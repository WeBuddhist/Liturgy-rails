༄༅། །གསོལ་འདེབས་མཚང་འདོན་མ་བཞུགས་སོ། །

རྒྱལ་ཀུན་འདུས་པའི་སྤྱི་གཟུགས་ནི། །
མཚན་ཉིད་ཀུན་ལྡན་བླ་མ་སྟེ། །
རང་སེམས་ཆོས་སྐུར་ངོ་སྤྲོད་པ། །
བདག་ལ་སངས་རྒྱས་ཀུན་ལས་ལྷག །

རེ་ས་ཁྱེད་ལས་མེད་པ་ལ། །
ཐུགས་རྗེའི་ཀློང་ནས་འདོར་ཕོད་དམ། །
བྱིས་དང་སྡིག་གྲོགས་འདུ་འཛིའི་ཁྲོད། །
ཆོས་མིན་ཆོས་ལྟར་བཅོས་པ་ཡི། །
སྐལ་པ་ངན་པ་བདག་ཅག་གི །
རྒྱུད་ལ་ངེས་འབྱུང་སྐྱོ་ཤས་དང་། །
བྱང་ཆུབ་སེམས་གཉིས་བསྐྱེད་བྱིན་རློབས། །

སྔོན་ཆད་ཚུལ་ཁྲིམས་གོས་ལ་རེ། །
དཀོན་མཆོག་དཀོར་ལ་བག་མེད་སྤྱད། །
ལྟར་སྣང་ཆོས་བརྒྱད་ཆོས་དང་འཁྲུལ། །
ད་ནི་འཆི་བ་སྙིང་དྲན་གྱི། །
རི་ཁྲོད་ནགས་ཁྲོད་དབེན་པ་རུ། །
གཏན་གྱི་མདུན་མ་སྒྲུབ་བྱིན་རློབས། །

ཆོས་ཀྱི་གེགས་ནི་ཅིར་ཡང་བརྫུ། །
གཉེན་བཤེས་འདུ་ལོང་ཟང་ཟིང་དང་། །
རང་དོན་ཤོར་བའི་གཞན་དོན་དང་། །
འགྲན་སེམས་ཕྲག་དོག་སྙན་གྲགས་བསྒྲུབ། །

དབེན་པ་བརྟེན་ཀྱང་ཞི་བདེར་གཞོལ། །
སྒོ་གསུམ་ཐ་མར་ཤོར་བའི་ཚེ། །
རང་སྐྱོན་རང་གིས་རིག་གྱུརནས། །
བླ་མའི་ཞལ་གདམས་དྲན་པར་ཤོག །

དྲན་ཤེས་མེད་པའི་དགེ་སྦྱོར་དང་། །
ལྟ་བས་མ་ཟིན་བསྐྱེད་རིམ་དང་། །
གོ་ཡུལ་ཤོར་བའི་སྟོང་ལྟ་སོགས། །
སངས་རྒྱས་ལམ་ལ་ཐག་རིང་བས། །

བླ་མ་མཆོག་ལ་མོས་གུས་དང་། །
ཟོལ་འགྲོགས་མེད་པའི་དད་པ་དང་། །
སེམས་ཅན་ཕ་མར་ཤེས་པ་ཡི། །
སྙིང་རྗེ་རྒྱུན་ཆད་མེད་པར་ཤོག །

ཆོས་ཕུང་བརྒྱད་ཁྲི་བཞི་སྟོང་གི །
རྩ་བ་བྱང་ཆུབ་སེམས་ལ་འདུ། །
དེ་ཕྱིར་གཞན་ལ་ཕན་སྲིད་ན། །
སྲོག་ཀྱང་གཏོང་བའི་སྙིང་སྟོབས་དང་། །
དེ་ལའང་བྱས་ཡུས་མི་སྐྱེ་བར། །
རང་བཞིན་མེད་པའི་སྟོང་ཉིད་ཀྱི། །
དྲན་པའི་རྩི་དང་མི་འབྲལ་ཤོག། །

སེམས་ཅན་ཀུན་ལ་བུ་གཅིག་ལྟར། །
བརྩེ་ཞིང་མཛད་པ་ཕྲིན་ལས་ཀུན། །
འགྲོ་ལ་ཕན་པ་ལྷུར་མཛད་པའི། །
སྐྱེས་བུ་དམ་པ་ཐམས་ཅད་དང་།།
བདག་ཉིད་སྨིན་སྒྲོལ་ལམ་བཀོད་པའི། །
བཀའ་དྲིན་ཅན་གྱི་བླ་མ་མཆོག །

ཞབས་པད་བསྐལ་བརྒྱར་བརྟན་པ་དང་། །
བདག་གི་མོས་གུས་ལྕགས་ཀྱུ་ཡིས། །
བླ་མ་དམ་པའི་ལུང་རྟོགས་ཀྱི། །
དགོངས་སྤྱོད་སྙིང་དབུས་སྤྱན་འདྲོངས་ནས། །
ཐུགས་ཡིད་གཅིག་ཏུ་འདྲེས་གྱུར་ཏེ། །
ཁམས་གསུམ་འཁོར་བའི་རྒྱ་མཚོ་འདི། །
ཅིག་ཅར་དོང་ནས་སྤྲུགས་ནུས་ཤོག །
