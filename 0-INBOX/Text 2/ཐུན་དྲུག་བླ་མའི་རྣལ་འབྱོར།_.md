ཐུན་དྲུག་བླ་མའི་རྣལ་འབྱོར་བཞུགས་སོ།། 

༑ན་མོ་གུ་རུ་མཉྫུ་གྷོ་ཥཱ་ཡ། 

རྡོ་རྗེ་འཆང་དབང་དཔལ་ལྡན་བླ་མ་ཡི། །

ཞབས་ཀྱི་པདྨ་གུས་པས་ཕྱག་བྱས་ཏེ། །

དམ་ཚིག་སྡོམ་པ་དག་བྱེད་རྨད་བྱུང་ཐབས། །

རྒྱུད་དང་རྗེ་བཙུན་བླ་མའི་གསུང་བཞིན་བཤད། །

(དེ་ཡང་མདོ་སྡེ་གོང་མ་གཉིས་ཀྱི་དཀྱིལ་འཁོར་ཆེན་པོར་དབང་བསྐུར་བ་ལེགས་པར་ཐོབ་པའི་གང་ཟག་རང་རང་གི་ཇི་སྲིད་སྔགས་སྡོམ་རྒྱུད་ལ་ལྡན་པ་དེ་སྲིད་དུ། རྩ་བ་དང་ཡན་ལག་གི་དམ་ཚིག་དང་སྡོམ་པ་རྣམས་ལ་ཉིན་ཞག་རེ་རེའི་ནང་སྔོ་བ་གྲང་བྱེད་ཅིང་། ཁྱད་པར་རིགས་ལྔའི་དམ་ཚིག་ཐུན་དྲུག་ཏུ་བསྐྱང་དགོས་ཏེ། དེ་ལྟར་མ་བྱས་ན་ཀ་ན་མ་ཐོ་བ་སྦོམ་པོར་འགྱུར་བའི་ཕྱིར་ཏེ། རྩེ་མོ་ལས། ཉིན་དང་མཚན་མོ་ལན་གསུམ་དུ། །ཉིན་རེ་བཞིན་ནི་འདའ་བྱ། །གང་ཚེ་ཉམས་གྱུར་རྣལ་འབྱོར་པ། །ཁ་ན་མ་ཐོ་སྦོམ་པོར་འགྱུུར། །ཞེས་གསུངས་པའི་ཕྱིར། དེ་ནས་བྱང་ཆུབ་སེམས་པ་གསང་སྔགས་ཀྱི་སྒོ་ནས་སྤྱད་པ་སྤྱོད་པར་འདོད་པ་རྣམས་ཀྱིས་ནི། བསྟེན་པ་དང་བཟའ་བ་དང་།བསྲུང་བའི་དམ་ཚིག་རྣམས་རྡོ་རྗེ་འཆང་ཆེན་པོས་གསུངས་པ་ལྟར་བསྟེན་དགོས་པས། དེ་དག་གི་ཚུལ་རྒྱས་པར། རྗེ་ཐམས་ཅད་མཁྱེན་པས་བཀའ་སྩལ་པའི་རྩ་ལྟུང་རྣམ་བཤད་དང་། བླ་མ་ལྔ་བཅུ་རྣམ་བཤད་དང་། རྣལ་འབྱོར་རོལ་པའི་དགའ་སྟོན། ཐུན་དྲུག་གི་རྣལ་འབྱོར་སོགས་སུ་བལྟ་བར་བྱ་ཞིང་། འདིར་ནི་དེ་དག་གི་དོན་ལས་དང་པོ་བ་རྣམས་ཀྱིས་ཐུན་དྲུག་ཏུ་ཉམས་སུ་ལེན་ཚུལ་འཁྱེར་བདེ་བའི་ཕྱིར་ཚིགས་སུ་བཅད་པའི་སྒོ་ནས་བརྗོད་པར་བྱའོ།)

སངས་རྒྱས་ཆོས་དང་ཚོགས་ཀྱི་མཆོག་རྣམས་ལ། །

བྱང་ཆུབ་བར་དུ་བདག་ནི་སྐྱབས་སུ་མཆི། །

བདག་གིས་སྦྱིན་སོགས་བགྱིས་པའི་ཚོགས་རྣམས་ཀྱིས། །

འགྲོ་བ་ཕན་ཕྱིར་སངས་རྒྱས་འགྲུབ་པར་ཤོག །(ཅེས་པ་ནི་སྐྱབས་འགྲོའི་བསླབ་བྱ་ཉིན་ལན་གསུམ་མཚན་ལན་གསུམ་དུ་སྐྱབས་སུ་འགྲོ་ཚུལ་དང་སེམས་བསྐྱེད་དུས་དྲུག་ཏུ་གཟུང་ཚུལ་ལོ།)

སེམས་ཅན་ཐམས་ཅད་ཉེ་རིང་ཆགས་སྡང་དང་། །བྲལ་(པར་བྱའོ་སྙམ་དུ་བཏང་སྙོམས་དང་)ནས་ཁྱད་པར་འཕགས་པའི་བདེ་བ་ཐོབ། །(པར་གྱུར་ཅིག་སྙམ་དུ་བྱམས་པ་དང་) བཟོད་དཀའི་སྡུག་བསྔལ་རྒྱ་མཚོ་ལས་བསྒྲལ་(བར་བྱའོ་སྙམ་དུ་སྙིང་རྗེ་དང་)ཏེ། །ཐར་པ་དམ་པའི་བདེ་དང་བྲལ་མ་གྱུར། (ཅིག་སྙམ་དུ་དགའ་བའོ།།ཞེས་པ་ནི་ཚད་མེད་བཞི་བསྒོམ་ཚུལ་ཏེ་རིགས་ལྔའི་དམ་ཚིག་གི་ནང་ནས་བྱམས་པ་དང་མི་འཇིགས་པའི་སྦྱིན་པ་བྱ་ཚུལ་རིན་འབྱུང་གི་དམ་ཚིག་བསྲུང་ཚུལ་གཉིས་འདིས་ཚང་། )

འགྲོ་ཀུན་སྲིད་ཞིའི་འཇིགས་པ་ལས་སྒྲོལ་ཕྱིར། །

རྫོགས་པའི་བྱང་ཆུབ་ཐོབ་པར་འདོད་པའི་སེམས། །

དེང་ནས་བཟུང་སྟེ་སངས་རྒྱས་མ་ཐོབ་བར། །

སྲོག་གི་ཕྱིར་ཡང་མི་བཏང་བཟུང་བར་བགྱི། །(སྙམ་དུ་སྨོན་སེམས་ཆོ་གས་གཟུང་ཕྱིར་ཞག་རེ་ཐུན་དྲུག་ཏུ་བཟུང་དགོས་པར་གསུངས་པའི་བསླབ་བྱའོ།མཇུག་སྡོམ་ལེན་འདོད་ན་མདུན་དུ་སྟོན་པ་འཁོར་བཅས་བཞུགས་པར་དགོགས་ནས)

ཇི་ལྟར་སྔོན་གྱི་བདེ་གཤེགས་ཀྱིས། །

བྱང་ཆུབ་ཐུགས་ནི་བསྐྱེད་པ་དང་། །

བྱང་ཆུབ་སེམས་དཔའི་བསླབ་པ་ལ། །

དེ་དག་རིམ་བཞིན་གནས་པ་ལྟར། །

བདག་ཀྱང་འགྲོ་ལ་ཕན་དོན་དུ། །

བྱང་ཆུབ་སེམས་ནི་བསྐྱེད་བགྱི་ཞིང་། །

བྱང་ཆུམ་སེམས་དཔའི་བསླབ་པ་ལ། །

རིམ་པ་བཞིན་དུ་བསླབ་པར་བགྱི། །(ཞེས་སོགས་གསུམ་བརྗོད་བྱང་སྡོམ་བླང་)

དེང་དུས་བདག་ཚེ་འབྲས་བུ་ཡོད། །

མི་ཡི་སྲིད་པ་ལེགས་པར་ཐོབ། །

དེ་རིང་སངས་རྒྱས་རིགས་སུ་སྐྱེས། །

སངས་རྒྱས་སྲས་སུ་ད་གྱུར་ཏོ། །

ད་ནི་བདག་གིས་ཅི་ནས་ཀྱང་། །

རིགས་དང་མཐུན་པའི་ལས་བརྩམས་ཏེ། །

སྐྱོན་མེད་བཙུན་པའི་རིགས་འདི་ལ། །

རྙོག་པར་མི་འགྱུར་དེ་ལྟར་བྱ། །(ཞེས་ཕན་ཡོན་བསམས་ཏེ་སྤྲོ་བ་བསྐྱེད། འདི་ནི་སེམས་བསྐྱེད་རྒྱས་པར་བྱ་བའི་ཕྱིར་དགའ་བ་བསྒོམས་པས་བྱང་ཆུབ་ཀྱི་སེམས་ཀྱི་ཕན་ཡོན་ཐུན་དྲུག་ཏུ་བསམ་ཚུལ་ལོ།)

མདུན་གྱི་ནམ་མཁར་ཡིད་འཕྲོག་རིན་ཆེན་ཁྲི། །

ཆུ་སྐྱེས་ཉི་ཟླ་རྒྱས་པའི་དཀྱིལ་འཁོར་སྟེང་། །

རྩ་བའི་བླ་མ་ཁྱབ་བདག་རྡོ་རྗེ་འཆང་། །

སྐུ་མདོག་སྔོན་པོ་ཞལ་གཅིག་ཕྱག་གཉིས་ཀྱིས། །

རྡོར་དྲིལ་བཟུང་ནས་རང་འདྲའི་ཡུམ་ལ་འཁྱུད། འ

མཚན་དཔེའི་གཟི་འབར་རིན་ཆེན་རྒྱན་མང་འཛིན། །

ཡིད་འཕྲོག་ལྷ་རྫས་གོས་ཀྱི་ན་བཟའ་ཅན། །

དྲན་པ་ཙམ་གྱིས་གདུང་བ་ཀུན་སེལ་ཅིང་། །

སྐྱབས་མཆོག་མ་ལུས་འདུས་པའི་བདག་ཉིད་དུ། །

སྐྱིལ་ཀྲུང་ཚུལ་གྱིས་བཞུགས་པའི་གནས་གསུམ་དུ། །

ཡི་གེ་གསུམ་མཚན་ཧཱུཾ་གི་འོད་ཟེར་གྱིས། །

རང་བཞིན་གནས་ནས་བླ་མ་རྡོ་རྗེ་འཆང་། །

ཛཿཧཱུཾ་བཾ་ཧོཿགཉིས་སུ་མེད་པར་གྱུར། །(འདི་ནི་མི་བསྐྱོད་པའི་རིགས་ཀྱི་དམ་ཚིག་སློབ་དཔོན་གཟུང་ཚུལ་ལོ།)

གང་གི་དྲིན་གྱིས་བདེ་ཆེན་ཉིད། །

སྐད་ཅིག་ཉིད་ལ་འཆར་བ་གང་། །

གང་འདུལ་ཇི་བཞིན་ངུར་སྨྲིག་རྣམ་རོལ་གྱིས། །

བླ་མ་རིན་པོ་ཆེ་ལ་གསོལ་བ་འདེབས། །

རབ་འབྱམས་རྒྱལ་བའི་འཁོར་ལོ་མཐའ་དག་ལས། །

ལྷག་པའི་ཞིང་གི་དམ་པར་ལེགས་བསྔགས་པའི། །

བླ་མ་རིན་པོ་ཆེ་ལ་གསོལ་བ་འདེབས། །

མཆོག་དང་ཐུན་མོང་དངོས་གྲུབ་མ་ལུས་ཀུན། །

མགོན་ཁྱོད་ཚུལ་བཞིན་བསྟེན་པའི་རྗེས་འགྲོ་བར། །

མཐོང་ནས་ལུས་དང་སྲོག་ཀྱང་ཡོངས་བཏང་སྟེ། །

ཁྱོད་མཉེས་ཁོ་ན་བསྒྲུབ་པར་བྱིན་གྱིས་རློབས། །(སྙམ་དུ་སེམས་པ་ནི་དགེ་བའི་བཤེས་མཉེན་ལ་དད་གུས་སྒོམ་ཚུལ་གྱི་གནད་བསྡུས་པའོ།)

བདག་ཉིད་རྡོ་རྗེ་སེམས་དཔའི་ང་རྒྱལ་གྱིས། །

ལྷན་སྐྱེས་བདེ་ཆེན་གསང་བའི་རྡོ་རྗེ་དང་།།

རང་བཞིན་སྤྲོས་བྲལ་གསང་བའི་གྷཎ་ྜཡི།།

མཚོན་བྱེད་བཟུང་ནས་བཅོམ་ལྡན་འདས་མར་འཁྱུད། །(སེམས་པ་འདི་གསུམ་ནི་རྡོར་དྲིལ་ཕྱག་རྒྱའི་དམ་ཚིག་བསྐྱང་ཚུལ་བླ་མེད་ལྟར་རོ།)

བདག་གི་ལུས་དང་དེ་བཞིན་ལོངས་སྤྱོངས་དང་། །

དུས་གསུམ་དགེ་ཚོགས་ཕུང་པོ་ཅི་མཆིས་པ། །

མ་གྱུར་སེམས་ཅན་ཀུན་གྱི་དོན་གྱི་ཕྱིར། །

དེང་ནས་ཕངས་པ་མེད་པར་བཏང་བར་བྱ། །(སྙམ་དུ་གཏོང་སེམས་སྤེལ་བ་ནི།རྒྱལ་སྲས་ཀྱི་སྦྱིན་པའི་ཉམས་ལེན་དང་རིན་འབྱུང་གི་དམ་ཚིག་ཟང་ཟིང་དང་ཆོས་སྦྱིན་བྱ་ཚུལ་ལོ། ལོང་བྱུང་ན་སོ་སོའི་མ་མཐའ་སྡོམ་གྱི་སྒོ་ནས་དམ་ཚིག་དང་སྡོམ་པ་རྣམས་ལ་བསྔོ་བགྲང་བྱེད་ཅིང་མ་ལྕོགས་ན་འདི་ཙམ་གྱིས་ཀྱང་སྡོམ་པ་གསུམ་ཀ་ལ་དྲན་ཤེས་བསྟེན་ཚུལ་དུ་འགྱུར་རོ།)

སོ་ཐར་བྱང་སེམས་རྡོ་རྗེ་ཐེག་པ་ཡི། །

རྣམ་དག་བཅས་པའི་བསླབ་པ་ཕྲ་མོ་ལས། །

རྨི་ལམ་དུ་ཡང་འདའ་བར་མི་བྱེད་པར། །

རྒྱལ་བའི་བཀའ་བཞིན་བདག་གིས་བསྒྲུབ་པར་བྱ། །(སྙམ་པ་ནི་སྔགས་སྡོམ་བཟུང་བས་རྩ་བ་དང་ཡན་ལག་གི་དམ་ཚིག་དང་སྡོམ་པ་ལ་བསྔོ་བགྲང་ཐུན་དྲུག་ཏུ་བྱེད་ཚུལ་ལོ།)

རྒྱལ་བའི་དགོངས་པ་ཇི་བཞིན་ལེགས་བཟུང་སྟེ། །(སྙམ་པ་ནི་པདྨའི་རིགས་ཀྱི་དམ་ཚིག་བསྐྱང་ཚུལ་ལོ།)

གང་འཚམས་ཐབས་ཀྱིས་འགྲོ་བ་ཡོངས་སུ་སྒྲོལ། །(སྙམ་པ་ནི་རྣམ་སྣང་གི་རིགས་ཀྱི་དམ་ཚིག་སེམས་ཅན་དོན་བྱེད་ཀྱི་ཚུལ་ཁྲིམས་བསྐྱང་ཚུལ་ལོ། མཇུག་ཏུ་བཟང་པོ་སྤྱོད་པ་སོགས་ཀྱི་སྨོན་ལམ་བྱའི།)

བཅས་པའི་མཚམས་ལས་ནམ་ཡང་མི་འདའ་བ། །

རིམ་གཉིས་ལམ་གྱི་རིམ་པ་མཐར་ཕྱིན་ཤོག ། །

ས་དང་ལམ་གྱི་ཡོན་ཏན་རབ་རྫོགས་ནས། །

རྡོ་རྗེ་འཆང་གི་གོ་འཕང་མྱུར་ཐོབ་ཤོག

(ཅེས་པ་འདི་ནི་རང་གི་སློབ་མ་སྡེ་སྣོད་འཛིན་པ་དུ་མས་བསྐུལ་བ་དང་། ཁྱད་པར་ཉིན་ཞག་ཕྲུགས་རེ་ལ་ཐུན་དྲུག་ཏུ་འདི་ལྟ་བུའི་ཉམས་ལེན་གྱི་སྡོམ་ཆེན་པོ་ཉམས་སུ་ལེན་པ་བྱུང་ན། བོད་ཀྱི་གསེར་ཆོས་སུ་གྲགས་པའི་ཟབ་ཟབ་མོ་རྣམས་ཀྱིས་ཆ་ཙམ་དུའང་བསྒྲུན་དུ་མེད་པའི་རྒྱལ་བ་དགྱེས་པའི་བཀའ་སྲོལ་དུ་རྗེ་ཡབ་སྲས་རྣམས་ཀྱིས་བསྔགས་ཤིང་། རྒྱུད་སྡེ་དང་མཐུན་པའི་ཉམས་ལེན་དུ་ཆོས་ཀྱི་རྗེས་སུ་འབྲང་པའི་བློས་རྟོགས་པའི་ཉེར་ལེན་གྱིས་བསྐུལ་བ་ལ་བརྟེན་ནས་རིག་པ་འཛིན་པ་ཆོས་ཀྱི་རྒྱལ་མཚན་གྱིས་སྦྱར་བའོ།།)
