བློ་ ཡིས་བླངས་ནས་བླ་མ་ཡི་དམ་དཀོན་མཆོག་གསམ་ལ་འབུལ།།

ཐུགས་རྗེ་དབང་གིས་བཞེས་ནས་བདག་ལ་བྱིན་གྱིས་བརླབ་ཏུ་གསོལ། །ཨི་དཾ་གུ་རུ་རཏྣ་མཎྜལ་ཀཾ་ནིརྱ་ཏཱ་ཡ་མི། །(དེ་ནས་བླ་ གོས་ཕྲག་པ་གཅིག་ཏུ་གཟར། པུས་མོ་གཡས་པའི་ལྷ་ང་ས་ལ་བཙུགས། ལག་ པ་གཉིས་ཐལ་མོ་སྦྱར་ཏེ་བསམ་པ་འདི་སྙམ་དུ། ག་རེ། བདག་སོགས་ནམ་མཁའི་ མཐའ་དང་མཉམ་པའི་སེམས་ཅན་ཐམས་ཅད་ཐོག་མ་མེད་པ་ནས་ད་ལྟའི་བར་དུ་ སྤྱིར་འཁོར་བ་དང་ཁྱད་པར་ངན་སོང་གསུམ་གྱི་སྡུག་བསྔལ་ཡོད་རྒུ་ཞིག་གྲངས་ མེད་པར་མྱོང་དུ་ཟིན་ཀྱང་། ད་དུང་སྐྱོ་བ་དང་སུན་པའི་བློ་སྐད་ཅིག་ཀྱང་མི་སྐྱེ་བར། སྡུག་བསྔལ་ལ་བདེ་བ། བདག་མེད་ལ་བདག་ཏུ་འཛིན་པའི་བློ་ངན་ལང་ཤོར་འདིའི་ མཐུས་ལས་དང་ཉོན་མོངས་པའི་གཞན་དབང་དུ་སོང་ནས། སླར་ཡང་འཁོར་བ་ དང་ངན་སོང་གི་སྡུག་བསྔལ་བཟོད་དཀའ་བ་སྔར་ལས་ཀྱང་ཆེས་ཆེར་མང་བ་ཞིག་ རང་དབང་མེད་པར་མྱོང་དགོས་པའི་ངང་ཚུལ་འདི་བསམ་ན་ངེས་པར་སྙིང་རླུང་ལྡང་ འོས་པའི་ཚོད་ཙམ་དུ་འདུག །སྔོན་བདག་ཅག་དང་མཚུངས་པར་འཁོར་བར་གནས་ པ་ལས་དགེ་བའི་བཤེས་གཉེན་རེ་དང་མཇལ་བའི་དྲིན་ལས་འཁོར་བའི་ཡིད་འབྱུང་ ཞིང་། སེམས་ཅན་ཀུན་ལའང་བརྩེ་བ་སྐྱེ་ནས་བྱང་ཆུབ་ཏུ་ཐུགས་བསྐྱེད་དེ། འདི་ ལྟ་བུའི་ཐེག་པ་ཆེན་པོའི་གསོ་སྦྱོང་གི་སྡོམ་པ་བཞེས་ནས་ཡང་དག་པར་བསྲུངས་ཤིང་ ལམ་ལ་བསླབས་ནས་སངས་རྒྱས་པ་ཡིན། དེ་བཞིན་བདག་གིས་ཀྱང་དགེ་བའི་བཤེས་ གཉེན་གྱི་སྐུ་དྲིན་ལས་ཐེག་པ་ཆེན་པོའི་ཆོས་དང་མཇལ་བའི་སྐབས་འདིར་བྱང་ཆུབ་ ཏུ་སེམས་བསྐྱེད་ནས་ལམ་ལ་བསླབ་ན་ངེས་པར་སངས་རྒྱས་ཀྱི་གོ་འཕང་ཐོབ་ནུས། ་ དེ་ལྟར་ནའང་། གཞི་དང་ལམ་དང་འབྲས་བུའི་གནས་སྐབས་ཐམས་ཅད་དུ་མེད་ དུ་མི་རུང་བའི་མཛའ་བཤེས་ཀྱི་གཙོ་བོ། འདི་ཕྱི་ལེགས་ཚོགས་ཐམས་ཅད་གང་ ལས་འབྱུང་བའི་ཡིད་བཞིན་ནོར་བུ། ཐོག་མ་མེད་པ་ནས་འཁོར་བའི་ཕྱི་མཐའ་མ་ རྫོགས་ཀྱི་བར་དུ་དྲིན་གྱིས་བསྐྱངས་ཤིང་སྐྱོང་བར་འགྱུར་བའི་མཁའ་ཁྱབ་ཀྱི་མ་རྒན་ ཉམ་ཐག་པ་འདི་དག་ཡལ་བར་དོར་ནས་རང་ཉིད་གཅིག་པུའི་བདེ་ཐབས་ཙམ་ལྷུར་ ལེན་པ་ནི་རྨོངས་པ་མི་ཟད་དུད་འགྲོ་རྣམས་དང་ཁྱད་པར་ཅུང་ཟད་ཀྱང་མི་འདུག། དེས་ན་བདག་གིས་མ་ནམ་མཁའ་དང་མཉམ་པའི་སེམས་ཅན་ཐམས་ཅད་ཀྱི་དོན་ དུ་ཡང་དག་པར་རྫོགས་པའི་སངས་རྒྱས་ཀྱི་གོ་འཕང་རིན་པོ་ཆེ་ཅི་ནས་ཀྱང་ཐོབ་ པར་བྱ། དེའི་ཕྱིར་དུ་རྒྱལ་བ་སྲས་དང་བཅས་པ་འདི་རྣམས་ཀྱི་སྤྱན་སྔར་ཐེག་པ་ ཆེན་པོའི་གསོ་སྦྱོང་གི་སྡོམ་པ་བླངས་ལ། སང་ཉི་མ་མ་ཤར་གྱི་བར་དུ་ལེགས་པར་ བསྲུང་བར་བྱའོ་སྙམ་པའི་བསམ་པ་མིག་ནས་མཆི་མ་འཁྲུགས་པ་དང་། ལུས་ཀྱི་ བ་སྤུ་ལྡང་བའི་ཚོད་ཙམ་གྱི་དུང་དུང་བ་བཅས་དང་། བླ་མ་ཐུབ་པའི་དབང་པོའི་ རྗེས་སུ་བཟླས་པར་མོས་ལ།)

དེ་ལྟར་སྡོམ་པ་བླངས་ཟིན་ནས་ ཉིན་ཞག་གཅིག་གི་བར་དུ་བསྲུང་རྒྱུའི་ཡན་ལག་གི་གྲངས་ནི། མཛོད་ལས། ཚུལ་ ཁྲིམས་ཡན་ལག་བག་ཡོད་པས། །ཡན་ལག་བརྟུལ་ཞུགས་ཡན་ལག་སྟེ། །བཞི་ གཅིག་དེ་བཞིན་གསུམ་རིམ་བཞིན། ཞེས་གསུངས་པ་ལྟར། །སོ་ཐར་བསྙེན་གནས་ ཀྱི་སྐབས་བསམ་བཞིན་དུ་སྲོག་ཆགས་གྲོག་སྦུར་ཡན་ཆད་ཀྱི་སྲོག་གཅོད་ པ་དང་། གཞན་གྱིས་བདག་ཏུ་བཟུང་བའི་རྫས་ཤིང་གི་འདབ་མ་ཡན་ཆད་ མ་བྱིན་པར་ལེན་པ་དང་། མི་ཚངས་པར་སྤྱོད་པ་དང་། རྫུན་ཚིག་སྟེ་ བཞི་སྤོང་བའི་ཡན་ལག་བཞི་དང་། མྱོས་འགྱུར་གྱི་རིགས་མཐའ་དག་སྤོང་ བའི་ཡན་ལག་དང་། དཀོན་མཆོག་མཆོད་པ་སོགས་ཆོས་ལྡན་གྱི་དགོས་ པ་མེད་པར་རྒོད་བག་ལ་སོགས་པའི་ཟློས་གར་དང་གླུ་དང་། རོལ་མོའི་སྒྲ་སོགས་ བྱེད་པ་དང་། རྒོད་བག་ཆགས་པ་ལ་སོགས་པའི་བསམ་པས་རྒྱུན་དུ་འདོགས་བཞིན་ མ་ཡིན་པའི་འཕྲེང་བ་ང་ང་རྒྱན་འདོགས་པ་དང་། ལུས་ལ་དྲིས་བྱུག་པ་སོགས་སྤོང་ བ་གར་སོགས་འཕྲེང་སོགས་སྤོང་བའི་ཡན་ལག་དང་། དགོས་མེད་དུ་རིན་ པོ་ཆེ་ལ་སོགས་པ་རིན་ཐང་ཅན་གྱི་ཁྲི་དང་སྟན་ཆེན་པོ་དང་། ཤིང་ལ་སོགས་པ་ དངོས་པོ་དམན་པའི་ཁྲི་སྟན་ཁྲུ་གང་ལོངས་པ་ཡན་ཆད་ཀྱི་ཁྲི་སྟན་མཐོན་པོ་ལ་འདུག་ པ་སྤོང་བ་ཁྲི་སྟན་ཆེ་མཐོ་སྤོང་བའི་ཡན་ལག་དང་། ཉི་མ་ཕྱེད་ཡོལ་བ་ནས་ ཞོ་འོ་མ་སོགས་དང་། འབྲས་ཆེན་དང་ཟན་ལ་སོགས་པའི་བཟའ་བཏུང་སྤོང་བ་དུས་ མ་ཡིན་པའི་ཁ་ཟས་སྤོང་བའི་ཡན་ལག་སྟེ་བརྒྱད་དང་། དེ་བརྒྱད་ཀྱི་དང་པོ་བཞི་ ཚུལ་ཁྲིམས་དང་། ལྔ་བ་བག་ཡོད་དང་། ལྷག་མ་གསུམ་བརྟུལ་ཞུགས་ཀྱི་ཡན་ ལག་ཏུ་འཇོག་པ་ཐེག་ཆེན་གསོ་སྦྱོང་གི་སྐབས་འདིར་ཡང་འདྲ་ལ། འོན་ཏེ་སྡོམ་ པ་འདི་དང་། བསྙེན་གནས་གཉིས་ཁྱད་མེད་པར་འགྱུར་རོ་སྙམ་ན། འདི་གཉིས་ ཁྱད་པར་ཤིན་ཏུ་ཆེ་སྟེ། ལེན་པའི་ཡུལ་མི་འདྲ། རྟེན་གྱི་ཁྱད་པར་མི་འདྲ། བསམ་ པའི་ཁྱད་པར་མི་འདྲ། ཇི་ལྟར་ལེན་པའི་ཆོ་ག་མི་འདྲ། བླངས་ནས་སྲུང་ཚུལ་མི་ འདྲ། མཐར་འབྲས་བུ་འགྲུབ་ཚུལ་མི་འདྲ་བའི་ཁྱད་པར་དུ་མ་ཡོད་དོ། །

ལེན་ པའི་ཡུལ་མི་འདྲ་བ་ནི། བསྙེན་གནས་ནི་དགེ་བསྙེན་ཡན་ཆད་སོ་ཐར་གྱི་སྡོམ་ ལྡན་གང་ཡང་རུང་བ་ཞིག་གི་དྲུང་དུ་བླངས་པས་སྐྱེ་ལ། ཐེག་ཆེན་གསོ་སྦྱོང་གི་སྡོམ་ པ་ལེན་ཡུལ་ནི། ཐེག་པ་ཆེན་པོའི་སྡོམ་ལྡན་ཞིག་དགོས་ཤིང་། རང་ཉིད་ཀྱིས་ རྟེན་གྱི་དྲུང་དུ་བླངས་ཀྱང་སྐྱེ་བས་སོ། །རྟེན་གྱི་ཁྱད་པར་མི་འདྲ་བ་ནི། བསྙེན་ གནས་ཀྱི་སྡོམ་པ་ནི་དགེ་ཚུལ་དང་དགེ་སློང་རྣམ་དག་གིས་ལེན་དུ་མི་རུང་སྟེ། དགེ་ ཚུལ་དང་། དགེ་སློང་གིས་བསྙེན་གནས་ཀྱི་སྡོམ་པ་བླངས་ནས་བདག་བསྙེན་གནས་ ཡན་ལག་བརྒྱད་པར་གཟུང་དུ་གསོལ་ཞེས་བསྙེན་གནས་པར་ཁས་བླངས་ན་སྡོམ་ པ་གོང་མ་གཏོང་བས་ཤིན་ཏུ་མི་རུང་ལ། ཐེག་པ་ཆེན་པོའི་གསོ་སྦྱོང་གི་སྡོམ་པ་ ནི། དགེ་སློང་རྣམ་དག་གིས་ལེན་དུ་རུང་བར་གསུངས་སོ། །བསམ་པའི་ཁྱད་ པར་མི་འདྲ་བ་ནི། བསྙེན་གནས་ཀྱི་སྡོམ་པ་ནི་ངེས་འབྱུང་གི་བསམ་པ་ཙམ་གྱིས་ བླངས་ན་སྐྱེ་ལ། ཐེག་པ་ཆེན་པོའི་གསོ་སྦྱོང་གི་སྡོམ་པ་ལ་ནི། ངེས་པར་བྱང་ཆུབ་ ཀྱི་སེམས་བཅོས་མ་ཚུན་ཆད་དུ་བསྐྱེད་པ་དགོས་པ་ནི་གོང་དུ་བཤད་ཟིན་པ་ལྟར་ རོ། །ཇི་ལྟར་ལེན་པའི་ཆོ་ག་མི་འདྲ་བ་ནི། བསྙེན་གནས་ཀྱི་སྡོམ་པ་ནི་འདུལ་ བ་ལུང་དང་། གནས་འཇོག་གི་མདོ་ནས་བཤད་པ་ལྟར་སྐྱབས་འགྲོ་རྩོམ་བྱེད་དུ་ བྱས་ནས་བསྙེན་གནས་ཡན་ལག་བརྒྱད་པར་ཁས་ལེན་ལན་གསུམ་བྱས་པས་ཐོབ་ པ་ཡིན་ལ། ཐེག་པ་ཆེན་པོའི་གསོ་སྦྱོང་གི་སྡོམ་པ་ནི། སངས་རྒྱས་རྣམས་ཀྱི་ལམ་ དུས་སུ་ཇི་ལྟར་མཛད་པ་དཔེར་བྱས་ཏེ། རང་ཉིད་ཀྱང་དེ་ལྟ་བུའི་རྣམ་ཐར་གྱི་རྗེས་ སུ་སློབ་པར་ཁས་ལེན་པ་ཡིན་པས་ཁྱད་པར་ཤིན་ཏུ་ཆེའོ། །

རིང་མིན་སྟོབས་བཅུ་མངའ་བར་སོན་གྱུར་ཅིག། (ཅེས་པ་འདི་ཡང་དད་དམ་མཚུངས་མེད་སྒེར་སྤེལ་གཞི་བའི་རྗེ་བཙུན་བློ་བཟང་ འཕྲིན་ལས་ཆོས་སྐྱིད་ནས་ཐེག་ཆེན་གསོ་སྦྱོང་གི་ཆོ་ག་པར་མ་དེང་སྐབས་ཡོད་པ་ རྣམས་ཕལ་ཆེར་འགོར་པདྨའི་རིགས་ཀྱི་དམ་རྒྱ་བཅིང་བ་དང་། ཨོཾ་སརྦ་ཏ་ཐཱ་ག་ཏ་སོགས་ཕྱག་འཚལ་ཞིང་ལུས་འབུལ་བ་སོགས་སྨྱུང་གནས་ལྟ་བུ་བྱ་རྒྱུད་ཀྱི་ཆོ་ག་དང་འབྲེལ་བ་ཤས་ཆེ་བས། དེ་དག་དང་མ་འབྲེལ་བའི་སྐབས་གསོ་སྦྱོང་རྐྱང་བ་ བྱ་ཚུལ་གྱི་ལག་ལེན་ནག་འགྲོས་སུ་འཁོད་པ་ཞིག་དགོས་ཞེས་རྟེན་བཅས་བསྐུལ་མ་མཛད་པ་བཞིན། ༸སྐུ་བཅར་མཚན་ཞབས་དར་ཧན་དགའ་ལྡན་ཁྲི་བྱང་སྤྲུལ་མིང་བས་ཤིང་སྤྲེལ་ཆོ་འཕྲུལ་ཟླ་བའི་ཡར་ཚེས་བཟང་པོར་དམ་པའི་ཕྱག་བཞེས་ལྟར་ཡི་གེར་བཀོད་པའོ།། །།
