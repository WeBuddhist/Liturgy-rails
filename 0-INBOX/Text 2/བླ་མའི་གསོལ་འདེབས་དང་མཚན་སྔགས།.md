བླ་མའི་གསོལ་འདེབས་དང་མཚན་སྔགས།

༄༅། །དེ་ནས། བློ་བཟང་རྒྱལ་བའི་དབང་པོ་ཙོང་ཁ་པ། །རྣམ་དཀར་འཕྲིན་ལས་ལྷུན་འགྲུབ་ཆོས་གྱི་རྗེ། །མདོ་སྔགས་བསྟན་པའི་རྒྱལ་མཚན་འཛིན་པའི་དཔལ། །སྐྱབས་མཆོག་དམ་པའི་ཞབས་ལ་གསོལ་བ་འདེབས། །ལན་གསུམ། ཨོཾ་ཨཱཿགུ་རུ་བཛྲ་དྷ་ར་སུ་མ་ཏི་ཀརྨ་ནི་རཱ་བྷོ་ག་དྷརྨ་དྷྭ་ཛ་ཤྲཱི་བྷ་དྲ་སརྦ་སིད་དྷི་ཧཱུཾ་ཧཱུྃ། ལན་བདུན། ངག་དབང་བློ་བཟང་རྒྱལ་བསྟན་སྙིང་པོའི་སྲོག །འཛིན་མཁས་རྒྱ་ཆེན་འཕྲིན་ལས་རོལ་བའི་མཚོ། །སྲིད་གསུམ་དབང་བསྒྱུར་མཚུངས་པ་མེད་པའི་རླབས། །རྣམ་པར་གྲོལ་བའི་སྡེ་ལ་གསོལ་བ་འདེབས། །ལན་གསུམ། ཨོཾ་ཨཱཿགུ་རུ་བཛྲ་དྷ་ར་ཝ་གིནྡྲ་སུ་མ་ཏི་ཤཱ་ས་ན་དྷ་ར་ས་མུ་དྲ་ཤྲཱི་བྷ་དྲ་སརྦ་སིད་དྷི་ཧཱུཾ་ཧཱུྃ། ཞེས་མཚན་སྔགས་ལན་བདུན་སོགས་རང་རང་གི་རྩ་བའི་བླ་མའི་གསོལ་འདེབས་དང་མཚན་སྔགས་བཟླས་ལ།

བདག་གི་རྗེ་བཙུན་བླ་མ་ཐུགས་རྗེ་ཅན། །ཉམ་ཐག་བདག་ལ་ཐུགས་རྗེས་གཟིགས་སུ་གསོལ། །དགྱེས་པའི་ཞལ་བཟང་བདག་ལ་བསྟན་དུ་གསོལ། །དབུགས་དབྱུང་གསུང་གི་གནང་བ་སྩལ་དུ་གསོལ། །རྩེ་གཅིག་སྙིང་ནས་གསོལ་བ་འདེབས་པའི་ཚེ། ། བདེ་ཆེན་ འཁོར་ ལོའི་ དབུས་ སུ་ ལེགས་བཞུགས་ནས། ། ཡིད་ལ་ བསམ་ པའི་ འདོད་ དོན་ མ་ ལུས་ པ།། གེགས་མེད་ལྷུན་གྱིས་འགྲུབ་པར་བྱིན་གྱིས་རློབས། ། ཚོགས་གཉིས་གསོག་དང་སྡིག་ལྟུང་བཤགས་པའི་ཚེ། ། མདུན་གྱི་ནམ་མཁའི་དབྱིངས་སུ་ལེགས་བཞུགས་ནས། ། རླབས་ཆེན་ཚོགས་གཉིས་ཡོངས་སུ་རྫོགས་པ་དང་། ། བཅས་རང་ཉེས་ལྟུང་བྱང་བར་བྱིན་གྱིས་རློབས། ། བྱང་ཆུབ་བར་གྱི་ཉིན་མཚན་དུས་ཀུན་ཏུ༑ ༑སྙིང་དབུས་ པདྨའི་ ཟེའུ་འབྲུར་ལེགས་ བཞུགས་ ནས།།བྱང་ཆུབ་སྒྲུབ་པའི་འགལ་རྐྱེན་ཀུན་ཞི་ཞིང་། ། མཐུན་རྐྱེན་བདེ་བླག་འགྲུབ་པར་བྱིན་གྱིས་རློབས། ། བདག་གི་བློ་སྣ་ཆོས་ལ་ཕྱོགས་པ་དང་། ། ཆོས་མིན་བསམ་སྦྱོར་ངན་པ་ཞི་བ་དང་། ། ཆོས་དང་མཐུན་པའི་ འདོད་ དོན་ མ་ ལུས་ པ། ། འབད་ མེད་ ལྷུན་ གྱིས་འགྲུབ་པར་བྱིན་གྱིས་རློབས། ། མདོར་ན་དེང་ནས་ཚེ་ རབས་ཐམས་ཅད་དང་། ། འདི་ཕྱི་བར་དོའི་གནས་སྐབས་ཐམས་ཅད་དུ ༑ ༑ རེ་སའི་སྐྱབས་གནས་བླ་མ་ཁྱེད་ལས་མེད། ། ཐུགས་རྗེའི་སྤྱན་གྱིས་ བདག་ལ་ གཟིགས་སུ་གསོལ། ། མགོན་སྐྱབས་ བྲལ་བའི་ཉམ་ཐག་བདག་ཅག་ལ། ། རྗེ་བཙུན་བླ་མའི་ཐུགས་རྗེས་རབ་དགོངས་ནས། ། སྐུ་དང་གསུང་གི་སྣང་བ་འབྲལ་མེད་དུ། ཉམས་སུ་མྱོང་བའི་སྐལ་བཟང་ཐོབ་པར་ཤོག།སྐྱེ་བ་ཀུན་ ཏུ་ ཡང་དག་བླ་མ་དང་། ། འབྲལ་མེད་ཆོས་ཀྱི་དཔལ་ལ་ལོངས་སྤྱོད་ཅིང་།ས་དང་ལམ་གྱི་ཡོན་ཏན་རབ་རྫོགས་ནས།།རྡོ་རྗེ་འཆང་གི་གོ་འཕང་མྱུར་ཐོབ་ཤོག།ཅེས་བརྗོད་པ་ནི།རྒྱལ་བ་རྒོད་ཚང་པས།བསྐྱེད་རིམ་སྒོམ་པ་མང་ལགས་ཏེ།།བླ་མ་སྒོམ་པའི་གོང་ན་མེད།།བཟླས་བརྗོད་བྱེད་པ་མང་ལགས་ཏེ། ། གསོལ་བ་འདེབས་པའི་གོང་ན་མེད།།རྒྱུན་དུ་གསོལ་བ་བཏབ་པ་ན༑ ༑འབྲལ་མེད་ཉམས་མྱོང་འོང་པར་ངེས།།ཞེས་གསུངས་པ་ལྟར།བདེ་ལེགས་ཐམས་ཅད་ཀྱི་རྩ་བ་བླ་མ་ལ་གསོལ་བ་འདེབས་པའོ། །
