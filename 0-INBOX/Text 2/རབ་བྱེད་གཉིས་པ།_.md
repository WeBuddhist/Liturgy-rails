རབ་བྱེད་གཉིས་པ། ཡ་རབས་བརྟགས་པ།

།རྟག་ཏུ་དམ་པའི་ཡོན་ཏན་རྣམས། །སྐྱེ་བོ་དམ་པས་ལྷག་པར་སྒྲོགས། །མ་ལ་ཡ་ཡི་ཙན་དན་དྲི། །རླུང་གིས་ཕྱོགས་བཅུར་རྒྱས་པར་བྱེད། །དམ་པ་དཔོན་དུ་བསྐོས་གྱུར་ན། །དོན་གྲུབ་པ་དང་བདེ་སྐྱིད་འཐོབ། །ནོར་བུ་རྒྱལ་མཚན་རྩེར་མཆོད་ན། །ཡུལ་ཕྱོགས་དགེ་ཞེས་མཁས་རྣམས་སྒྲོགས། །རྒྱལ་ངན་གཞན་གྱིས་གཙེས་པ་ན། །ལྷག་པར་ཆོས་རྒྱལ་དྲན་པར་འགྱུར། །རིམས་ཀྱིས་བཏབ་པའི་སེམས་ཅན་རྣམས། །གངས་ཆུ་འབའ་ཞིག་ཡིད་ལ་བྱེད། །སྡིག་སྤྱོད་རྒྱལ་པོས་གཙེས་པ་ལ། །མི་དབང་ཆོས་རྒྱལ་ལྷག་པར་སྐྱོང༌། །འབྱུང་པོའི་གདོན་གྱིས་བཏབ་པ་ལ། །གསང་སྔགས་གྲུབ་པས་རྗེས་སུ་འཛིན། །དམ་པ་སྡིག་པ་ཆུང་ཡང་སྤོང༌། །དམན་རྣམས་ཆེན་པོའང་དེ་ལྟ་མིན། །ཞོ་ལ་རྡུལ་ཕྲན་འབྱར་བ་སེལ། །ཆང་ལ་ཕབ་ཀྱང་ལྷག་པར་འདེབས། །སྐྱེ་བོ་དམ་པ་རྒུད་གྱུར་ཀྱང༌། །སྤྱོད་པའི་ཁྱད་པར་ལྷག་པར་མཛེས། །མེ་ནི་ཐུར་དུ་ཁ་བསྟན་ཀྱང༌། །མེ་ལྕེ་གྱེན་དུ་འབར་བ་མཐོང༌། །དམ་པ་རྒྱང་ན་གནས་ན་ཡང༌། །འཁོར་འདབས་ཕན་པས་རིང་ནས་སྐྱོང༌། །མཁའ་ལ་སྤྲིན་ཆེན་འཁྲིགས་པ་ཡིས། །ས་ཡི་ལོ་ཏོག་ཁྱད་པར་འཕེལ། །གསོན་ཚེ་སྙན་གྲགས་དགའ་བའི་རྒྱུ། །འཇིག་རྟེན་གཞན་དུ་བསོད་ནམས་དགའ། །དེ་གཉིས་མེད་པའི་ནོར་ཙམ་གྱིས། །མཁས་རྣམས་དགའ་བ་བསྐྱེད་མི་ནུས། །ཕྱི་རྗེས་རིང་དུ་ལྟ་བ་དང༌། །བག་ཡོད་པ་ལ་བཟོད་སྲན་ཆེ། །བརྩོན་འགྲུས་ཆེ་ཞིང་བརྟན་ལ་གྲིམས། །བྲན་གཡོག་ཡིན་ཡང་དཔོན་དུ་འགྱུར། །སྦྱིན་ལ་རྟག་ཏུ་སེམས་སྤྲོ་བ། །དེ་ཡི་གྲགས་པ་རླུང་བཞིན་ལྡང༌། །ཕོངས་པ་སློང་བ་འདུ་བ་བཞིན། །འབྱལ་བར་འདོད་རྣམས་དེ་བས་མང༌། །བྱིན་ནས་སླར་ཡང་མི་ལེན་ལ། །དམན་པའི་བརྙས་པ་དང་དུ་ལེན། །ཕན་པ་ཆུང་ངུའང་མི་བརྗེད་པ། །བདག་ཉིད་ཆེན་པོའི་ཆེ་རྟགས་ཡིན། །དམ་པའི་ཡོན་ཏན་སྦས་གྱུར་ཀྱང༌། །འཇིག་རྟེན་ཀུན་ལ་ཁྱབ་པར་གསལ། །སྣ་མའི་མེ་ཏོག་ལེགས་བཀབ་ཀྱང༌། །དྲི་ཞིམ་ཀུན་ཏུ་ཁྱབ་པར་འགྱུར། །རྒྱལ་པོ་རང་ཡུལ་ཆེ་བ་ཙམ། །དམ་པ་གང་དུ་ཕྱིན་སར་བཀུར། །མེ་ཏོག་ཕལ་ཆེར་ཉིན་རེའི་རྒྱན། །གཙུག་གི་ནོར་བུ་གང་དུའང་མཆོད། །ལྗོན་ཤིང་དུད་པ་འབྲས་བུ་མང༌། །རྨ་བྱ་དལ་བ་མཇུག་སྒྲོ་བཟང༌། །རྟ་མཆོག་དུལ་བ་འགྲོ་མགྱོགས་ཏེ། །དམ་པ་དུལ་བ་མཁས་པའི་རྟགས། །དམ་པ་རྣམས་དང་ཕལ་པ་ལ། །བྱས་པ་མཉམ་ཡང་དྲིན་མི་འདྲ། །ཞིང་ལ་ས་བོན་ཁྱད་མེད་ཀྱང༌། །ལོ་ཏོག་ཁྱད་པར་དཔག་ཏུ་མེད། །བདག་ཉིད་ཆེ་ལ་ཕན་བཏགས་ན། །ཅུང་ཟད་ཙམ་ལའང་འབྲས་བུ་འབྱིན། །སྐྱུ་རུ་ར་གཅིག་བྱིན་པ་ལ། །ཆོས་རྒྱལ་བུ་དང་མཉམ་ལ་ལྟོས། །བཟང་པོའི་ཆོ་རིགས་སྤྱོད་པས་བསྲུང༌། །སྤྱོད་པ་ཉམས་ན་རིགས་དོན་མེད། །ཙན་དན་དྲི་བཟང་སྐྱེ་བོ་དགའ། །དེ་བསྲེགས་སོལ་བ་སུ་ཞིག་ལེན། །ཆེན་པོ་རེ་ཞིག་རྒུད་གྱུར་ཀྱང༌། །དེ་ལ་གདུང་བ་བསྐྱེད་མི་དགོས། །ཟླ་བ་རེ་ཞིག་གཟས་ཟིན་ཡང༌། །དེ་མ་ཐག་ཏུ་གྲོལ་བར་འགྱུར། །ཆེན་པོས་དགྲ་ལ་བྱམས་བྱས་ན། །དགྲ་ཉིད་དེ་ཡི་དབང་དུ་འགྱུར། །མང་པོས་བཀུར་བས་ཀུན་བསྐྱངས་པས། །ཀུན་གྱིས་རྒྱལ་པོར་དབང་བསྐུར་ཏོ། །དམ་པ་ཇི་ལྟར་རྒྱུད་གྱུར་ཀྱང༌། །སྡིག་དང་འདྲེས་པའི་ཟས་མི་ཟ། །སེང་གེ་བཀྲེས་ཀྱང་མི་གཙང་བའི། །ངན་སྐྱུགས་ཟ་བར་མི་བྱེད་དོ། །དམ་པ་སྲོག་ལ་བབས་ན་ཡང༌། །རང་བཞིན་བཟང་པོ་ག་ལ་འདོར། །ས་ལེ་སྦྲམ་ནི་བསྲེགས་བཅད་ཀྱང༌། །དེ་ཡི་ཁ་དོག་ཉམས་མི་འགྱུར། །སྐྱེས་མཆོག་རྣམས་ལ་དམན་པ་རྣམས། །ཁྲོ་ཡང་ལན་དུ་ག་ལ་ཁྲོ། །ཅེ་སྤྱང་དྲེགས་པས་སྐད་འབྱིན་ཡང༌། །རི་དྭགས་རྒྱལ་པོས་སྙིང་རྗེས་སྐྱོང༌། །བདག་ཉིད་ཆེ་ལ་སྐྱེ་བོ་རྣམས། །སྐྱོན་འཚོལ་འགྱུར་གྱི་དམན་ལ་མིན། །འཕན་ཟེལ་རིན་ཆེན་ལ་ལྟ་ཡི། །མགལ་དུམ་ལ་ཤེ་སུ་ཞིག་སྤྱོད། །བསྟོད་པས་དགའ་བར་མི་འགྱུར་ལ། །སྨད་པས་མི་དགར་མི་འགྱུར་ཞིང༌། །རང་གི་ཡོན་ཏན་ལེགས་གནས་པ། སྐྱེ་བོ་དམ་པའི་མཚན་ཉིད་ཡིན། །སྡིག་པ་དང་ནི་མཐུ་རྩལ་ལས། །བྱུང་བའི་ལོངས་སྤྱོད་ལོངས་སྤྱོད་མིན། །ཁྱི་དང་བྱི་ལ་འགྲངས་ན་ཡང༌། །ངོ་ཚ་བོར་བའི་རྣམ་ཐར་ཡིན། །འཁོར་ལ་ཕུན་སུམ་ཚོགས་གྱུར་ན། །རྗེ་དཔོན་ཉིད་ཀྱི་ཆེ་བ་ཡིན། །རྟ་ལ་རྒྱན་དུ་བྱས་པ་དེ། །བདག་པོ་ཉིད་ལ་མི་མཛེས་སམ། །ཇི་ལྟ་ཇི་ལྟར་རྗེ་དཔོན་གྱིས། །འཁོར་ལ་དྲིན་གྱིས་བསྐྱངས་གྱུར་པ། །དེ་ལྟ་དེ་ལྟར་འཁོར་གཡོག་རྣམས། །རྗེ་དཔོན་ཉིད་ཀྱི་བྱ་བ་སྒྲུབ། །བདག་ཉིད་ཆེན་པོ་གནས་པའི་སར། །མཁས་པ་གཞན་དག་སུ་ཡིས་བརྩི། །མཁའ་ལ་ཉི་མ་ཤར་བ་ན། །རྒྱུ་སྐར་མང་ཡང་མཐོང་མི་འགྱུར། །
