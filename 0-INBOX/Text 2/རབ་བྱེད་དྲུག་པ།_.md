རབ་བྱེད་དྲུག་པ། རང་བཞིན་གྱི་ཚུལ་བརྟག་པ།

།བདག་ཉིད་དཔོན་དུ་བསྐོས་གྱུར་ན། །དེ་ཡི་བྱ་བ་ཤེས་པ་དཀོན། །གཞན་ལ་ལྟ་བའི་མིག་ཡོད་ཀྱང་། །རང་ཉིད་བལྟ་ན་མེ་ལོང་དགོས། །ཤེས་རབ་ཕྱོགས་གཅིག་ལྡན་པས་ཀྱང་། །བྱ་བ་ཀུན་ལ་མཁས་པ་དཀའ། །ཤིན་ཏུ་གསལ་བའི་མིག་གིས་ཀྱང་། །སྒྲ་ལ་ཉན་པར་ནུས་མ་ཡིན། །དྲང་པོར་སྨྲ་བ་སྐྱོན་བཅས་དང་། །འཁྱོག་སྨྲ་ཡོན་ཏན་ལྡན་པ་སྲིད། །ལམ་དྲང་འགའ་ཞིག་ནོར་འཇིགས་ལ། །དུང་དཀར་གཡས་འཁྱིལ་བཀྲ་ཤིས་བྱེད། །བསོད་ནམས་མེད་ན་ཡོན་ཏན་འགའ། །ཡོན་ཏན་ཉིད་ཀྱིས་རང་ཉིད་འཇོམས། །མུ་ཏིག་འཛིན་པའི་ཉ་ཕྱིས་རྣམས། །མུ་ཏིག་ཉིད་ཀྱིས་སྲོག་དང་འབྲལ། །ཡོན་ཏན་ལྡན་ཡང་བསྟེན་དྲགས་ན། །ཕལ་ཆེར་སྐྱོ་བ་སྐྱེ་བ་སྲིད། །བུར་ཤིང་ཤིན་ཏུ་ཞིམ་པ་ཡང་། །རྒྱུན་དུ་བསྟེན་ན་ཕལ་ཆེར་འདོར། །རང་བཞིན་བཟང་པོ་རྣམས་ལ་ཡང་། །རྒྱུན་དུ་གཙེས་ན་གནོད་སེམས་སྐྱེ། །ཙན་དན་བསིལ་བ་ཡིན་མོད་ཀྱི། །གཙུབ་པར་གྱུར་ན་འབར་བར་འགྱུར། །རྒྱལ་པོ་ཤིན་ཏུ་མང་མོད་ཀྱི། །ཆོས་བཞིན་སྐྱོང་བ་ཤིན་ཏུ་ཉུང་། །མཁའ་ལ་ལྷག་གནས་མང་ན་ཡང་། །འོད་གསལ་ཉི་ཟླ་ལྟ་བུ་མེད། །གང་ཞིག་གནོད་པ་བྱེད་ནུས་པ། །དེ་ཡི་ཕན་པ་བྱེད་པར་ནུས། །མགོ་བོ་གཅོད་པར་ནུས་པ་ཡི། །རྒྱལ་པོས་རྒྱལ་སྲིད་སྦྱིན་པར་ནུས། །བློན་པོ་བློ་ལྡན་དྲང་པོ་ཡིས། །རྗེ་དང་འབངས་ཀྱི་དོན་ཀུན་གྲུབ། །མདའ་དྲང་མཁས་པས་འཕངས་གྱུར་ན། །གང་ལ་གཏད་པར་འཕོག་པ་ཡིན། །རྒྱལ་པོ་ཉིད་ཀྱིས་མ་བརྩིས་ན། །ཐམས་ཅད་ཤེས་པའང་བཀུར་མི་འགྱུར། །སྲོག་གི་དབང་པོས་མ་བཟུང་བའི། །མི་རོ་བཟང་ཡང་སུ་ཞིག་ལེན། །མང་པོ་གཅིག་ཏུ་བློ་མཐུན་ན། །ཉམ་ཆུང་གིས་ཀྱང་དོན་ཆེན་འགྲུབ། །སྲོག་ཆགས་གྲོག་མའི་ཚོགས་འདུས་པས། །སེང་གེའི་ཕྲུ་གུ་བསད་ཅེས་གྲགས། །ཞུམ་བཅས་སྩོལ་བ་དོར་བའི་མི། །མཐུ་སྟོབས་ལྡན་ཡང་རྒུད་པར་འགྱུར། །གླང་པོ་ཤིན་ཏུ་སྟོབས་ལྡན་ཡང་། །གླང་རྫི་ཆུང་ངུས་བྲན་བཞིན་འཁོལ། །ང་རྒྱལ་རྩོལ་བ་ཆེར་བསྐྱེད་ན། །ཆེན་པོ་རྣམས་ཀྱང་ཟིལ་གྱིས་གནོན། །དུང་དཀར་ཤིན་ཏུ་ལུས་ཆུང་ཡང་། །ཆུ་སྲིན་ཆེན་པོའི་གཤེད་མ་ཡིན། །ཆེན་པོས་ང་རྒྱལ་བྱ་མི་དགོས། །ངན་པས་ང་རྒྱལ་བྱས་ཅི་ཕན། །རིན་ཆེན་ཚིག་གིས་སྒྲོགས་མི་དགོས། །མཆིང་བུ་བསྒྲགས་ཀྱང་སུ་ཞིག་ཉོ། །བདག་ཉིད་ཆེན་པོའི་འབྱོར་པ་རྟག །སྐྱེས་ངན་ཕུན་ཚོགས་རྒུད་ལ་ཉེ། །ཉི་མ་རྒྱུན་དུ་འོད་འབྱིན་ཏེ། །ཟླ་བ་ཉ་བར་གྱུར་ནས་འབྲི། །རྒྱལ་པོས་གོ་འཕང་བསྟོད་དྲགས་ན། །དེ་ཡི་ཐ་མ་བརླག་པའི་རྒྱུ། །སྒོ་ང་མཁའ་ལ་འཕངས་གྱུར་ན། །ཆག་པ་ཉིད་ལས་གཞན་ཅི་འབྱུང་། །སྐྱེ་བོ་ཕལ་ཆེར་རང་ཉིད་དང་། །མཐུན་པའི་རིགས་ཀྱིས་གནོད་པ་སྐྱེལ། །ཉི་མའི་འོད་ཟེར་ཤར་བ་ན། །འོད་ཟེར་གཞན་རྣམས་བརླག་པར་འགྱུར། །ཕན་བྱེད་དགྲ་བོ་ཡིན་ཡང་བསྟེན། །གཉེན་ཡང་གནོད་ན་སྤང་བར་བྱ། །རྒྱ་མཚོའི་ནོར་བུ་རིན་གྱིས་ཉོ། །ཁོང་པའི་ཟུག་རྔུ་སྨན་གྱིས་འབྱིན། །ནང་དུ་ཅུང་ཟད་འབྱོར་གྱུར་ན། །ཕྱི་རུ་དྲེགས་པའི་རྣམ་འགྱུར་སྟོན། །ཆུ་ཡིས་ཡོངས་སུ་གང་བའི་ཚེ། །སྤྲིན་རྣམས་གཡོ་ཞིང་འབྲུག་སྒྲ་སྒྲོགས། །ཡོན་ཏན་ཀུན་ལྡན་དཀོན་པ་ལྟར། །ཡོན་ཏན་ཅི་ཡང་མེད་པའང་དཀོན། །སྐྱོན་དང་ཡོན་ཏན་འདྲེས་པ་ལས། །ཡོན་ཏན་གཙོ་ཆེ་མཁས་པས་བསྟེན། །དང་པོ་ཉིད་ནས་དགྲ་བོ་དང་། །མཛའ་བོ་ཉིད་དུ་ངེས་པ་མེད། །ཁ་ཟས་མ་ཞུ་དུག་ཏུ་འགྱུར། །དུག་ཀྱང་ཤེས་ན་སྨན་དུ་འགྱུར། །གང་ན་ལས་མཐུན་ཡོད་དེ་འབྱོར། །ལས་ཀྱི་འདུ་འཕྲོ་མེད་དེ་རྒུད། །ཁྱིམ་དུ་བཟུང་ཡང་མི་ཆགས་པའི། །ངང་པ་མཚོ་ལས་བསྐྲད་ཀྱང་འཁོར། །མཁས་རྣམས་རིག་པ་ཡོན་གྱིས་སློབ། །བླུན་པོ་རིག་པ་བསླབས་ཀྱང་སྤོང་། །ཕལ་ཆེར་ན་ལ་སྨན་ཟ་ཡང་། །ལ་ལ་གསོན་པོར་ལྕེབ་པ་ཡོད། །རང་དབང་ཐམས་ཅད་བདེ་བ་སྟེ། །གཞན་དབང་ཐམས་ཅད་སྡུག་བསྔལ་ཡིན། །ཐུན་མོང་བ་ནི་རྩོད་གཞི་སྟེ། །དམ་བཅས་པ་ནི་འཆིང་བའི་རྒྱུ། །ནང་ན་ཡོན་ཏན་ཀུན་ལྡན་ཡང་། །ཆ་ལུགས་ངན་ན་ཀུན་གྱིས་བརྙས། །པགས་བྱེའུ་མཁས་ཀྱང་སྤུ་མེད་པས། །འདབ་ཆགས་ཀུན་གྱིས་སྤང་ཞེས་གྲགས། །གནས་མིན་ཧ་ཅང་དྲང་པོ་ཡང་། །རང་གཞན་གཉིས་ཀ་བརླག་པ་ཡོད། །མདའ་ཡིས་ཕ་རོལ་གསོད་པའམ། །ཡང་ན་རང་ཉིད་འཆག་པར་འགྱུར། །ཆར་དང་ཆུ་བོ་རྒྱ་མཚོར་འབབ། །བློ་དང་རིག་པ་མཁས་ལ་འགོ། ནོར་དང་སྐྱེ་བོ་རྒྱལ་པོས་སྡུད། །དྲོ་ཞིང་བརླན་པར་ནགས་ཚལ་སྐྱེ། །དབྱར་གྱི་ཆུ་མིག་རྩྭ་ཡི་མེ། །སྤྲིན་པར་ཉི་མ་ཟླ་བ་ཉ། །བློ་ངན་རིག་པ་ངན་དང་མཛའ། །དགོས་ཚེ་འབྱུང་བའི་ངེས་པ་མེད། །བླུན་པོ་སྨྲ་བ་ཉུང་ན་མཛེས། །རྒྱལ་པོ་ལྐོག་ཏུ་འདུག་ན་མཛེས། །སྒྱུ་མ་བརྒྱ་ལམ་མཐོང་ན་མཛེས། །ནོར་ཆེན་དཀོན་ན་རིན་ཐང་ཆེ། །ཧ་ཅང་བྱམས་པ་ཆེ་དྲགས་ན། །དེ་ཉིད་འཁོན་པའི་རྒྱུ་རུ་འགྱུར། །འཇིག་རྟེན་རྩོད་པ་ཕལ་ཆེ་བ། །འབྲེལ་བ་ཉིད་ལས་བྱུང་བ་མང་། །ཤིན་ཏུ་རྩོད་པ་ཆེན་པོ་ཡང་། །ལྷག་པར་མཛའ་བའི་རྒྱུ་སྲིད་དེ། །རྩོད་པ་བྱུང་བའི་མཐའ་ཕལ་ཆེར། །འདུམ་པར་གྱུར་པ་མང་བར་མཐོང་། །སེར་སྣ་ཅན་གྱི་ནོར་རྣམས་དང་། །ཕྲག་དོག་ཅན་གྱི་གྲོགས་པོ་དང་། །བློ་ངན་གྱི་ནི་རིག་པ་རྣམས། །ཡོད་ཀྱང་དགའ་བ་བསྐྱེད་མི་ནུས། །བརྐམ་ཆགས་ཅན་ནི་ནོར་གྱིས་དགའ། །ང་རྒྱལ་ཅན་ནི་བསྟོད་པས་མགུ། །བླུན་པོ་རང་དང་མཐུན་པས་དགའ། །དམ་པ་བདེན་པ་སྨྲ་བས་མགུ། །སྐྱེ་བོ་ངན་པའི་ཡོན་ཏན་དང་། །སྨྲ་སྤོབས་ཞན་པའི་རིག་པ་དང་། །རྗེ་དཔོན་ངན་པའི་བཀའ་དྲིན་རྣམས། །གཞན་ལ་ཕན་པའི་གོ་སྐབས་དཀའ། །གང་ལ་ནོར་ཡོད་སྨྲ་བ་མཛེས། །ནོར་མེད་སྨྲ་བ་བདེན་ཡང་སྤོང་། །མ་ལ་ཡ་ལས་བྱུང་གྱུར་ན། །སྡོང་དུམ་ཕལ་ལའང་རིན་ཐང་ཆེ། །སྨྲ་མང་ཉེས་པ་འཛིན་པའི་རྒྱུ། །མི་སྨྲ་ཉེས་པ་སྤོང་བའི་གཞི། །ནེ་ཙོ་སྨྲ་བས་གཟེབ་ཏུ་བཅུག །འདབ་ཆགས་ལྐུགས་པ་བདེ་བར་རྒྱུ། །མི་གང་དགྲ་ལ་གཡོ་མེད་པས། །རྣམ་པ་ཀུན་ཏུ་ཕན་བཏགས་ན། །དགྲ་ཡང་དེ་ལ་གཡོ་མེད་པས། །འདུད་པ་དངོས་པོའི་ཆེ་བ་ཡིན། །སྟོབས་བྲལ་མི་ཡིས་ཁྲོས་ཅི་ཕན། །སྟོབས་ལྡན་དོན་ལ་ཁྲོ་ཅི་དགོས། །དེས་ན་བྱ་བ་སྒྲུབ་པ་ལ། །ཁྲོ་བ་དོན་མེད་རང་སྲེག་ཡིན། །སྦྱིན་པས་བསྡུས་ན་དགྲ་ཡང་འདུ། །སྦྱིན་མེད་གཉེན་ཡང་རིང་དུ་སྤོང་། །བ་ཡི་འོ་མ་ཟད་པ་ན། །བེའུ་བཟུང་ཡང་འཆོར་བར་འགྱུར། །བྱ་བ་འགའ་ལ་མཁས་པ་ཡིས། །གཞན་དག་ཤེས་པའི་ངེས་པ་མེད། །ཆུ་ལས་འོ་མ་འབྱེད་ཤེས་པའི། །ངང་པས་གཟུགས་བརྙན་ཟས་སུ་སྒོམ། །རྟག་ཏུ་བྱམས་པས་སྐྱོང་བ་ཡི། །རྗེ་བོས་བྲན་གཡོག་རྙེད་པར་སླ། །པདྨའི་མཚོ་ལ་ངང་པ་དག །མ་བསྡུས་པར་ཡང་ངང་གིས་འདུ། །ལོངས་སྤྱོད་ལྡན་པས་སྤྱོད་པ་དང་། །མཁས་པར་གྱུར་ནས་དུལ་བ་དང་། །ཆེན་པོས་དམན་པ་ལེགས་སྐྱོང་བ། །གསུམ་པོ་གཞན་བདེ་རང་ལ་ཕན། །བསོད་ནམས་ཤུགས་ཀྱིས་དོན་བསྒྲུབ་པ། །ཉི་མའི་འོད་བཞིན་ལྟོས་པ་མེད། །འབད་པའི་ཤུགས་ཀྱིས་དོན་བསྒྲུབ་པ། །མར་མེའི་འོད་བཞིན་ཀུན་ལ་ལྟོས། །ཆེན་པོ་རྣམས་ལ་རྟེན་བཅས་ན། །དམན་པ་རྣམས་ཀྱང་ཆེན་པོ་ཐོབ། །ལྗོན་ཤིང་ཆེ་ལ་བརྟེན་པ་ཡིས། །འཁྲི་ཤིང་རྩེ་མོར་ཕྱིན་ལ་ལྟོས། །ཡོན་ཏན་ལྡན་ལ་སྐྱོན་ཡོད་ཀྱང་། །ཡོན་ཏན་དགའ་རྣམས་ཉེ་བར་བསྟེན། །ཆར་གྱིས་མཁར་ལ་གནོད་ན་ཡང་། །འཇིག་རྟེན་པ་རྣམས་དགའ་བ་སྐྱེ། །ཡོན་ཏན་མེད་ན་ཆ་ལུགས་ཀྱིས། །མཁས་རྣམས་དགའ་བ་བསྐྱེད་མི་ནུས། །རྟ་མཆོག་འགྲོས་དང་མི་ལྡན་ན། །ཤིན་ཏུ་བརླིང་ཡང་རིན་ཐང་ཆུང་། །ནོར་ལྡན་བླུན་པོའི་ནང་ན་མང་། །དཔའ་བོ་གཅན་གཟན་ཁྲོད་ན་མོད། །ལེགས་བཤད་མཁས་པའི་ཁྲོད་འབྱིན་པ། །དམ་པ་འཇིག་རྟེན་འདི་ན་དཀོན། །གང་དང་གང་ལ་དེ་ཡོད་པ། །དེ་ཡིས་དེ་ཡི་གྲགས་པ་ཐོབ། །མཁས་པའི་གྲགས་པ་མཁས་པ་དང་། །དཔའ་བོས་དཔའ་བོ་ཐོས་པ་བཞིན། །ཆེན་པོ་རྣམས་ཀྱི་མཆོད་བྱ་བ། །དམན་པ་རྣམས་ཀྱིས་བརྙས་པར་འགྱུར། །དབང་ཕྱུག་ཆེན་པོའི་སྤྱི་བོའི་རྒྱན། །ཟླ་བ་ལྷ་མིན་ཟས་སུ་བྱེད། །རིག་པ་གླེགས་བམ་ལ་གནས་དང་། །མ་བསྒྲུབས་པ་ཡི་གསང་སྔགས་དང་། །བརྗེད་ངས་ཅན་གྱིས་བསླབ་པ་རྣམས། །དགོས་པའི་ཚེ་ན་སླུ་བ་མང་། །བློ་ལྡན་ནོར་གྱིས་འབྱོར་ན་ཡང་། །ལེ་ལོ་ཅན་ལ་མཐོན་པོ་དཀའ། །རྣ་བ་ཐོག་མར་སྐྱེས་ན་ཡང་། །རྭ་རྩེ་མཐོ་བ་མ་མཐོང་ངམ། །ཁྱི་ཕག་རྣམས་ལ་དྲི་ཞིམ་དང་། །དམུས་ལོང་བ་ལ་སྒྲོན་མེ་དང་། །མ་ཞུ་བ་ལ་ཁ་ཟས་དང་། །བླུན་པོ་རྣམས་ལ་ཆོས་ཅི་དགོས། །ཡོན་ཏན་ཅན་དང་གསེར་བཟང་པོ། །གཡུལ་ངོར་དཔའ་དང་རྟ་བཟང་པོ། །སྨན་པ་མཁས་དང་རྒྱན་བཟང་པོ། །གང་དུ་ཕྱིན་པ་དེ་རུ་བྲིན། །བློ་དང་བརྩོན་འགྲུས་ཡོད་གྱུར་པ། །དེ་ཡིས་སྒྲུབ་པར་མི་ནུས་གང་། །དགྲ་ངན་དཔུང་ཚོགས་བཅུ་གཉིས་པོ། །སྐྱ་སེང་བུ་ཡིས་བཅོམ་ཞེས་ཐོས། །ཕལ་ཆེར་བུ་ཚའི་སྤྱོད་པ་ནི། །རིགས་རྒྱུད་སྔ་མའི་རྗེས་སུ་འབྲང་། །ཁུ་བྱུག་ཕྲུ་གུ་བྱེའུ་ཁྲ་རུ། །འགྲོ་བ་བརྒྱ་ལམ་སྲིད་པ་ཙམ། །རི་དང་ཆུ་བོ་གླང་པོ་རྟ། །ཤིང་དང་འོད་ཟེར་ནོར་བུ་རྡོ།
