རབ་བྱེད་བཞི་པ། སྤེལ་མ་བརྟག་པ།

།དམན་པ་ལོངས་སྤྱོད་ཆེ་ན་ཡང་། །རིགས་ལྡན་རྒུད་པས་ཟིལ་གྱིས་གནོན། །བཀྲེས་པ་སྟག་གི་ངར་སྒྲ་ཡིས། །སྤྲེའུ་ཤིང་གི་རྩེ་ལས་ལྷུང་། །བླུན་པོའི་ཡོན་ཏན་ཁར་འབྱིན་ཏེ། །མཁས་པའི་ཡོན་ཏན་ཁོང་དུ་སྦེད། །སོག་མ་ཆུ་ཡི་སྟེང་ན་འཕྱོ། །ནོར་བུ་སྟེང་དུ་བཞག་ཀྱང་འབྱིང་། །ཡོན་ཏན་ཆུང་རྣམས་ང་རྒྱལ་ཆེ། །མཁས་པར་གྱུར་ན་དུལ་བར་གནས། །ཆུ་ཕྲན་རྟག་ཏུ་ཀུ་ཅོ་ཆེ། །རྒྱ་མཚོ་ཅ་ཅོ་ག་ལ་སྒྲོགས། །སྐྱེ་བོ་དམན་རྣམས་དམ་པ་ལ། །ཁྱད་གསོད་བྱེད་ཀྱིས་དམ་པས་མིན། །སེང་གེས་ཝ་ཚོགས་ལེགས་སྐྱོང་གི། ཝ་ཉིད་ཝ་ཡི་རིགས་ལ་འགྲན། །དམ་པ་ཁྲོས་ཀྱང་བཏུད་ན་ཞི། །དམན་རྣམས་བཏུད་ན་ལྷག་པར་རེངས། །གསེར་དངུལ་སྲ་ཡང་བཞུ་ནུས་ཀྱི། །ཁྱི་ལུད་བཞུས་ན་དྲི་ངན་འབྱུང་། །མཁས་ལ་ཡོན་ཏན་ཀུན་ལྡན་ཏེ། །བླུན་པོ་ལ་སྐྱོན་འབའ་ཞིག་ཡིན། །རིན་པོ་ཆེ་ལས་དགོས་དགུ་འབྱུང་། །སྦྲུལ་གདུག་ལས་ནི་ཉེས་ཀུན་བསྐྱེད། །སྡིག་སྤྱོད་ནགས་ན་གནས་ཀྱང་ཉམས། །དམ་པ་གྲོང་ན་གནས་ཀྱང་དུལ། །ནགས་ཀྱི་གཅན་གཟན་ཁྲོ་བ་དང་། །རྟ་མཆོག་གྲོང་གནས་དུལ་བར་མཐོང་། །སྐྱེས་མཆོག་རང་གི་སྐྱོན་ལ་ལྟ། །སྐྱེ་བོ་ངན་པ་གཞན་སྐྱོན་འཚོལ། །རྨ་བྱ་རང་གི་ལུས་ལ་རྟོག །སྲིན་བྱ་གཞན་ལ་ལྟས་ངན་གཏོང་། །དམ་པ་དུལ་བས་རང་གཞན་སྐྱོང་། །ངན་པ་རེངས་པས་རང་གཞན་སྡུག །འབྲས་ལྡན་ཤིང་གིས་རང་གཞན་སྲུང་། །ཤིང་སྐམ་རེངས་པས་རང་གཞན་སྲེག །འབྱོར་པའི་དུས་ན་ཐམས་ཅད་གཉེན། །གལ་ཏེ་རྒུད་ན་ཀུན་ཀྱང་དགྲ། །རིན་ཆེན་གླིང་དུ་རིང་ནས་འདུ། །མཚོ་བསྐམས་པ་ན་སུ་ཡང་སྤོང་། །བླུན་པོ་འབྱོར་པ་ཐོབ་ན་བདེ། །སྐྱེས་མཆོག་ཐམས་ཅད་བྱིན་ན་བདེ། །མཛེ་ཅན་ཕྲུགས་ན་བདེ་བར་འཛིན། །མཁས་པ་མཛེ་ལ་འཇིགས་པར་ལྟ། །ཆེ་ལ་གནོད་བྱེད་གྲོགས་སུ་འགྱུར། །དམན་ལ་གནོད་བྱེད་གནོད་པར་འགྱུར། །ནགས་མེ་རླུང་གིས་སྦོར་མོད་ཀྱི། །དེ་ཡིས་སྒྲོན་མེ་ཆུང་ངུ་གསོད། །འདི་བཤེས་འདི་ནི་དགྲ་བོ་ཞེས། །བློ་གྲོས་ཆུང་རྣམས་སོ་སོར་འབྱེད། །བློ་ཆེན་ཀུན་ལ་བྱམས་པ་ལ། །ཕན་བྱེད་གང་ཡིན་ངེས་པ་མེད། །ཡོན་ཏན་ལྡན་པ་ཡོན་ཏན་ལ། །དགའ་ཡི་ཡོན་ཏན་མེད་རྣམས་མིན། །སྦྲང་རྩི་འཛིན་པ་མེ་ཏོག་ལ། །དགའ་ཡི་ཤ་སྦྲང་དེ་ལྟ་མིན། །མཁས་པ་མཁས་པའི་ནང་ན་མཛེས། །བླུན་པོས་མཁས་པ་ཇི་ལྟར་གོ། ཙན་དན་གསེར་བས་རིན་ཆེ་བ། །བླུན་པོས་སོལ་བར་བྱས་ལ་ལྟོས། །མཁས་པ་རང་གིས་དཔྱོད་ཤེས་ཀྱི། །བླུན་པོ་གྲགས་པའི་རྗེས་སུ་འབྲང་། །ཁྱི་རྒན་ཀུ་ཅོ་འདོན་པ་ན། །རྒྱུ་མཚན་མེད་པར་གཞན་དག་རྒྱུག །མཁས་པ་ཤིན་ཏུ་རྒུད་ན་ཡང་། །ལེགས་བཤད་གཏམ་གྱིས་གཞན་དགའ་བྱེད། །བླུན་པོ་འབྱོར་པར་གྱུར་ཚེ་ཡང་། །རྩོད་པ་ཁོ་ནས་རང་གཞན་སྲེག །ཁ་ཅིག་སྨྲ་བས་གྲུབ་པར་བརྩི། །ལ་ལ་མི་སྨྲ་དོན་ལ་འབད། །ཁྱི་ངན་དགྲ་ལ་ཁུས་འདེབས་ཏེ། །ཆུ་སྐྱར་བྱི་ལ་སྒྲ་མེད་འཇབ། །དམ་པ་རྒོལ་ཡང་ཕན་པ་སྒྲུབ། །སྐྱེ་བོ་ངན་པ་མཛའ་ཡང་གནོད། །ལྷ་རྣམས་ཁྲོ་བས་སེམས་ཅན་སྲུང་། །གཤིན་རྗེ་འཛུམ་ཡང་ཕ་རོལ་གསོད། །སྐྱེ་བོ་དམ་པ་རིན་ཆེན་བཞིན། །རྣམ་པ་ཀུན་ཏུ་འགྱུར་ལྡོག་མེད། །སྐྱེ་བོ་ངན་པ་སྲང་མདའ་བཞིན། །ཅུང་ཟད་ཙམ་གྱིས་མཐོ་དམའ་བྱེད། །བློ་མཐུན་རིང་ན་གནས་ཀྱང་ཕན། །མི་མཐུན་ཉེ་ན་ལྷག་པར་འབྲལ། །འདམ་སྐྱེས་འདམ་གྱིས་མི་གོས་ལ། །དེ་ཉིད་ཉི་མས་རྟག་ཏུ་སྐྱོང་། །ཇི་སྲིད་ངོ་ཚ་ཡོད་གྱུར་པ། །དེ་སྲིད་ཡོན་ཏན་རྒྱན་གྱི་མཆོག །ངོ་ཚ་བརྒལ་ན་ཡོན་ཏན་ཉིད། །ཕྱོགས་སུ་བྱས་ཏེ་གཏམ་ངན་འཕེལ། །དམ་པ་མ་བཅོལ་ལེགས་པར་སྟོན། །དམན་ལ་དྲིས་ཀྱང་ལོག་པར་ཤོད། །རྒྱལ་བའི་སྲས་ལ་བརྙས་ཀྱང་བརྩེ། །འཆི་བདག་མཆོད་ཀྱང་གསོད་པ་ཡིན། །གཅིག་ལ་ཕན་པར་འགྱུར་པ་ཡི། །བྱ་བ་གཞན་ལ་གནོད་པ་སྲིད། །ཟླ་བ་ཤར་ན་ཀུ་མུད་ཏ། །ཁ་འབྱེད་འགྱུར་གྱི་པདྨོ་ཟུམ། །ཉེས་པ་ལ་ལས་དོན་གྲུབ་པ། །བྱུང་ཡང་མཁས་རྣམས་ག་ལ་སྨོན། །ལེགས་པར་བསྒྲུབས་ཀྱང་སྐྱོན་སྲིད་མོད། །དེ་ལ་མཁས་རྣམས་ཁྲེལ་བ་མེད། །རྙེད་པ་ལ་ལ་རྙེད་པ་སྟེ། །རྙེད་པ་ལ་ལ་དགྲ་བོ་ཡིན། །རྒོད་མ་མངལ་ལྡན་ནོར་འཕེལ་བྱེད། །དྲེའུ་མོ་སྦྲུམ་ན་འཆི་བ་རྙེད། །དམ་པ་དབྱེ་དཀའ་འདུམ་སླ་སྟེ། །དམན་པ་དབྱེ་སླ་འདུམ་པ་དཀའ། །ལྗོན་ཤིང་དང་ནི་སོལ་བ་ཡི། །གཅད་དང་སྦྱོར་བའི་ཁྱད་པར་ལྟོས། །ཉམ་ཆུང་གྱུར་ཀྱང་ཟོན་བྱས་ན། །ཆེན་པོ་ཡིས་ཀྱང་གཞོམ་པར་དཀའ། །ཆེན་པོ་ཡིན་ཀྱང་བག་ཕེབས་ན། །ཉམ་ཆུང་རྣམས་ཀྱིས་བརླགས་པ་མང་། །ནོར་གྱིས་འབྱོར་ན་སྟོབས་ཀྱང་འཕེལ། །ནོར་མེད་པ་ལ་སྟོབས་ཀྱང་འབྲི། །བྱི་བས་དབྱིག་ལྡན་ནོར་ཕྲོགས་པས། །བརྐུ་བའི་ནུས་པ་ཉམས་ཞེས་ཐོས། །མི་གང་བསོད་ནམས་ལ་སྤྱོད་པ། །བཏང་ཡང་ལོངས་སྤྱོད་ཆར་བཞིན་འབབ། །བསོད་ནམས་མེད་པའི་ལོངས་སྤྱོད་དེ།
