རབ་བྱེད་བརྒྱད་པ། བྱ་བ་བརྟག་པ།

།བློ་ལྡན་བྱ་བ་ཅུང་ཟད་ཀྱང་། །རྒྱུན་དུ་གྲོས་ཀྱིས་བསྒྲུབ་པར་བྱ། །གྲུབ་པར་གྱུར་ན་ལྟ་ཅི་སྨོས། །མ་གྲུབ་ན་ཡང་མཛེས་པའི་རྒྱུ། །སེམས་ཅན་མོས་པ་སྣ་ཆོགས་པས། །ཐམས་ཅད་མགུ་བ་སུས་ཀྱང་དཀའ། །རང་ཉིད་ཡོན་ཏན་ལྡན་བྱས་ན། །ཐམས་ཅད་དགའ་བ་དེ་ལ་ཉེ། །ཤིན་ཏུ་རྒས་པར་གྱུར་ཚེ་ཡང་། །ཐོས་པ་མང་དུ་གསག་པར་བྱ། །ཕྱི་མར་ཐོས་པས་ཕན་པ་ཙམ། །སྦྱིན་པ་ཡིས་ཀྱང་ག་ལ་ཕན། །ཡོན་ཏན་ཀུན་རྫོགས་སྐྱེ་བོ་བསྟེན། །ཡང་ན་ཐ་མལ་འགྲོགས་པ་བདེ། །བུམ་པ་ཆུ་ཡིས་གང་བ་འམ། །ཡང་ན་སྟོང་པར་ཁུར་བ་སླ། །ཡོན་ཏན་ཅུང་ཟད་བསླབས་པའི་མི། །སུ་ཞིག་གིས་ནི་བསྟེན་པར་ནུས། །ཆུས་ཕྱེད་གང་བའི་བུམ་པ་དེ། །མགོ་ལ་འཁུར་བ་སུ་ཡིས་ནུས། །མི་གང་སྐྱེ་བོ་དམ་པ་དང་། །དམན་པའི་ཁྱད་པར་ལེགས་ཤེས་ནས། །དེ་ཡི་བྱ་བ་བསྒྲུབ་ཤེས་པ། །ཕུན་སུམ་ཚོགས་པའི་གཞི་ཆེན་ཡིན། །བློ་གྲོས་ལྡན་པས་ལེགས་བསྐྱངས་ན། །སྐྱེ་བོ་དམན་པའང་མཆོག་ཏུ་འགྱུར། །སློབ་ཤེས་དག་གིས་ནེ་ཙོ་ལ། །བསླབས་ན་འདོན་པ་ཤེས་པར་འགྱུར། །ཉམས་སྟོབས་ཆུང་བའི་སྐྱེ་བོས་ཀྱང་། །ཆེན་པོ་གཞན་ལ་བསྟེན་ན་འགྲུབ། །ཆུ་ཡི་ཐིགས་པ་ཉམ་ཆུང་ཡང་། །མཚོ་དང་འདྲེས་ན་སྐེམ་མི་ནུས། །རང་ལ་བློ་གྲོས་མི་ལྡན་ན། །བློ་ཆེན་གཞན་ལ་ལེགས་པར་འདྲི། །ལག་པས་དགྲ་བོ་མི་གསོད་ན། །མཚོན་ཆ་ལེན་པར་མི་བྱེད་དམ། །གནོད་པར་བྱེད་པའི་དགྲ་བོ་ཡང་། །ཐབས་དང་ལྡན་ན་གྲོགས་སུ་འགྱུར། །དུག་ཆེན་ལུས་ལ་གནོད་མོད་ཀྱི། །སྦྱོར་བ་ཤེས་ན་སྨན་དུ་འགྱུར། །རིགས་པས་རྙེད་པའི་ཟས་ནོར་བླང་། །མི་འོས་གཞན་ལ་རློམ་སེམས་སྤང་། །ཤིང་ཏོག་ཤིང་རྩེ་ལས་བླང་གི། དེ་ལས་ཐལ་ན་ས་ལ་ལྷུང་། །མཁས་པས་སྒྲིམ་པར་མི་བྱེད་པ། །དེ་ཡི་བར་ལ་ཉེས་པ་འབྱུང་། །བློ་དང་ལྡན་པས་རབ་བསྒྲིམས་ན། །ཉེས་པ་འབྱུང་བའི་གོ་སྐབས་དཀའ། །ཇི་སྲིད་རང་སྟོབས་མ་རྫོགས་པར། །དེ་ཡི་བར་དུ་དགྲ་རྣམས་བཀུར། །ནུས་པ་རྫོགས་ནས་གང་རིགས་པ། །བྱ་ཞེས་བསྟན་བཅོས་གཞན་དག་སྨྲ། །དགྲ་བོ་སྙན་པར་སྨྲ་ན་ཡང་། །བློ་དང་ལྡན་པས་ཡིད་མི་བརྟན། །ཆུ་སྐྱར་བྱི་ལ་དེས་པ་ཡིས། །ཕ་རོལ་གསོད་ལ་རྟག་ཏུ་བརྩོན། །ཡུལ་གྱི་བདག་པོར་ཞེ་སྡང་ཡང་། །དེ་ལ་དགའ་བས་བསྟེན་ཏེ་བསྡད། །ས་ལ་རྐང་པ་འདྲེད་གྱུར་ན། །ས་ཉིད་ལ་ནི་བརྟེན་པ་བཞིན། །ཧ་ཅང་འདོད་ལ་བརྐམ་པ་ཡིས། །སྐྱེ་བོ་མྱུར་དུ་བརླག་པར་འགྱུར། །མཆིལ་པའི་ཤ་ལ་ཆགས་པ་ཡིས། །ཉ་རྣམས་དེ་མ་ཐག་ཏུ་གསོད། །མཆོད་འོས་པ་དང་འཁོར་རྣམས་ཀྱང་། །རྟག་ཏུ་སྦྱིན་པས་བསྡུ་བར་བྱ། །མཆོད་གཏོར་ལ་ནི་ལྷ་རྣམས་དང་། །ཡི་དྭགས་བར་དུ་མགུ་བས་བསྲུང་། །ཆེན་པོ་རྣམས་ཀྱིས་རྩེད་མོ་དང་། །བདེ་དང་ཟས་ལ་ཆགས་པ་སྤང་། །འདོད་ལ་ཆགས་པའི་ལེ་ལན་གྱིས། །འབོད་གྲོགས་ལང་ཀར་བསད་ཅེས་གྲགས། །བྱམས་རྩོད་གཉིས་ཀ་ཆེན་པོ་དང་། །བྱ་ཡི་དམན་དང་ནམ་ཡང་མིན། །ཉོ་ཚོང་རིན་ཆེན་ལ་བྱ་ཡི། །སྦྱར་དུག་ལ་ནི་དེ་ལྟར་མིན། །དཔྱ་བ་མང་ཕྱིན་རྒྱལ་པོའི་མཛོད། །ཧ་ཅང་མི་བསྡུ་ཉུང་ངུས་ཁེངས། །གྲོག་མཁར་སྦྲང་རྩི་ཡར་ངོ་ཡི། །ཟླ་བ་ཉུང་ངུ་ཉུང་ངུས་གང་། །རྗེ་བོས་འབངས་ལ་མི་གནོད་པར། །རིགས་པའི་ལམ་གྱི་དཔྱ་རྣམས་བསྡུ། །ས་ལའི་ཤིང་ལས་སྤོས་ཀྱི་བཅུད། །གཟགས་པ་མང་ན་བསྐམ་པར་འགྱུར། །རྗེ་བོ་ལྷག་པར་དེས་པ་དགོས། །རྒྱུ་མཚན་ཆུང་ངུས་ཁྲོ་མི་རུང་། །སྦྲུལ་ལ་ནོར་བུ་ཡོད་ན་ཡང་། །སེམས་ཤེས་སུ་ཞིག་དྲུང་དུ་སྡོད། །གང་ཞིག་ནོར་འདོད་སྐྱེ་བོས་ཀྱང་། །ཆོས་ཉིད་གཙོ་བོར་བསྲུང་བར་བྱ། །ཆོས་ལས་ཉམས་པའི་ནོར་ཙམ་ནི། །འཇིག་རྟེན་འདིར་ཡང་ག་ལ་རྟག །ཧ་ཅང་བྱམས་ཐལ་གཉེན་ལའང་མིན། །ཤིན་ཏུ་གནོད་ཆེ་དགྲ་ལའང་མིན། །གཉེན་ལ་རེ་བ་འཁོན་གཞི་སྟེ། །གནོད་ལན་ཀུན་གྱིས་བསྒྲུབ་པར་སླ། །འཇམ་པོས་འཇམ་པོ་འཇོམས་བྱེད་ཅིང་། །འཇམ་པོས་རྩུབ་པོའང་འཇོམས་པར་བྱེད། །འཇམ་པོས་ཀུན་འགྲུབ་དེ་ཡི་ཕྱིར། །འཇམ་ཉིད་རྣོ་ཞེས་མཁས་རྣམས་སྨྲ། །སུ་ལའང་བདག་གི་དགྲ་ཡིན་པ། །འགའ་ཡང་བདག་ལ་མི་བྱསས་ཞེས། །མི་བྱམས་ན་ཡང་དེ་མི་བསྒྲག །བསྒྲགས་ན་དེ་ཉིད་འབྱེད་པའི་སེལ། །ངོ་ཚ་ཁྲེལ་ལ་མི་ལྟ་ཞིང་། །མཆོད་དང་བརྙས་པའི་ཁྱད་མི་ཤེས། །ཟས་དང་ནོར་ལ་ཆགས་ལྡན་པ། །དེ་འདྲའི་གནས་སུ་བསྡད་མི་བྱ། །ཡུལ་གཞན་ལེགས་པར་མ་བརྟགས་པར། །སྔ་མའི་གནས་ནི་འདོར་མི་བྱ། །རྐང་པ་ཡ་གཅིག་མ་བཙུགས་པར། །གཉིས་ཀ་བཏེགས་ན་འགྱེལ་བའི་རྒྱུ། །རང་སྤྱོད་འབད་པས་སྦ་བར་བྱ། །ཕལ་ཆེར་གསལ་བར་བསྟན་པས་ཉམས། །སྤྲེ་འུ་ཟློས་གར་མི་བྱེད་ན། །མགུལ་དུ་ཐག་པ་ཅི་སྟེ་འདོགས། །ཉེས་པ་མངོན་སུམ་མཐོང་ན་ཡང་། །གནས་མ་ཡིན་ལ་སྨྲ་མི་བྱ། །ལྟས་ངན་མཐོང་ན་སྨྲ་བ་པོའི། །ཐོག་ཏུ་འབྱུང་ཞེས་འཇིག་རྟེན་སྒྲོགས། །གཞན་ལ་བརྙས་པར་འགྱུར་བ་ཡི། །ནོར་དང་ཟས་ཀྱིས་ཅི་ཞིག་བྱ། །ཁྱི་ཕག་མི་གཙང་ཟ་བ་ལ། །མཁས་པ་སུ་ཞིག་ཡིད་སྨོན་བྱེད། །ཕ་རོལ་གནད་དུ་འབེབས་པའི་ཚིག །དགྲ་བོ་ལ་ཡང་སྨྲ་མི་བྱ། །བྲག་ཆ་བཞིན་དུ་རང་ཉིད་ལ། །དེ་མ་ཐག་ཏུ་ལེ་ལན་འབྱུང་། །གལ་ཏེ་དགྲ་ལ་གནོད་འདོད་ན། །རང་ཉིད་ཡོན་ཏན་ལྡན་པར་བྱ། །དེ་ཡིས་དགྲ་ཡང་སེམས་སྲེག་ཅིང་། །རང་ཡང་བསོད་ནམས་འཕེལ་བར་འགྱུར། །མི་བསྲུན་འགྲོ་ལ་བྱམས་སེམས་བསྐྱེད། །རྩུབ་སྤྱོད་ཁོ་ནས་གདུལ་བར་བྱ། །རང་གི་ལུས་ལ་ཕན་འདོད་རྣམས། །གཏར་དཔྱད་ཀྱིས་ནི་ནད་རྣམས་འདོན། །གནོད་པར་བྱེད་པ་ཆུང་ངུ་ཡང་། །མྱུར་དུ་བསལ་ལ་འདུམ་པར་བྱ། །ཡུར་བ་ཆུང་ངུའི་རྗེས་ཞུགས་ནས། །འོད་པ་ཆེན་པོ་མ་མཐོང་ངམ། །མཁས་པས་བྱ་བ་མ་ཡིན་པའི། །བྱ་བ་ཤེས་ཀྱང་མི་བྱའོ། །གླང་པོས་དགྲ་སྡེ་བཅོམ་པ་ཡིས། །རྒྱུན་དུ་རྒྱལ་པོས་བཅིངས་ལ་ལྟོས། །རང་ཕྱོགས་སྡང་ཡང་སྤང་མི་བྱ། །དགྲ་བོ་བྱམས་ཀྱང་འཁུ་མི་བྱ། །བྱ་རོག་བྱ་རོག་གིས་འཚེ་ཡང་། །ཉིན་མོ་ལོང་བ་བསྟེན་ན་ཕུང་། །བྱ་བ་ཆེ་ཆུང་གང་བྱེད་ཀྱང་། །མཁས་པས་རྟག་ཏུ་བསྒྲིམས་ཏེ་བྱ། །སེང་གེས་རི་བོང་གླང་ཆེན་གཉིས། །གསོད་པའི་ཚེ་ན་སྒྲིམ་ལྷོད་མེད། །ཡོན་ཏན་ལྡན་པ་མི་བཀུར་བའི། །གནས་དེར་མཁས་པ་སུ་ཞིག་སྡོད། །ཆུ་ཤེལ་མེ་རྡོར་བྱེད་པ་ཡི། །ཡུལ་དེར་ཆུ་ཤེལ་འཚོང་ངམ་ཅི། །མཁས་པས་གཞན་ལ་བཤད་པ་འམ། །ཡང་ན་ཞི་བའི་ནགས་སུ་བསྒྲུབ། །ནོར་བུ་སྤྱི་གཙུག་རྒྱན་བྱེད་དམ། །མིན་ན་རྒྱ་མཚོའི་གླིང་ལ་གནས། །བདག་བས་མཐོ་བའི་དམ་པ་ལ། །བསྙེན་ན་བདག་ལ་ཕན་པ་སྒྲུབ། །ལྷུན་པོའི་ངོས་ལ་གནས་པ་ཡི། །བྱ་རྣམས་གསེར་དུ་སྣང་བར་འགྱུར། །ཆེན་པོ་ཕྲག་དོག་ལྡན་པ་ལ། །བརྟེན་ན་ཆེན་པོ་ཐོབ་མི་འགྱུར། །ཉི་མའི་འོད་ལ་ཉེ་བ་ཡིས། །ཟླ་བ་མར་ངོར་གྱུར་ལ་ལྟོས། །མི་གང་མཛའ་བ་མི་བརྟན་པ། །དེ་དང་འགྲོགས་པར་སུ་ཞིག་འཐམ། །ནམ་མཁའི་འཇའ་ཚོན་མདོག་མཛེས་ཀྱང་། །རྒྱན་དུ་རེ་ན་བླུན་པོ་འཁྲུལ། །བདག་ཉིད་གང་ལ་མི་དགའ་བ། །གཞན་ལ་ཀུན་ཏུ་དེ་མི་བྱ། །གཞན་གྱིས་ཅུང་ཟད་གཙེས་པའི་ཚེ། །རང་ལ་བསམ་པ་ཅི་འབྱུང་སོམས། །གང་ཞིག་བདག་ཉིད་གང་དགའ་བ། །དེ་ཉིད་གཞན་ལ་སྦྱར་བྱས་ན། །གཞན་གྱིས་བདག་ལ་གང་དགའ་བ། །དེ་ཡི་ཚུལ་གྱིས་མཆོད་པར་གྱུར། །དམུ་རྒོད་རྣམས་དང་མཛའ་བ་དང་། །རྩོད་པ་གཉིས་ཀ་མཁས་པས་སྤང་། །འགྲས་པ་དང་ནི་འདྲིས་པའི་ཚུལ། །གཉིས་ཀ་བཏེག་ལ་བྱར་མི་རུང་། །དམ་པ་དག་ལ་བརྟེན་པ་དང་། །མཁས་པ་བདག་ལ་འདྲི་བ་དང་། །གཞུང་བཟང་པ་དང་འགྲོགས་པ་ནི། །སུ་ལ་ཡོད་པ་རྒྱུན་དུ་བདེ། །དུས་སུ་མ་བབ་པ་ཡི་གཏམ། །གང་སྨྲས་དེ་ལ་ཀུན་གྱིས་བརྙས། །མང་པོ་སྨྲ་བའི་གཏམ་ཚིག་ལས། །སྨྱོན་པར་རྗེས་སུ་མི་དཔོག་གམ། །ཉམ་ཆུང་རྣམས་ཀྱིས་རང་ཉིད་ཀྱི། །སྨྲ་བ་ཐམས་ཅད་འཁྲུལ་བར་བསམ། །དེ་ལྟར་གོ་ནས་མི་སྨྲ་བའི། །མི་ལ་གཞན་གྱིས་བཀུར་བ་ཉེ། །ཡུལ་དང་དུས་ལ་བབ་པའི་ཚེ། །རབ་ཏུ་བསྒྲིམས་ཏེ་ལན་འགའ་སྨྲ། །ལེགས་བཤད་ཡིན་ཡང་མངས་གྱུར་ན། །ཚོང་ཟོང་ལྷག་བཞིན་བྲིན་མི་འགྱུར། །རང་སྐྱོན་མཁས་པ་རྣམས་ཀྱིས་ཀྱང་། །སྐྱོན་དུ་གོ་བ་ཤིན་ཏུ་དཀའ། །མང་པོས་བདག་ལ་དེ་སྒྲོགས་ན། །སྐྱོན་ཅན་ཡིན་པར་རྗེས་སུ་དཔོག །གོ་ཡང་སྐྱོན་དེ་མི་འདོར་ན། །མི་དེ་གདོན་གྱིས་བཏབ་པ་ཡིན། །ད་དུང་གཉེན་པོ་མི་འཇུག་ན། །མི་ཡིན་སྙམ་དུ་མ་སེམས་ཤིག །ཅུང་ཟད་བསམ་པ་ཡོད་རྣམས་ཀྱིས། །སྐྱོན་དེ་བརྣགས་ལ་གདོན་པར་བྱ། །དེ་ལྟར་གཉེན་པོ་སྤྱོད་པ་ཡི། །མི་དེ་གོང་ནས་གོང་དུ་འགྲོ། །བློ་དང་ལྡན་པའམ་བྱམས་ཀྱང་རུང་། །བསྒོ་བ་ཉན་པའམ་དཔའ་ཡང་རུང་། །གཞན་གྱི་བྱ་བ་མི་ཤེས་ཀྱང་། །དེ་དང་དེ་ཡི་སྤྱོད་པས་བསྐྱང་། །མི་བསྲུན་པ་ཡི་འབྲེལ་བ་དང་། །རིང་ནས་འགྲོགས་ཀྱང་སྤངས་ན་བདེ། །རྟག་ཏུ་འགུལ་བའི་སོ་བཟང་ཡང་། །བཏོན་ན་བདེ་ཞེས་རྒད་པོ་སྨྲ། །རྒྱུན་དུ་འཁང་ཞིང་འཁོར་བ་ཡི། །འབྲེལ་བ་ཅུང་ཟད་བྱིན་ལ་བསྐྲད། །སྦྲུལ་གྱིས་ཟིན་པའི་རྨ་དག་ནི། །གཅོད་མ་ནུས་ན་སྲོག་དང་འབྲལ། །གལ་ཏེ་ཆེན་པོ་ཐོབ་གྱུར་ན། །ཕྲན་ཚེགས་གཞན་ལ་སེར་སྣ་སྤང་། །ཡུལ་འཁོར་དབང་དུ་འདུ་བ་ན། །དབྱིག་ལ་སེར་སྣ་བྱེད་དམ་ཅི། །མཁས་པ་ལོངས་སྤྱོད་གསོག་འདོད་ན། །ཆ་ཤས་སྦྱིན་པ་སྲུང་བ་ཆེ། །ཁྲོན་པའི་ཆུ་ནི་འཕེལ་འདོད་ན། །འཆུ་བ་ཉིད་ནི་གདམས་ངག་ཡིན། །གང་ཞིག་ཕུན་སུམ་ཚོགས་འདོད་ན། །དེ་ཡིས་འདུ་འགོད་མང་དུ་བྱ། །འདུ་འཛི་སྡུག་བསྔལ་རྒྱུར་མཐོང་ན། །ཕུན་སུམ་ཚོགས་པའི་རེ་བ་ཐོངས། །བྱ་བ་གང་ཞིག་བསྒྲུབ་ན་ཡང་། །སྐྱོན་དང་ཡོན་ཏན་གཉིས་གཉིས་སོམས། །གཉིས་ཀ་མཉམ་ཡང་སྤང་དགོས་ན། །ཉེས་པ་ཤས་ཆེ་སྨོས་ཅི་དགོས། །མཁས་པ་དྲང་པོ་གུས་པས་བསྟེན། །མཁས་པ་གཡོ་ཅན་ཤེས་ན་གཟབ། །རྨོངས་པ་དྲང་པོ་བྱམས་པས་བསྐྱང་། །རྨོངས་པ་གཡོ་ཅན་མྱུར་དུ་སྤང་། །ནོར་དང་གཡོ་འཁོར་མེད་ན་ཡང་། །མཛའ་བོ་བློ་ལྡན་གྲོགས་ཡོད་ན། །དུད་འགྲོས་ཀྱང་ནི་དོན་གྲུབ་ན། །མི་ཡི་འགྲོ་བས་སྨོས་ཅི་དགོས། །གང་ཞིག་ཤེས་པའི་བྱ་བ་བསྒོ། །མི་ནུས་པ་ལ་འགའ་མི་སྦྱར། །ཆུ་ལ་ཤིང་རྟས་མི་བགྲོད་དེ། །ཐང་ལ་གྲུ་ཡིས་ཇི་ལྟར་བགྲོད། །ཡུན་རིང་འཁོན་པའི་དགྲ་བོ་དང་། །མཛའ་བ་བྱེད་ཀྱང་བསྲེ་མི་བྱ། །ཤིན་ཏུ་ཁོལ་བའི་ཆུ་ཡིས་ཀྱང་། །མེ་དང་ཕྲད་ན་མི་གསོད་དམ། །གཞུང་བཟང་ངོ་ཚ་ཁྲེལ་ཡོད་ན། །དགྲ་བོ་ལ་ཡང་ཡིད་བརྟན་རུང་། །གཞུང་བཟང་དགྲ་ལ་སྐྱབས་སོང་བས། །སྲོག་གི་བར་དུ་བསྐྱབས་པ་ཐོས། །རང་ལ་ངན་སེམས་མེད་ཅེས་ཏེ། །ཕལ་པ་ཀུན་ལ་ཡིད་མི་བརྟན། །རི་དྭགས་རྒྱུན་དུ་སེམས་བཟང་ཡང་། །མཆེ་བ་ཅན་རྣམས་ཟས་སུ་བསྒོམ། །བླུན་པོ་ལོག་པའི་ལམ་འགྲོ་ན། །བླུན་པོ་ཉིད་དུ་གོ་བས་ཆོག །མཁས་པ་ལོག་པའི་ལམ་འགྲོ་ན། །རྒྱུ་མཚན་གང་ཞིག་དཔྱད་དགོས་སོ། །མཁས་པས་ལོངས་སྤྱོད་ལྷག་པར་སྤྱད། །མི་སྟེར་ན་ཡང་འཕྲོག་པ་མེད། །ལོངས་སྤྱོད་ཉམས་པའི་ས་ཕྱོགས་སྤང་། །མི་འཕྲོད་ན་ཡང་གཉའ་སྐྱིས་འཇོམས། །རང་གིས་ལེགས་པར་ཤེས་ན་ཡང་། །བྱ་མ་ཐམས་ཅད་གྲོས་ཀྱིས་བསྒྲུབ། །མི་གང་གྲོས་ལ་མི་དགའ་བ། །འགྱོད་པ་རིན་གྱིས་ཉོ་བ་ཡིན། །ཉམ་ངའི་གྲོས་དག་བྱེད་པ་ན། །མཛའ་ཡང་གསུམ་དུ་གྲོས་མི་བྱ། །རོ་ལངས་སྒྲུབ་པའི་གྲོགས་ངན་ན། །ཐོག་མར་སྒྲུབ་པོ་ཟ་བར་འགྱུར། །ལེན་ཤེས་སྙོམས་པར་གཏོང་ལ་མཁས། །དུལ་ཞིང་ཀུན་གྱི་ངོ་བསྲུང་བ། །འཇིགས་མེད་བྱས་པ་ཆུད་མི་གཟན། །མི་དེ་འདོད་ན་ས་ཀུན་ཐོབ། །གལ་ཏེ་དགྲ་བོ་སྐྱབས་འོང་ན། །དེ་ལ་མཆོད་ནས་སྙན་པ་བརྗོད། །ཁྭ་ཏས་བྱི་བ་ལ་བརྟེན་ནས། །བདེ་བ་ཐོབ་ཅེས་སྔོན་རབས་སྨྲ། །གྲོགས་པོ་ངན་དང་ཐོས་པ་ངན། །རྟོག་པ་ངན་དང་བྱ་བ་ངན། །མཁས་པ་རྣམས་ཀྱིས་མི་བྱ་སྟེ། །བྱེད་ན་བླུན་པོ་གུད་ན་མེད། །ལེགས་པར་བརྟགས་ནས་བྱེད་པ་ལ། །བྱ་བ་ཉམས་པ་ག་ལ་སྲིད། །མིག་ལྡན་བརྟགས་ནས་འགྲོ་བ་ན། །གཡང་སར་གོམ་པ་འདོར་རམ་ཅི། །རང་གཞན་གཉིས་ཀ་འཕེལ་འགྱུར་བའི། །ཡོན་ཏན་སློབ་ན་མཁས་པའི་རྟགས། །རིག་པ་འགའ་ཞིག་མདའ་མཁན་ལྟར། །ཤེས་པ་ཉིད་ན་ཆོ་འབྲང་འཇིག །རང་ཉིད་ཅིས་ཀྱང་མཐོ་འདོད་ན། །གཞན་ལ་ཕན་པ་འབའ་ཞིག་གྱིས། །བྱད་ལ་བྱི་དོར་བྱེད་པ་རྣམས། །སྔོན་དུ་མེ་ལོང་མི་འབྱིད་དམ། །ཕ་རོལ་ཅི་ནས་གཞོམ་འདོད་ན། །རང་གི་ཡོན་ཏན་འབད་པས་སྒྲུབ། །དགྲ་བོ་གསོད་པར་འདོད་པ་རྣམས། །མཚོན་ཆ་འབད་ནས་སྒྲུབ་ལ་ལྟོས། །གཡོན་ཅན་ཚིག་ལ་དྲོན་འཇུག་པས། །དྲང་པོས་དེ་ལ་བརྟག་དཔྱད་དགོས། །བསླུས་རྗེས་ཁོ་བོ་དྲང་ངོ་ཞེས། །རང་ལ་བསྟོད་པས་ཅི་ཞིག་ཕན། །གཡོན་ཅན་སོགས་ལ་གཡོ་སོགས་དང་། །དྲང་པོ་སོགས་ལ་དྲང་པོ་སྟེ། །མི་བརྟན་པ་ལ་བརྟན་དགོས་ཞེས། །བྱ་བ་སྔོན་གྱི་ཡིག་ཚངས་ཡིན། །མཁས་པས་རྟག་ཏུ་དགྲ་ལ་ཡང་། །གཉེན་དེས་རྐྱེན་དུ་བྱས་ན་ལེགས། །དེ་ཡིས་འདུམ་ལ་མི་ཕན་ཡང་། །ངེས་པར་གནོད་སེམས་འཇིལ་བའི་སྨན། །ཚིག་ངན་བརྗོད་པས་འཇིག་རྟེན་ན། །འདོད་པ་འགྲུབ་པ་མི་སྲིད་ཀྱིས། །ཡིད་ལ་རང་འདོད་བསྒྲུབ་ན་ཡང་། །ངག་གིས་ཀུན་དང་མཐུན་པར་གྱིས། །རང་དང་གཞན་གྱི་དོན་གྲུབ་ན། །འཇམ་རྩུབ་ཇི་ལྟར་བྱས་ཀྱང་རུང་། །ཐབས་ལ་མཁས་པའི་སྤྱོད་པ་ལ། །གཡོ་སྒྱུ་ཐུབ་པས་མ་གསུངས་སོ། །ཐ་མར་ཤིན་ཏུ་ཕན་པའི་གྲོས། །འཕྲལ་ལ་སྡུག་ཀྱང་མཁས་པས་བསྟེན། །རྒས་ཤིང་གདུངས་ན་ཡོན་ཏན་རྣམས། །བུ་གཅིག་པ་ལ་མཁས་རྣམས་སློབ། །ལོངས་སྤྱོད་ཧ་ཅང་སྤེལ་དྲགས་ན། །དེ་ལ་རྒུད་པ་ཤིན་ཏུ་ཉེ། །ཆུ་ཡིས་ཤིན་ཏུ་གང་བ་ཡི། །རྫིང་བུ་སྟོང་ངམ་ཡང་ན་འབྲལ། །སྐྱེ་བོ་འགའ་ལ་ཕན་པའི་ལས། །ལ་ལས་དེ་ལ་འཇུག་མི་རུང་། །སྒོག་སྐྱ་རླུང་ལ་ཕན་མོད་ཀྱི། །མཁྲིས་པའི་ནད་ལ་དུག་ཏུ་འགྱུར། །ཉམ་ཆུང་རྟག་ཏུ་བསྐྱང་དགོས་དང་། །འགྲོགས་ན་ཆེན་པོ་ཉམས་དོགས་ཡོད། །ཞུན་མར་སྣོད་དུ་བཞག་གྱུར་ན། །སྣོད་ལ་བྱི་བས་མི་རྨུག་གམ། །ངན་པ་རྣམས་ལ་བརྟེན་བྱས་ན། །ངན་པའི་ཤན་གྱིས་རང་ལ་གནོད། །ཡུར་བའི་ཆུ་ལ་བརྟེན་པ་ཡི། །ཉ་རྣམས་ཞིང་ལ་འགྲེམས་ལ་ལྟོས། །གཞུང་ངན་བདག་ལ་བརྟེན་འཆའ་ན། །དེ་ལ་ཅུང་ཟད་བྱིན་ཏེ་བསྐྲད། །ཁྱིམ་དུ་ལྟས་ངན་བྱུང་གྱུར་ན། །ནོར་ལས་ཕྲིས་ཏེ་བཟློག་པ་དགོས། །དམ་པ་གཞན་དུ་འགྲོ་ན་ཡང་། །བསྙེན་བཀུར་བྱས་ཏེ་ཉེ་བར་བཞག །ནོར་བུ་རིན་ཆེན་རྟག་མཆོད་ན། །བཀྲ་ཤིས་དཔལ་དང་བསམ་པ་འགྲུབ། །དམན་པའི་སྐྱེ་བོ་བསྟོད་དྲགས་ན། །ཕྱི་ནས་དེ་ཉིད་ཁྱད་དུ་གསོད། །མཁའ་ལ་མི་གཙང་གཏོར་གྱུར་ན། །འཐོར་བ་པོ་ཡི་སྤྱི་བོར་བབ། །སྐྱེ་བོ་བློ་ལྡན་མགོ་སྨད་ན། །སྨོད་བྱེད་ཉིད་ལ་ཉེས་པ་འཕོག །མར་མེ་ཐུར་དུ་ཁ་བསྟན་ན། །སྟོན་པ་པོ་ཡི་ལག་པ་སྲེག །གང་དང་གང་གི་གནས་ཡིན་པར། །དེ་དང་དེ་ཡི་གནས་སུ་བཞག །གཙུག་རྒྱན་རྐང་པར་མི་གདགས་ཏེ། །རྐང་པའི་རྐང་རྒྱན་སྤྱི་གཙུག་མིན། །བྱ་བ་ཆེན་པོ་སྒྲུབ་པའི་ཚེ། །འབད་པ་ཡིས་ནི་གྲོགས་བཟང་བསྟེན། །ནགས་ཚལ་ཆེན་པོ་མེས་བསྲེག་ན། །རླུང་ཞིག་ཅི་ནས་གྲོགས་སུ་དགོས། །བྱམས་པས་ཁྱབ་པའི་ཚིག་སྨྲ་བ། །བསྒྲུབ་སླ་གཞན་དགའ་སྡུད་པའི་མཆོག །ནོར་གྱིས་མགུ་བར་སུ་ཡིས་ནུས། །ལུས་སྲོག་བཏང་ཡང་ཕྱེད་མི་ཚིམ། །ཕོངས་ཀྱང་གདུང་བར་མི་བྱ་སྟེ།
