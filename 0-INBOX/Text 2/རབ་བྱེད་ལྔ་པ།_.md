རབ་བྱེད་ལྔ་པ། ངན་སྤྱོད་བརྟག་པ།

།གཡོན་ཅན་སྙན་པར་སྨྲ་བ་ནི། །རང་དོན་ཡིན་གྱི་གུས་ཕྱིར་མིན། །སྲིན་བྱ་སྙན་ནས་དགོད་པ་ནི། །ལྟས་ངན་གཏོད་ཀྱི་དགའ་ནས་མིན། །ངན་པས་ཐོག་མར་ཚིག་གིས་འབྲིད། །བག་ཕེབས་གྱུར་ན་ཕྱི་ནས་བསླུ། །ཉ་པས་ཟས་ཀྱིས་ཁ་བྲིད་ནས། །ཉ་རྣམས་གསོད་པར་བྱེད་ལ་ལྟོས། །དམན་པ་ཇི་སྲིད་ཉམ་ཆུང་བ། །དེ་ཡི་བར་ལ་རང་བཞིན་བཟང་། །དུག་གི་ཚེར་མ་མ་སྨིན་པའི། །བར་དུ་གཞན་ལ་འབིགས་མི་ནུས། །ཡིད་ལ་བསམ་པ་གཞན་བྱས་ནས། །ཚིག་ཏུ་སྨྲ་བ་གཞན་བྱེད་པ། །གཡོན་ཅན་ཞེས་སུ་གྲགས་པ་སྟེ། །བླུན་པོ་མཁས་པར་འཆོས་པ་ཡིན། །ཧ་ཅང་གཡོ་སྒྱུ་མང་དྲགས་ན། །རེ་ཞིག་གྲུབ་ཀྱང་ཐ་མར་བརླག །གཟིག་ལྤགས་བཀབ་པའི་བོང་བུ་ཡིས། །ལོ་ཏོག་ཟོས་མཐར་གཞན་གྱིས་བསད། །ཤེས་རབ་ལྡན་ན་མངོན་སུམ་གྱི། །རྫུན་གྱིས་ཕ་རོལ་བསླུ་བར་ནུས། །རྐུན་པོ་རྣམས་ཀྱིས་ཁྱིར་བཏགས་པས། །ར་སྐྱེས་ཁུར་བ་བྲམ་ཟེས་བོར། །ཉེས་པ་མངོན་སུམ་བྱེད་བཞིན་དུ། །གཡོན་ཅན་ཚིག་གིས་ཕ་རོལ་བསླུ། །ལྷ་ཆེན་སྨྲེ་སྔགས་ཤོར་བ་ལ། །སྡུག་བསྔལ་བདེན་པ་བསྒྲགས་སོ་ལོ། །གཡོ་ཅན་ཚུལ་གཟོབ་ངག་འཇམ་ལ། །མ་བརྟགས་བར་དུ་ཡིད་མི་བརྟན། །རྨ་བྱ་གཟུགས་མཛེས་སྒྲ་སྙན་ཡང་། །དེ་ཡི་ཁ་ཟས་དུག་ཆེན་ཡིན། །གཡོ་ཅན་བཟང་པོའི་ཚུལ་བཟུང་ན། །ཕྱི་ནས་དོན་ལ་བསླུ་བ་ཡོད། །རི་དྭགས་རྔ་མ་བསྟན་ནས་ནི། །ཁྲེལ་འདས་བོང་བུའི་ཤ་དག་འཚོང་། །ངོ་ཚ་ཆུང་རྣམས་གཞན་ནོར་ལ། །བདག་གི་ཆེ་བ་སྒྲུབ་པ་ཡོད། །མགྲོན་ལ་གྲོགས་ཀྱི་གོས་བཏིང་ནས། །བདག་གི་གུས་རྟགས་སྟོན་པར་བྱེད། །གཏིང་ནས་ཉམས་པའི་ཁྲེལ་འདས་རྣམས། །རང་གི་གཏམ་ངན་ངོ་སོར་འཁྱེར། །ཀཉྩའི་ཡུལ་གྱི་རྒྱལ་རིགས་འགའ། །ཕ་བསད་པ་ལ་རྒྱལ་རྔ་རྡུང་། །བློ་ཆུང་རྣམས་ཀྱིས་ཕན་བཏགས་པ། །འགའ་ཞིག་གནོད་པ་ཆེན་པོར་འགྱུར། །སྐྱ་ཀའི་ཕྲུ་གུས་མ་ཡི་སྤུ། །འཐོག་པ་དྲིན་ལན་ཡིན་པར་སྒོམ། །གཞུང་ངན་གཞན་གྱི་བཀའ་དྲིན་ལས། །བྱུང་བ་རང་གི་ཆེ་བར་སྒོམ། །ཀླུ་ཡིས་འབད་ནས་ཕབ་པའི་ཆར། །ཞིང་པ་རང་གི་དཔལ་དུ་རྩི། །བློ་ངན་ལས་ལ་སྤྱོད་པ་ལ། །རང་གི་འབད་པས་ཐོབ་པར་སེམས། །ཁྱི་རྒན་ཡ་རྐན་ཕུག་པའི་ཁྲག །རུས་པའི་བཅུད་དུ་བསྒོམས་ནས་འཆའ། །མི་མཁས་པ་དག་གཉེན་བཅོམ་ནས། །འབྲེལ་མེད་གཞན་དག་གསོ་བར་མཐོང་། །མགོ་བོ་བཅད་ནས་མཇུག་མའི་རྒྱན། །སྨྱོན་པ་མིན་པ་སུ་ཞིག་བྱེད། །བླུན་པོས་དགོས་པའི་སར་མི་བྱིན། །མི་དགོས་པ་ལ་ལྷུག་པར་འཆོར། །ཆུ་མིག་ངན་པ་དབྱར་འབབ་ཀྱི། །དཔྱིད་དུས་དགོས་པའི་ཚེ་ན་སྐམ། །སྐྱེ་བོ་དམ་པ་ཞི་བ་ལ། །ངན་པ་ལྷག་པར་འཚེ་བར་བྱེད། །སྣུམ་དང་ལྡན་པའི་སྡོང་བུ་ལ། །མེ་ལྕེ་ཆགས་ཀྱི་གཞན་ལ་མིན། །རྩུབ་པས་རྩུབ་པ་ཐུལ་འགྱུར་གྱི། །ཞི་བས་འདུལ་བར་ག་ལ་ནུས། །ཕོལ་མིག་བསྲེགས་དང་བཅད་པས་འདོན། །ཞི་ཆོས་དེ་ཡི་དུག་ཏུ་འགྱུར། །རྒྱལ་པོས་ཆོས་བཞིན་ས་བསྐྱང་སྟེ། །མིན་ན་རྒྱལ་པོ་ཉམས་པའི་རྟགས། །ཉི་མས་མུན་པ་མི་སེལ་ན། །དེ་ནི་གཟའ་ཡིས་ཟིན་པའི་རྟགས། །སྡིག་སྤྱོད་རྗེ་བོར་བསྐོས་པ་དང་། །ཁང་པའི་ཡང་ཐོག་གྲུམ་པ་དང་། །རི་རྩེ་ཤིན་ཏུ་ཞིག་པ་ཡི། །འོག་ན་གནས་རྣམས་རྒྱུན་དུ་སྐྲག །གལ་ཏེ་རིག་པ་ཡོད་ན་ཡང་། །སྐྱེ་བོ་རང་བཞིན་ངན་པ་སྤང་། །དུག་སྦྲུལ་ནོར་བུས་མགོ་བརྒྱན་ཡང་། །མཁས་པ་སུ་ཞིག་པང་དུ་ལེན། །བསྙེན་དྲགས་རྒྱལ་པོ་བཞེད་བཞིན་དཀའ། །བཀྱེད་ན་ཐུགས་ཀྱིས་འཛིན་པ་དཀོན། །བར་མར་གནས་པ་གཡང་ཟ་བས། །རྒྱལ་པོ་ལ་ལ་མེ་ཆེན་འདྲ། །ཆུང་མ་ངན་དང་མཛའ་བོ་ངན། །རྒྱལ་པོ་ངན་པ་སུ་ཞིག་བསྟེན། །གཅན་གཟན་གདུག་པའི་ནགས་ཁྲོད་དུ། །སེམས་ཤེས་རྟག་ཏུ་སྡོད་དམ་ཅི། །དྲེགས་པས་ཡོན་ཏན་ཉམས་འགྱུར་ཞིང་། །འདོད་པས་ངོ་ཚ་ཉམས་པར་བྱེད། །གཡོག་འཁོར་རྟག་ཏུ་སུན་ཕྱུང་ན། །རྗེ་དཔོན་ཉམས་པར་འགྱུར་བ་ཡིན། །ཕན་པར་སྨྲ་བ་དཀོན་པ་སྟེ། །དེ་དག་བས་ཀྱང་ཉན་པ་དཀོན། །སྨན་པ་མཁས་པ་རྙེད་དཀའ་སྟེ། །དེ་ཡི་ཚིག་བཞིན་བྱེད་པ་ཉུང་། །ཧ་ཅང་རློམ་པ་ཆེས་དྲགས་ན། །སྡུག་བསྔལ་བརྒྱུད་མར་ཐོག་ཏུ་བབ། །སེང་གེ་ང་རྒྱལ་ཆེས་པའི་རྒྱུས། །ཝ་ཡིས་ཁུར་པོ་བསྐུར་ཞེས་ཐོས། །བྱ་རོག་དག་གིས་བསྐུངས་པ་དང་། །གཞུང་ངན་པ་ལ་ཕན་བཏགས་དང་། །ཞིང་ངན་ས་བོན་བཏབ་པ་རྣམས། །རེ་བ་མང་སྟེ་སྤྱོད་པ་ཉུང་། །མ་བརྟགས་པར་ནི་སུ་ལ་ཡང་། །ཡིད་ཆེས་པ་དང་བསྒོ་མི་བྱ། །བག་ཕབ་པ་ལས་སྐྱོན་བྱུང་དང་། །གྲོས་བཏབ་པ་ལས་དགྲར་གྱུར་མང་། །འཇིག་རྟེན་ངན་པ་མང་ཡོད་ཀྱང་། །སྐྱེ་བོ་ངན་འདྲའི་འཇིགས་པ་མེད། །ངན་པ་གཞན་དག་བཅོས་རུང་གི། སྐྱེ་བོ་ངན་པ་བཅོས་ན་འཇིག །སྐྱེ་བོ་ངན་པ་ཡོན་ཏན་བརྒྱས། །ཕན་པར་བྱས་ཀྱང་མགུ་མི་ནུས། །གང་ཞིག་བརྟེན་པ་དགྲར་འགྱུར་འདི། །སྐྱེ་བོ་ངན་པའི་ཁྱད་ཆོས་ཡིན། །མཆིལ་པ་ཇི་ལྟར་གསོས་གྱུར་ཀྱང་། །བག་ཕེབས་འགྱུར་བ་མི་སྲིད་དོ། །རྟག་ཏུ་འབྱེད་ལ་བརྩོན་པའི་མིས། །མཛའ་བོ་བརྟན་པོའང་ཕྱེད་འགྱུར་ཏེ། །ཆུ་ཡིས་རྟག་ཏུ་བཅོས་གྱུར་ན། །བྲག་ལ་སེར་ག་མི་འབྱུང་ངམ། །དགྲ་བོ་དེ་དག་གཞོམ་པར་སླ། །ཕན་པའི་བཟློག་རྟགས་ཁར་འབྱིན་པའི། །དགྲ་བོ་དེ་དག་ཇི་ལྟར་གཞོམ། །རྨ་གཞན་མཁས་ན་འཚོ་སྲིད་ཀྱི། །ཉེས་པའི་ཚིག་རྨ་འདྲུབ་པ་མེད། །ཁྭ་ཡིས་འུག་པ་རྨ་ཕབ་པས། །བསྐལ་པའི་བར་དུ་འགྲས་ལ་ལྟོས།
