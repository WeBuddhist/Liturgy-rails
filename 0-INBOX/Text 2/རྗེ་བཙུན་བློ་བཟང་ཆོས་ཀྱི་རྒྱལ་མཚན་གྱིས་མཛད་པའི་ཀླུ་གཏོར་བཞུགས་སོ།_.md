༄༅། །རྗེ་བཙུན་བློ་བཟང་ཆོས་ཀྱི་རྒྱལ་མཚན་གྱིས་མཛད་པའི་ཀླུ་གཏོར་བཞུགས་སོ། །

༄༅། ། ན་མོ་གུ་རུ་ལོ་ཀེ་ཤཱ་རཱ་ཡ། འདིར་ཀླུ་གཏོར་གཏོང་བར་འདོད་ པས། དཀར་གསུམ་མངར་གསུམ་དར་སྣ་རིན་པོ་ཆེའི་རིགས་དང་འབྲུ་སྣ་ཀླུ་སྨན་སོགས་གཏོར་སྣོད་དུ་ཆུ་གཙང་དང་སྦྱར་བ་ལག་ལེན་བཞིན་བཤམས།། སྐྱབས་འགྲོ་སེམས་བསྐྱེད་སྔོན་དུ་གཏང་། རང་ཆོ་ག་གསུམ་སྐྱེད་ཀྱི་སྒོ་ནས་ སྤྱན་རས་གཟིགས་སུ་བསྐྱེད། ཡེ་ཤེས་པ་བཅུག དབང་བསྐུར་རྒྱས་བཏབ་ སོགས་བརྒྱ་རྩའི་ཐོག་མར་བྱ་བ་ཇི་བཞིན་དུ་བྱེད། དེ་ནས་ཀླུ་གཏོར་གྱི་རྫས་དེ་ དག ། ཨ་མྲྀ་ཏས་བསངས། སྭ་བྷ་ཝས་སྦྱངས། སྟོང་པའི་ངང་ལས་ཧཱུཾ་ལས་ རིན་པོ་ཆེ་ལས་གྲུབ་པའི་སྣོད་ཡངས་ཤིང་རྒྱ་ཆེ་བ་རྣམས་ཀྱི་ནང་དུ་མཆོད་ གཏོར་མཐུན་པའི་དམ་རྫས་དང་བཅས་པ་རྣམས། ཨོཾ་དང་རང་རང་གི་མིང་ ཡིག་ཐིག་ལེས་བརྒྱན་པ་ལས་བྱུང་བའི་ཡེ་ཤེས་ཀྱི་བདུད་རྩི་དང་། སྡུག་བསྔལ་ ཐམས་ཅད་སེལ་བའི་སྨན་དང་། དགོས་འདོད་འབྱུང་བའི་ཡིད་བཞིན་གྱི་ ནོར་བུའི་ཚོགས་དང་། ཡིད་དུ་འོང་བའི་གནས་དང་ཕོ་བྲང་ལ་ སོགས་པ་ མདོར་ན། ཀླུ་རྒྱལ་འཁོར་དང་བཅས་པ་ལ་གང་ལ་གང་འདོད་པའི་ལོངས་ སྤྱོད་ཕུན་སུམ་ཚོགས་པར་གྱུར།

སྔགས་དྲུག་དང་ཕྱག་རྒྱ་དྲུག་དང་། ཨོཾ་ཨརྒྷཾཨཿ ཧཱུཾ། ཞེས་པ་ནས། ཤཔྟ་ཨཱཿ ཧཱུཾ། ཞེས་པའི་བར་གྱིས་བྱིན་བརླབས། དེ་ ནས་མགྲོན་བསྐྱེད་ཅིང་སྤྱན་དྲང་བ་ནི། རང་གི་མདུན་དུ་རིན་པོ་ཆེ་ལས་གྲུབ་ པའི་བྲག་གསེར་གྱི་བྱ་སྐྱིབས་ཅན་གྱི་འོག་ཏུ། མཚོ་དང་། རྫིང་བུ་དང་། ལྟེང་ཀ་དང་། ནེའུ་སིང་མེ་ཏོག་གིས་བརྒྱན་པ་དང་། ཆུ་མིག་དང་། ཁྲོན་པ་ དང་། འབབ་ཆུ་དང་། ལུ་མ་དང་། སྤང་ལྗོངས་དང་། སྐྱེད་མོས་ཚལ་ལ་ སོགས་པ་ཀླུའི་གནས་དང་། ཕོ་བྲང་རིན་པོ་ཆེ་ལས་གྲུབ་པ་ཡིད་དུ་འོང་བ་ རྣམས་ཀྱི་ནང་དུ་ཀླུ་རྒྱལ་ཝ་རུ་ཎས་གཙོ་བྱས། ཀླུ་ཆེན་པོ་བརྒྱད་དང་ཀླུ་དང་ ས་བདག་མ་ལུས་པའི་ཚོགས་ཀྱི་འཁོར་གྱིས་བསྐོར་བ་འཁོད་པར་གྱུར།

རང་ གི་ཐུགས་ཀའི་ཧིའི་འོད་ཟེར་གྱིས་རྒྱ་མཚོ་ཆེན་པོ་ནས་ཀླུ་རྒྱལ་འཁོར་བཅས་ སྤྱན་དྲངས་པར་བསམས་ལ་ངག་ཏུ༑ གདེངས་ཀ་ནོར་བུ་འབར་བའི་འོད་ཟེར་ གྱིས། ། ས་འོག་མུན་པ་མ་ལུས་སེལ་མཛད་པའི། ། བསྟན་ལ་རབ་དགའ་ མཐུ་སྟོབས་བསམ་མི་ཁྱབ། ། ཀླུ་རྒྱལ་མ་ལུས་འཁོར་བཅས་གནས་འདིར་ གཤེགས། ། ཨོཾ་ནཱ་ག་རཱ་ཛ་ས་པ་རི་ཝཱ་ར་ཨེཧྱེ་ཧི་ས་མ་ཡ་ཛཿ ཛཿ ཛཿ ཧཱུཾ་བཾ་ ཧོཿ པདྨ་ཀ་མ་ལ་ཡེ་སྟྭཾ། ནཱ་ག་རཱ་ཛ་ས་པ་རི་ཝཱ་ར་ཨརྒྷཾ། ཞེས་སོགས་ཀྱིས་ མཆོད། དེ་ནས་རང་ཉིད་སྤྱན་རས་གཟིགས་སུ་གསལ་བའི་ཕྱག་གི་མཐིལ་ལས་ བདུད་རྩི་འོ་མའི་རྒྱུན་ལྟ་བུ་བབས་པ་དང་འདྲེས་པའི། མདུན་གྱི་མཆོད་ གཏོར་དམ་རྫས་རིན་པོ་ཆེ་དང་སྨན་སྣ་ཚོགས་ཡེ་ཤེས་ལས་གྲུབ་པའི་འདོད་ དགུའི་རྒྱ་མཚོ་ཆེན་པོ་ལ་མགྲོན་རྣམས་ཀྱིས་ལོངས་སྤྱད་པས་ལུས་དང་ཡན་ལག་དང་དབང་པོ་ཉམས་པ་གསོས། སྡུག་བསྔལ་དང་སྡིག་སྒྲིབ་དག ཟག་མེད་ ཀྱི་བདེ་བས་རྒྱུད་གང་བར་བསམས་ལ། ལག་པ་གཡས་པས་མཆོད་གཏོར་ནང་དུ་ཆུ་རྒྱུན་སྒྲེང་། གཡོན་པས་སེ་གོལ་གདབ། ཨོཾ་བཱ་སུ་ཀི་མཾ་སྭཱཧཱ། ཞེས་བརྒྱ་རྩ་བཟླ་བ་ནི་ཀླུ་གཏོར་གྱི་དངོས་གཞིའོ། །

དེ་ནས།ཀླུ་ཚོགས་ཀུན་གྱི་རྗེར་ གྱུར་ཅིང་༑ ༑

གདེངས་ཀ་ནོར་བུའི་འོད་ཟེར་གྱིས། །

འདོད་དགུའི་དངོས་ གྲུབ་མཆོག་སྩོལ་བའི། །

ཝ་རུ་ཎ་ལ་ཕྱག་འཚལ་ལོ། །

མཐའ་ཡས་དང་ནི་འཇོག་པོ་དང་། །

སྟོབས་ཀྱི་རྒྱུ་དང་རིགས་ལྡན་དང་། །

ནོར་རྒྱས་དང་ནི་དུང་སྐྱོང་དང་། །

པདྨ་དང་ནི་ཆུ་ལྷ་དང་། །

དགའ་བོ་དང་ནི་ཉེར་དགའ་བོ། །

རྒྱ་མཚོ་དང་ནི་རྒྱ་མཚོ་ཆེ། །

དཔལ་གྱི་འོད་དང་འོད་ཆེན་དང་། །

གཟུགས་ཆེན་སྦྲུལ་བཟང་སྟོབས་པོ་ཆེ། །

ཀླུ་ཡི་རྒྱལ་པོ་འཁོར་དང་བཅས། །

འདིར་གཤེགས་བརྒྱན་པའི་གཏོར་མ་བཞེས་༑ ༑

མཛེས་པའི་དཀར་གསུམ་མཆོད་པ་དང་། །

མངར་གསུམ་འོ་མའི་རྒྱ་མཚོ་འཁྱིལ། །

རིན་ཆེན་མཛེས་པའི་ཐེམ་སྐས་ཅན། །

ཀླུ་ཡི་སྡུག་བསྔལ་སྐྱོབ་བྱེད་པའི། །

མཐུན་པའི་རྫས་སུ་འདི་གྱུར་ཅིག༑

མཆོད་སྦྱིན་གཏོར་མ་འདི་བཞེས་ལ། །

བདག་ཅག་ཡོན་མཆོད་འཁོར་བཅས་ལ། །

བཞི་བརྒྱ་རྩ་བཞིའི་ནད་རྣམས་དང་། །

ཀླུ་ལས་གྱུར་པའི་ནད་ཀྱི་རིགས། །

མཐོང་བའི་དུག་དང་རེག་པའི་དུག་༑

བསམ་ངན་དུག་དང་ཁ་རླངས་དུག །

སྦྱོར་བའི་དུག་ལ་སོགས་པ་ཀུན། །

མ་ལུས་མྱུར་དུ་ཞི་གྱུར་ཅིག །

བདག་ལས་གཅེས་པའི་ རྫས་ཁྱེར་ལ། །

ཀླུ་ཡི་ནད་རྣམས་ཞི་བར་མཛོད། །

འཕྲིན་ལས་རྣམ་བཞིའི་དངོས་གྲུབ་དང་། །

དཔལ་འབྱོར་ལོངས་སྤྱོད་རྒྱས་པར་མཛོད༑ ༑
