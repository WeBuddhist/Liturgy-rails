རྡོ་རྗེ་རྣམ་པར་འཇོམས་པའི་སྒོ་ནས་དཀར་ཕྱོགས་རྣམས་ལ་རི་ཁྲུས་ཀླུང་ཁྲུས་བཅས་བཞུགས་སོ། །

(༄༅། །འདིར་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པའི་སྒོ་ནས་དཀར་ཕྱོགས་སྐྱོང་བའི་སྲུང་མ་ལྷ་ཀླུ་གཞི་བདག་རྣམས་དང་རི་ཀླུང་གནས་ཁང་སོགས་ལ་ཉམས་གྲིབ་སེལ་ཕྱིར་ཁྲུས་ཀྱི་ཆོ་ག་བྱེད་པར་འདོད་པས་མེ་ལོང་དང་ཁྲུས་གཞོང་བུམ་པ་དྲི་བཟང་གི་ཆུས་བཀང་བ་ཁ་རྒྱན་བཅས་ཁྲུས་ཀྱི་ཡོ་བྱད་དྲི་གླུད་རྣམས་ཚོགས་པར་བྱས་ལ། སྐྱབས་འགྲོ་སེམས་བསྐྱེད་ཚད་མེད་བཞི་སྔོན་དུ་བཏང༌། )སྭ་བྷཱ་ཝས་སྟོང་པར་སྦྱངས། སྟོང་པའི་ངང་ལས་སྣ་ཚོགས་པདྨ་དང་ཉི་མའི་སྟེང་དུ། རང་གི་སེམས་ཉིད་ཧཱུཾ་ལས་སྣ་ཚོགས་རྡོ་རྗེ་ཧཱུཾ་གིས་མཚན་པ་དེ་ལས་འོད་འཕྲོས། དོན་གཉིས་བྱས། ཚུར་འདུས་ཡོངས་སུ་གྱུར་པ་ལས། རང་ཉིད་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པ་སྐུ་མདོག་ལྗང་སྔོན་ཞལ་གཅིག་ཕྱག་གཉིས་སྤྱན་གསུམ་པ་ཕྱག་གཡས་སྣ་ཚོགས་རྡོ་རྗེ་ཐུགས་ཀར་འཛིན་ཅིང༌། ཕྱག་གཡོན་སྣ་ཚོགས་རྡོ་རྗེས་ཡུ་བ་བྱས་པའི་དྲིལ་བུ་དཀུར་བརྟེན་པ། དབུ་སྐྲ་རལ་པའི་ཐོར་ཚུགས་ཅན། རིན་པོ་ཆེའི་རྒྱན་གྱིས་བརྒྱན་ཅིང༌། དར་གྱི་ན་བཟའ་གསོལ་པ། འོད་ཀྱི་དཀྱིལ་འཁོར་གྱི་དབུས་ན་རྡོ་རྗེའི་སྐྱིལ་ཀྲུང་གིས་བཞུགས་པའི་སྤྱི་བོར་ཨོཾ། མགྲིན་པར་ཨཱཿཐུགས་ཀར་ཧཱུཾ། དེ་ལས་འོད་འཕྲོས། རང་བཞིན་གྱི་གནས་ནས་བསྒོམས་པ་དང་འདྲ་བའི་ཡེ་ཤེས་པ་དང༌། དབང་གི་ལྷ་རྣམས་སྤྱན་དྲངས་ཛཿཧཱུཾ་བཾ་ཧོཿདབང་གི་ལྷས་དབང་བསྐུར། མི་བསྐྱོད་པས་དབུ་བརྒྱན་པར་གྱུར། ཨོཾ་སརྦ་ཏ་ཐཱ་ག་ཏ་ཨརྒཾ་ྷནས་ཤཔྟའི་བར་གྱིས་མཆོད།

རྒྱལ་བ་ཀུན་གྱི་མཁྱེན་བརྩེའི་ལྷ་ལམ་ལས། །

མཐུ་ཆེན་སྟོབས་ཀྱི་དབྱར་སྐྱེས་ཕྲེང་བ་ཅན། །

འཕྲིན་ལས་རྣམ་བཞིའི་གྲུ་ཆར་འབེབས་མཁས་པ། །

དཔལ་ལྡན་ཁྲོ་བོའི་དབང་པོ་ལ་ཕྱག་འཚལ། ། (ཞེས་པས་བསྟོད།) རང་རྣམ་འཇོམས་སུ་གསལ་བའི་ཐུགས་ཀར་ཉི་གདན་གྱི་དབུས་སུ་ཧཱུཾ་ཡིག་གི་མཐར་སྔགས་འཕྲེང་གིས་བསྐོར་བ་ལས་འོད་ཟེར་འཕྲོས། རང་གཞན་གྱི་ནད་གདོན་སྡིག་སྒྲིབ་དང༌། མི་གཙང་བ་ལ་སོགས་པ་ཐམས་ཅད་སྦྱངས། སངས་རྒྱས་སྲས་བཅས་ཀྱི་བྱིན་རླབས་ཐམས་ཅད་འོད་ཟེར་གྱི་རྣམ་པར་བསྡུས་ནས་རང་གི་ལུས་ལ་ཐིམ་པར་བསམས་ལ། (གཟུངས་རིང་ནི།) ༈ན་མོ་རཏྣ་ཏྲ་ཡཱ་ཡ། ན་མཤྩ་ཎྜ་བཛྲ་པཱ་ཎ་ཡེ། མཧཱ་ཡཀྵ་སེ་ནཱ་པ་ཏ་ཡེ། ཏདྱ་ཐཱ། ཨོཾ་ཏྲུ་ཌ་ཏྲུ་ཌ། ཏྲོ་ཊ་ཡ། ཏྲོ་ཊ་ཡ། སྥུ་ཊ། སྥུ་ཊ། སྥོ་ཊ་ཡ། སྥོ་ཊ་ཡ། གྷཱུརྞ། གྷཱུརྞ། གྷཱ་ཱུརྞཱ་པ་ཡ། གྷཱུརྞཱ་པ་ཡ། སརྦ་སཏྭཱ་ནི། བོ་དྷ་ཡ། བོ་དྷ་ཡ། སཾ་བོ་དྷ་ཡ། སཾ་བོ་དྷ་ཡ། བྷྲ་མ་བྷྲ་མ། སཾ་བྷྲ་མ་ཡ། སཾ་བྷྲ་མ་ཡ། སརྦ་བྷུ་ཏ་བོ་དྷ་ནི། ཀུ་ཊ། ཀུ་ཊ། སཾ་ཀུ་ཊ་ཡ། སཾ་ཀུ་ཊ་ཡ། སརྦ་ཤ་ཏྲཱུཾ། གྷ་ཊ། གྷ་ཊ། སཾ་གྷ་ཊ་ཡ། སཾ་གྷ་ཊ་ཡ། སརྦ་བིདྱཾ། བཛྲ་བཛྲ། སྥོ་ཊ་ཡ། བཛྲ་བཛྲ། ཀ་ཊ། བཛྲ་བཛྲ། མ་ཊ། བཛྲ་བཛྲ། མ་ཐ། བཛྲ་བཛྲ། ཨཊྚ་ཧཱ་ས་བཛྲ་ནཱི་ལ་བཛྲ། སུ་བཛྲ་ཡ་སྭཱཧཱ། ཧེ་ཕུ་ལུ། ནི་རུ་ཕུ་ལུ། གྲ་ཧྣ་ཀུ་ལུ། མི་ལི་ཙུ་ལུ། ཀུ་རུ་ཀུ་ལུ། བཛྲ་བི་ཛ་ཡཱ་ཡ་སྭཱཧཱ། ཀཱི་ལི་ཀཱི་ལཱ་ཡ་སྭཱཧཱ། ཀ་ཊ། ཀ་ཊ། མ་ཊ། མ་ཊ། ར་ཊ། ར་ཊ། མོ་ཊ་ན་པྲ་མོ་ཊ་ནཱ་ཡ་སྭཱཧཱ། ཙ་ར། ནི་ཙ་ར། ཧ་ར་ཧ་ར། ས་ར་ས་ར། མཱ་ར་ཡ། མཱ་ར་ཡ། བཛྲ་བི་དཱ་ར་ཎཱ་ཡ་སྭཱཧཱ། ཙྪིནྡ་ཙྪིནྡ། བྷིནྡ་བྷིནྡ། མཧཱ་ཀཱི་ལི་ཀཱི་ལཱ་ཡ་སྭཱཧཱ། བྷནྡ། བྷནྡ། ཀྲོ་དྷ། ཀྲོ་དྷ། བཛྲ་ཀཱི་ལི་ཀཱི་ལཱ་ཡ་སྭཱཧཱ། ཙུ་རུ་ཙུ་རུ་ཙཎྜ་ཀཱི་ལི་ཀཱི་ལཱ་ཡ་སྭཱཧཱ། ཏྲཱ་ས་ཡ། ཏྲཱ་ས་ཡ། བཛྲ་ཀཱི་ལི་ཀཱི་ལཱ་ཡ་སྭཱཧཱ། ཧ་ར་ཧ་ར་བཛྲ་དྷ་ར་ཡ་སྭཱཧཱ། པྲ་ཧ་ར་པྲ་ཧ་ར། བཛྲ་པྲ་བྷཉྫ་ནཱ་ཡ་སྭཱཧཱ། མ་ཏི་སྠི་ར་བཛྲ། ཤུ་ཏི་སྠི་ར་བཛྲ། པྲ་ཏི་སྠིར་བཛྲ། མཧཱ་བཛྲ། ཨ་པྲ་ཏི་ཧ་ཏ་བཛྲ། ཨ་མོ་གྷ་བཛྲ། ཨེ་ཧྱེ་ཧི་བཛྲ། ཤཱི་གྷྲཾ་བཛྲ་དྷ་ར་ཡེ་སྭཱཧཱ། དྷ་ར་དྷ་ར། དྷི་རི་དྷི་རི། དྷུ་རུ་དྷུ་རུ། སརྦ་བཛྲ་ཀུ་ལ་མཱ་བརྟཱ་ཡེ་སྭཱཧཱ། གནོད་བྱེད་འབྱུང་པོ་མཱ་ར་ཡ་ཕཊ། ན་མཿས་མནྟ་བཛྲ་ཏཱཾ། སརྦ་བ་ལ་ཨཱ་བརྟ་ཡ་མཧཱ་བ་ལ། ཀ་ཊ་བེ་ཏ་ཏ་ལེ། ཨ་ཙ་ལེ་མཎྜལ་མཱ་ཡོ། ཨ་ཏི་བཛྲ། མཧཱ་བ་ལ་བེ་ག་ར་ཎ། ཨ་ཛི་ཏི། ཛྭ་ལ་ཛྭ་ལ། ཏི་ཊ་ཏི་ཊི། བིཾ་ག་ལེ། ད་ཧ་ད་ཧ་ཏེ་ཛོ་བ་ཏི། ཏི་ལི་ཏི་ལི། བནྡྷ་བནྡྷ། མཧཱ་བ་ལེ། བཛྲ་ཨཾ་ཀུ་ཤ་ཛྭ་ལ་ཡེ་སྭཱཧཱ། ན་མོ་རཏྣ་ཏྲ་ཡཱ་ཡ། ན་མ་ཤྩཎྜ་བཛྲ་པཱ་ཎ་ཡེ། མཧཱ་ཡཀྵ་སེ་ནཱ་པ་ཏ་ཡེ། ཏདྱ་ཐཱ། ཨོཾ་ཧ་ར་ཧ་ར་བཛྲ། མ་ཐ་མ་ཐ་བཛྲ། དྷུ་ན་དྷུ་ན་བཛྲ། ད་ཧ་ད་ཧ་བཛྲ། པ་ཙ་པ་ཙ་བཛྲ། དྷ་ར་དྷ་ར་བཛྲ། དྷཱ་ར་ཡ་དྷཱ་ར་ཡ་བཛྲ། དཱ་ར་ཎ་དཱ་ར་ཎ་བཛྲ། ཙྪིནྡ་ཙིནྡ་བཛྲ། བྷིནྡ་བྷིནྡ་བཛྲ་ཧཱུཾ་ཕཊ། ན་མ་ཤྩཎྜ་བཛྲ་ཀྲོ་དྷཱ་ཡ། ཧུ་ལུ་ཧུ་ལུ། ཏིཥྛ་ཏིཥྛ། བྷནྡྷ་བནྡྷ། ཧ་ན་ཧ་ན། ཨ་མྲ་ཏ་ཧཱུཾ་ཕཊ། ཅེས་གང་མང་བཟླའོ། ། (མཇུག་ཏུ། )དགེ་བ་འདི་ཡིས་མྱུར་དུ་བདག། རྡོ་རྗེ་རྣམ་འཇོམས་འགྲུབ་གྱུར་ནས། ། འགྲོ་བ་གཅིག་ཀྱང་མ་ལུས་པ། ། དེ་ཡི་ས་ལ་འགོད་པར་ཤོག། (དེ་ནས་བུམ་པ་སྒྲུབ་པ་ནི། )ཨ་མྲ་ཏས་བསངས། སྭ་བྷ་ཝས་སྟོང་པར་སྦྱངས། སྟོང་པའི་ངང་ལས་བཾ་ལས་བྱུང་བའི་རིན་པོ་ཆེའི་བུམ་པ་དཀར་པོ་ལྟོ་བ་ཆེ་བ། མགྲིན་པ་རིང་བ། མཆུ་འཕྱང་བ་ལ་སོགས་པའི་མཚན་ཉིད་ཐམས་ཅད་དང་ལྡན་པའི་ནང་དུ་སྣ་ཚོགས་པདྨ་དང་ཉི་མའི་སྟེང་དུ་ཧཱུཾ་ལས་སྣ་ཚོགས་རྡོ་རྗེ་ཧཱུཾ་གིས་མཚན་པ་དེ་ལས་འོད་འཕྲོས། དོན་གཉིས་བྱས། འདུས་ཡོངས་སུ་གྱུར་པ་ལས། རྡོ་རྗེ་རྣམ་པར་འཇོམས་པ་སྐུ་མདོག་ལྗང་སྔོན་ཞལ་གཅིག་ཕྱག་གཉིས་སྤྱན་གསུམ་པ་ཕྱག་གཡས་སྣ་ཚོགས་རྡོ་རྗེ་ཐུགས་ཀར་འཛིན་ཅིང༌། ཕྱག་གཡོན་སྣ་ཚོགས་རྡོ་རྗེས་ཡུ་བ་བྱས་པའི་དྲིལ་བུ་དཀུར་བརྟེན་པ། དབུ་སྐྲ་རལ་བའི་ཐོར་ཚུགས་ཅན། རིན་པོ་ཆེའི་རྒྱན་གྱིས་བརྒྱན་ཅིང༌། དར་གྱི་ན་བཟའ་གསོལ་བ། འོད་ཀྱི་དཀྱིལ་འཁོར་གྱི་དབུས་ན་རྡོ་རྗེའི་སྐྱིལ་ཀྲུང་གིས་བཞུགས་པའི་སྤྱི་བོར་ཨོཾ། མགྲིན་པར་ཨཱཿཐུགས་ཀར་ཧཱུཾ། དེ་ལས་འོད་འཕྲོས། རང་བཞིན་གྱི་གནས་ནས་བསྒོམས་པ་དང་འདྲ་བའི་ཡེ་ཤེས་པ་དང༌། དབང་གི་ལྷ་རྣམས་སྤྱན་དྲངས། ཛཿཧཱུཾ་བཾ་ཧོཿདབང་གི་ལྷས་དབང་བསྐུར། མི་བསྐྱོད་པས་དབུ་བརྒྱན་པར་གྱུར། ཨོཾ་སརྦ་ཏ་ཐཱ་ག་ཏ་ཨརྒཾ་ྷནས། ཤཔྟའི་བར་གྱིས་མཆོད།

རྒྱལ་བ་ཀུན་གྱི་མཁྱེན་བརྩེའི་ལྷ་ལམ་ལས། །

མཐུ་ཆེན་སྟོབས་ཀྱི་དབྱར་སྐྱེས་ཕྲེང་བ་ཅན། །

འཕྲིན་ལས་རྣམ་བཞིའི་གྲུ་ཆར་འབེབས་མཁས་པ། །

དཔལ་ལྡན་ཁྲོ་བོའི་དབང་པོ་ལ་ཕྱག་འཚལ། ། ཞེས་པས་བསྟོད། མདུན་བསྐྱེད་རྡོ་རྗེ་རྣམ་འཇོམས་སུ་གསལ་བའི་ཐུགས་ཀར་ཉི་གདན་གྱི་དབུས་སུ་ཧཱུཾ་ཡིག་གི་མཐར་སྔགས་འཕྲེང་གིས་བསྐོར་བ་ལས་འོད་ཟེར་འཕྲོས། རང་གཞན་གྱི་ནད་གདོན་སྡིག་སྒྲིབ་ཐམས་ཅད་སྦྱངས། ཁྱད་པར་དུ་ཡང་ཕྱོགས་བཅུའི་སངས་རྒྱས་བྱང་སེམས་རྣམས་ཀྱི་སྐུ་གསུང་ཐུགས་ཀྱི་བྱིན་རླབས་ཐམས་ཅད་བདུད་རྩི་འོད་ཟེར་གྱི་རྣམ་པར་སྤྱན་དྲངས། སྔགས་འཕྲེང་ལ་ཐིམ། དེ་ལས་བདུད་རྩིའི་རྒྱུན་བབས་བུམ་པ་གང་བར་བསམས་ལ། ན་མཤྩཎྜ་བཛྲ་ཀྲོ་དྷཱ་ཡ། ཧུ་ལུ་ཧུ་ལུ། ཏིཥྛ་ཏིཥྛ། བནྡྷ་བནྡྷ། ཧ་ན་ཧ་ན། ཨ་མྲི་ཏེ་ཧཱུཾ་ཕཊ། (སྔགས་མང་དུ་བཟླས་མཐར། ལྷ་འོད་དུ་ཞུ་ནས་བུམ་པའི་ཆུའི་རྣམ་པར་གྱུར་པར་མོས་བུམ་པའི་ཆུས་བླ་མ་ཡི་དམ་སངས་རྒྱས་བྱང་སེམས་དཔའ་བོ་མཁའ་འགྲོ་སྐྱེས་བུ་གསུམ་གྱི་ཆོས་སྐྱོང་རྣམས་ལ་ཁྲུས་གསོལ་རྒྱས་བསྡུས་སྐབས་དང་སྦྱར་ནས་བྱ། དེ་ནས་ཁྲུས་ཆུ་ཡང་བུམ་པར་བླུག། ཡུལ་ལྗོངས་གནས་དགོན་རི་ཀླུང་རྣམས་ལ་མེ་ལོང་བསྟན་ནས་དེ་ལ་གཟུགས་བརྙན་ཤར་བ་ལ་ཁྲུས་གསོལ་ཞིང༌། དམིགས་པ་ནི་རང་རྣམ་འཇོམས་སུ་གསལ་བའི་ཐུགས་ཀ་ནས་ཕྱོགས་བཅུའི་རྒྱལ་བ་སྲས་བཅས་ཐམས་ཅད་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པ་སྐུ་ཆེ་ཆུང་དཔག་ཏུ་མེད་པའི་རྣམ་པར་སྤྱན་དྲངས་པ་ཡུལ་ཕྱོགས་ཀྱི་སྟེང་ཐམས་ཅད་ཁྱབ་པར་བཞུགས་པའི་སྐུ་ལས་བདུད་རྩིའི་རྒྱུན་བབས་པས་སྣོད་བཅུད་ཀྱི་མ་དག་པའི་དྲི་མ་ཐམས་ཅད་དག་པའི་མོས་པ་དང་བཅས་ཏེ། )ཇི་ལྟར་བལྟམས་པ་ཙམ་གྱིས་ནི། ། (ཞེས་པ་ནས།) དེ་བཞིན་བདག་གིས་ཁྲུས་བགྱིའོ། ། ན་མཤྩཎྜ་བཛྲ་ཀྲོ་དྷཱ་ཡ། ཧུ་ལུ་ཧུ་ལུ། ཏིཥྛ་ཏིཥྛ། བནྡྷ་བནྡྷ། ཧ་ན་ཧ་ན། ཨ་མྲ་ཏེ་ཧཱུཾ་ཕཊ། ཡུལ་ཕྱོགས་འདིར་གནས་ཀྱི་བསྲུང་མ་ལྷ་ཀླུ་གཞི་བདག་བླ་དཔོན་འབངས་འཁོར་སྡེ་བཅས་གནས་དགོན་ཡུལ་ལྗོངས་རྣམས་ལ་དམ་ཚིག་ཉམས་ཆག་འཁོན་གྲིབ་བཙོག་གྲིབ་ཕོག་པ་སོགས་སངས་ཀྱིས་དག་ཅིང༌། བསམ་ངན་སྦྱོར་རྩུབ་སོགས་ཞི་ནས་དགའ་བདེ་རྒྱ་ཆེན་པོས་འཚོ་བར་གྱུར་ཅིག། ཚེ་རིང་མཆེད་ལྔ་བསྟན་སྐྱོངས་བཅས། ། (ཞེས་པ་ནས།) ཡུལ་ལྷ་གཞི་བདག་རྣམས་ལ་ཁྲུས་བགྱིའོ། ། ན་མཤྩཎྜ་བཛྲ་ཀྲོ་དྷཱ་ཡ། ཧུ་ལུ་ཧུ་ལུ། ཏིཥྛ་ཏིཥྛ། བནྡྷ་བནྡྷ། ཧ་ན་ཧ་ན། ཨ་མྲ་ཏེ་ཧཱུཾ་ཕཊ། (མཐར་སྔགས་དང་སྤེལ་ཚིག་གོང་བཞིན་སྦྱར། )

རི་བྲག་ཆུ་ཀླུང་ནགས་ཚལ་སྤང་གི་ལྗོངས། །

ཁང་ཞིང་ཟས་ནོར་ཡོ་བྱད་བསྲུང་བ་དང༌། །

)ཡུལ་ཕྱོགས་འདིར་གནས་ཀྱི་བསྲུང་མ་ལྷ་ཀླུ་གཞི་བདག་བླ་དཔོན་འབངས་འཁོར་སྡེ་བཅས་གནས་དགོན་ཡུལ་ལྗོངས་ཐམས་ཅད་ཀྱི་དམ་ཚིག་ཉམས་ཆག་འཁོན་གྲིབ་བཙོག་གྲིབ་སོགས་དང༌། དེས་བསྐྱེད་པའི་ནད་རིམས་གདོན་བགེགས་བཀྲ་མི་ཤིས་པ་སོགས་སྡིག་པ་དང་སྒྲིབ་པ་བར་གཅོད་མི་མཐུན་པའི་ཕྱོགས་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། ནད་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། གདོན་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། རིམས་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། འཁོན་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། གྲིབ་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། བྱད་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། བྱུར་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། སྐལ་བ་ངན་པ་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། ལྟས་ངན་པ་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། བཀྲ་མི་ཤིས་པ་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། ཉེ་བར་འཚེ་བ་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། དུས་མ་ཡིན་པར་འཆི་བ་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། གཟའ་དང་རྒྱུ་སྐར་གྱི་གནོད་པ་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། རྒྱལ་པོ་དང་ཆོམ་རྐུན་གྱི་འཇིགས་པ་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། བདུད་ཀྱི་ཞགས་པ་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། གཤིན་རྗེའི་ཕོ་ཉ་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། བགེགས་དང་ལོག་འདྲེན་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། བར་གཅོད་མི་མཐུན་པའི་ཕྱོགས་ཐམས་ཅད་ཨ་པ་ན་ཡེ་སྭཱཧཱ། (ཞེས་ལན་གསུམ་མམ་བདུན་ལ་སོགས་པ་བརྗོད། དེ་ནས།) ན་མོ་རྗེ་བཙུན་རྩ་བ་དང་བརྒྱུད་པར་བཅས་པའི་དཔལ་ལྡན་བླ་མ་དམ་པ་རྣམས་ཀྱི་བཀའ་བདེན་པ་དང༌། སངས་རྒྱས་ཀྱི་བཀའ་བདེན་པ་དང༌། ཆོས་ཀྱི་བཀའ་བདེན་པ་དང༌། དགེ་འདུན་གྱི་བཀའ་བདེན་པ་དང༌། བཅོམ་ལྡན་འདས་རྡོ་རྗེ་རྣམ་པར་འཇོམས་པའི་ལྷ་ཚོགས་རྣམས་ཀྱི་སྐུ་གསུང་ཐུགས་གསང་བ་བླ་ན་མེད་པའི་བདེན་པ་དང༌། བདེན་པ་ཆེན་པོའི་བྱིན་གྱིས་བརླབས་པ་ལ་བརྟེན་ནས་ཡུལ་ཕྱོགས་འདིར་གནས་ཀྱི་ནད་རིམས་གདོན་བགེགས་གནོད་ཅིང་འཚེ་བ་ངན་པ་ཐམས་ཅད་དྲི་ཆུ་དང་དྲི་གླུད་འདིས་ཚིམས་པར་གྱིས་ལ། རང་རང་སོ་སོའི་གནས་སུ་སོང་ཞིག། བདེན་པ་རྒྱལ་གྱུར་བརྫུན་མ་ལགས། ། བདེན་པ་དང་བདེན་པའི་ཚིག་འདི་དག་གིས་ཡུལ་ཕྱོགས་འདི་རྣམས་བདུད་དང་མི་མཐུན་པའི་ཕྱོགས་ཐམས་ཅད་ལས་རྒྱལ་བར་གྱུར་ཅིག། ཛ་ཡ་ཛ་ཡ་སིདྡྷི་སིདྡྷི་སྭཱཧཱ། (དེ་རྗེས་བསྲུང་བའི་སྐབས་ཀྱི་དམིགས་པ་ནི།) སྟེང་གི་ལྷ་ཚོགས་ཐམས་ཅད་སྣོད་བཅུད་ཡུལ་ཁམས་ཐམས་ཅད་ལ་ཐིམ་པས་སྣོད་ཐམས་ཅད་དག་པ་ལྷའི་གཞལ་ཡས་དང༌། བཅུད་ཐམས་ཅད་རྡོ་རྗེ་རྣམ་འཇོམས་སུ་གྱུར་པའི་སྐུ་ལས་འོད་ཟེར་འཕྲོས་པས་ཡུལ་ཁམས་རྣམས་ཀྱི་ཕྱི་རོལ་ཏུ་རྡོ་རྗེའི་ས་གཞིའི་གུར་བླ་རེ་མེ་རིས་ཡོངས་སུ་བསྐོར་བའི་བསྲུང་བ་ཆེན་པོར་གྱུར་པའི་མོས་པ་དང་བཅས། རང་ལྷར་གསལ་བའི་ཐུགས་ཀའི་ཧཱུཾ་ལས་འོད་ཟེར་འཕྲོས། ཡུལ་ཕྱོགས་འདི་དག་གི་སྟེང་འོག་ཕྱོགས་མཚམས་ཐམས་ཅད་རྡོ་རྗེ་རྡུལ་ཕྲ་རབ་ཀྱི་རང་བཞིན་དུ་གྱུར་ཏེ། ནད་གདོན་སྡིག་སྒྲིབ་ཐམས་ཅད་ཀྱི་གནོད་པས་མི་མཚུགས་པའི་བསྲུང་བ་ཆེན་པོར་བྱིན་གྱིས་བརླབས་པར་གྱུར། (སྔགས་ལན་གསུམ་མམ་བདུན་བརྗོད་དེ།) ཡུལ་ཕྱོགས་འདིར་གནས་ཀྱི་བསྲུང་མ་ལྷ་ཀླུ་གཞི་བདག་བླ་དཔོན་འབངས་འཁོར་སྡེ་བཅས་གནས་དགོན་ཡུལ་ལྗོངས་ཐམས་ཅད་ཀྱི་དམ་ཚིག་ཉམས་ཆག་འཁོན་གྲིབ་བཙོག་གྲིབ་སོགས་དང༌། དེས་བསྐྱེད་པའི་ནད་རིམས་གདོན་བགེགས་བཀྲ་མི་ཤིས་པ་སོགས་སྡིག་པ་དང་སྒྲིབ་པ་བར་གཅོད་མི་མཐུན་པའི་ཕྱོགས་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། ནད་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། གདོན་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། རིམས་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། འཁོན་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། གྲིབ་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། བྱད་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། བྱུར་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། སྐལ་བ་ངན་པ་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། ལྟས་ངན་པ་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། བཀྲ་མི་ཤིས་པ་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། ཉེ་བར་འཚེ་བ་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། དུས་མ་ཡིན་པར་འཆི་བ་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། གཟའ་དང་རྒྱུ་སྐར་གྱི་གནོད་པ་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། རྒྱལ་པོ་དང་ཆོམ་རྐུན་གྱི་འཇིགས་པ་ཐམས་ཅད་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། བདུད་ཀྱི་ཞགས་པ་ཐམས་ཅད་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། གཤིན་རྗེའི་ཕོ་ཉ་ཐམས་ཅད་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། བགེགས་དང་ལོག་འདྲེན་ཐམས་ཅད་ཐམས་ཅད་ལས་རཀྵ་རཀྵ། བར་གཅོད་མི་མཐུན་པའི་ཕྱོགས་ཐམས་ཅད་རཀྵ་རཀྵ། (ཞེས་ཚར་གསུམ་མམ་བདུན་བརྗོད།)
