རྗེ་བཙུན་རྩ་བའི་བླ་མ་ལ་བཏུད་ནས། །

ལམ་ཟབ་ནང་གི་བླ་མའི་རྣལ་འབྱོར་བཤད། །འདིར་རྗེ་བཙུན་མཐུ་སྟོབས་ཀྱི་དབང་ཕྱུག་དཔལ་བིརྺ་པས། རྣལ་འབྱོར་གྱི་དབང་ཕྱུག་དཔལ་ས་སྐྱ་པ་ལ་དངོས་སུ་གནང་བའི་གདམས་ངག་ཟབ་མོ་ནང་གི་བླ་མའི་རྣལ་འབྱོར་ཉམས་སུ་ལེན་པར་འདོད་ན། སྟན་བདེ་བ་ལ་བསམ་གཏན་གྱི་ཡན་ལག་གིས་འདུག་ལ། སྐྱབས་འགྲོ་སེམས་བསྐྱེད་སྔོན་དུ་འགྲོ་བས། རང་སྐད་ཅིག་གིས་ཀྱེ་རྡོ་རྗེར་གསལ་བའི་ཐུགས་ཀ་ནས་འོད་འཕྲོས་པས། དཔལ་ཧེ་རུ་ཀ་དང་བླ་མ་ཐ་མི་དད་པ་མདུན་དུ་སྤྱན་དྲངས་ལ། ཕྱི་ནང་གསང་བ་དེ་ཁོ་ན་ཉིད་ཀྱིས་མཆོད། གུས་བཏུད་དྲག་པོ་བྱས་ཏེ་བདག་ཉིད་ལ་ཐིམ་པས། རང་གི་ལུས་སྐད་ཅིག་གི་དཀྱིལ་འཁོར་ལུས་དཀྱིལ་ལས་འབྱུང་བ་ལྟར། སྤྱི་བོར་སྣ་ཚོགས་རྡོ་རྗེ་བསྒོམ་པ་ནས། རྐང་མཐིལ་གནོད་མཛེས་སུ་བསམ་པའི་བར་ཤིན་ཏུ་གསལ་བར་བསྒོམས་ནས། རང་གི་སྤྱི་བོར་རྩ་བའི་བླ་མ་ནས་རྡོ་རྗེ་འཆང་གི་བར་གྱི་བརྒྱུད་པ་རྣམས་ཀྱི་རྡོ་རྗེ་ཡབ་ཡུམ་གྱི་རྣམ་པ་ཅན། འོག་མའི་སྤྱི་བོར་གོང་མ་བཞུགས་པ་ཐོ་བརྩེགས་ཀྱི་ཚུལ་དུ་བསམས་ལ། དེ་རྣམས་ལ་མོས་གུས་དྲག་ཏུ་བྱ་ཞིང༌། རྩ་བ་དང་བརྒྱུད་པར་བཅས་པའི་བླ་མ་དམ་པ་རྣམས་ཀྱིས་བདག་གི་རྒྱུད་བྱིན་གྱིས་བརླབ་ཏུ་གསོལ། ཞེས་གསོལ་བ་བཏབ་པས་བླ་མ་རྣམས་ཀྱི་ཐུགས་ཀ་ནས་འོད་འཕྲོས། བླ་མ་ཡི་དམ་སངས་རྒྱས་བྱང་སེམས་ཐམས་ཅད་ཀྱི་བྱིན་རླབས་མ་ལུས་པ་བསྡུས་ནས་བླ་མ་རྣམས་ལ་ཐིམ་པར་བསྒོམ།

དེ་ནས་རྡོ་རྗེ་འཆང་བདག་མེད་མ་ལ་ཐིམ། དེ་བིརྺ་པ་ལ། དེ་རྗེ་བཙུན་ས་སྐྱ་པ་ཆེན་པོ་ལ་ཐིམ། དེ་ནས་རང་གི་རྩ་བའི་བླ་མ་ལ་ཐིམ་པའི་བར་བྱས་ནས། རྩ་བའི་བླ་མ་ལ། བླ་མ་དམ་པས་བདག་ལ་བུམ་པའི་དབང་བསྐུར་དུ་གསོལ། ཞེས་གསོལ་བ་བཏབ་པས།བླ་མ་ཡབ་ཡུམ་སྙོམས་པར་ཞུགས་པ་ལས་བྱུང་བའི་བྱང་ཆུབ་ཀྱི་སེམས་རང་གི་སྤྱི་བོ་ནས་ཞུགས། བདེ་ཆེན་གྱི་འཁོར་ལོའི་དབུས་ན་རྣམ་པར་སྣང་མཛད་ཡབ་ཡུམ་དང༌། རྩ་འདབ་རྣམས་ལ་ལྷ་མོ་སོ་གཉིས་ཡོད་པ་ལ་ཐིམ་པར་བསམས་ལ། ལྷ་རྣམས་གསལ་གདབ། རླུང་བུམ་པ་ཅན་བཟུང་ལ། བྱང་སེམས་དེས་སྨིན་མཚམས་ཡན་ཆད་གང༌། ལུས་ཀྱི་དྲི་མ་དག བུམ་པའི་དབང་ཐོབ་པར་བསམ་མོ། །ཡང་སྤྱི་བོར་རྩ་བའི་བླ་མ་ལ། གསང་བའི་དབང་བསྐུར་དུ་གསོལ། ཞེས་གསོལ་བ་བཏབ་པས། སྦྱོར་མཚམས་ནས་བྱང་སེམས་ཀྱི་ཆུ་རྒྱུན་བྱུང༌། མགྲིན་པའི་འཁོར་ལོའི་ལྷ་རྣམས་ལ་ཐིམ་པར་བསམས་ལ། ལྷ་རྣམས་གསལ་གདབ། བུམ་པ་ཅན་བཟུང་ལ། བྱང་སེམས་ཀྱིས་མགྲིན་པ་ཡན་ཆད་གང༌། ངག་གི་དྲི་མ་དག གསང་བའི་དབང་ཐོབ་པར་བསམ་མོ། །དེ་བཞིན་དུ་ཤེས་རབ་ཡེ་ཤེས་ཀྱི་དབང་བསྐུར་དུ་གསོལ། ཞེས་གསོལ་བ་བཏབ་པས། ཡང་སྦྱོར་མཚམས་ནས་བྱང་སེམས་ཀྱི་ཆུ་རྒྱུན་བྱུང༌། སྙིང་གའི་འཁོར་ལོའི་ལྷ་རྣམས་ལ་ཐིམ་པར་བསམས་ལ། ལྷ་རྣམས་གསལ་གདབ། བུམ་པ་ཅན་བཟུང་ལ། བྱང་སེམས་ཀྱིས་སྙིང་ག་ཡན་ཆད་གང༌། ཡིད་ཀྱི་དྲི་མ་དག ཤེས་རབ་ཡེ་ཤེས་ཀྱི་དབང་ཐོབ་པར་བསམ་མོ། །དེ་བཞིན་དུ་དབང་བཞི་པ་བསྐུར་དུ་གསོལ། ཞེས་གསོལ་བ་བཏབ་པས། བླ་མ་ཡབ་ཡུམ་བདེ་བ་ཆེན་པོའི་རང་བཞིན་དུ་ཞུ་ནས། ལྟེ་བ་དང༌། གསང་གནས་དང༌། དབང་པོ་དང༌། ཡན་ལག་གི་ལྷ་རྣམས་ལ་ཐིམ་པར་བསམས་ལ། ལྷ་རྣམས་གསལ་གདབ། བུམ་པ་ཅན་བྱས་ལ། བྱང་སེམས་དེས་སྤྱི་བོ་ནས་རྐང་མཐིལ་ཡན་ཆད་གང༌། སྒོ་གསུམ་གྱི་དྲི་མ་དག དབང་བཞི་པ་ཐོབ་པར་བསམ་མོ། །

དེ་ནས་ལྟ་བ་ཅི་ནུས་སུ་བསྐྱངས་ལ་དགེ་བ་བསྔོ་བ་སོགས་བྱའོ། །དེ་ལྟར་ན་དབང་བཞི་པོ་རེ་རེ་ལའང་ལམ་བཞི་ཆར་ཚང་བ་ཡིན་ཏེ། ལྷའི་རྣམ་པ་ལ་སེམས་འཛིན་པ་བུམ་དབང་གི་ལམ། རླུང་བུམ་པ་ཅན་དུ་བྱ་བ་གསང་དབང་གི་ལམ། རང་རང་གི་འཁོར་ལོ་རྣམས་བྱང་སེམས་ཀྱིས་ཁྱབ་པར་བྱས་པ་གསུམ་པའི་ལམ། དེ་དག་ལས་ལྷན་ཅིག་སྐྱེས་པའི་ཡེ་ཤེས་སྐྱེས་པ་ནི་བཞི་པའི་ལམ་ཡིན་པའི་ཕྱིར་རོ། །ཟབ་མོ་ནང་གི་བླ་མ་སྒྲུབ་པའི་ཐབས། །སྙན་བརྒྱུད་ཡི་གེར་བྲིས་ལ་ཉེས་པ་གང༌། །བླ་མ་མཁའ་འགྲོའི་སྤྱན་སྔར་བདག་འཆགས་ཤིང༌། །དགེ་བས་དཔལ་ལྡན་བླ་མ་མཉེས་བྱེད་ཤོག ནང་གི་བླ་མའི་རྣལ་འབྱོར་གྱི་གདམས་ངག་ལམ་འབྲས་ཀྱི་བཅུད་བསྡུས་པ་ལྟ་བུ་འདི་ནི། བླ་མ་དམ་པ་རྣམས་ཀྱི་བྲན་དུ་གྱུར་པ་ཤཱཀྱའི་དགེ་སློང་ཀུན་དགའ་བཟང་པོས་སྦྱར་བའོ།། །།
