ལམ་རིམ་བླ་བརྒྱུད་སྨྲ་བ་ཟླ་མེད་མ། །

༄༅། ། སྨྲ་བ་ཟླ་མེད་འདྲེན་མཆོག་ཤཱཀྱའི་ཏོག ། རྒྱལ་བའི་མཁྱེན་ རབ་ཀུན་བསྡུས་འཇམ་པའི་དབྱངས། ། ཟབ་མོའི་དོན་གཟིགས་འཕགས་ མཆོག་ཀླུ་སྒྲུབ་ཞབས། ། སྨྲ་བའི་གཙུག་རྒྱན་གསུམ་ལ་གསོལ་བ་འདེབས། ། འཕགས་པའི་དགོངས་པ་གསལ་མཛད་ཟླ་བ་གྲགས༑ ༑དེ་སྲས་ཐུ་བོ་རིགས་ པའི་ཁུ་བྱུག་ཆེ། ། རྒྱལ་སྲས་རིགས་ པའི་ཁུ་བྱུག་གཉིས་པའི་ཞབས། ། རིགས་པའི་དབང་ཕྱུག་གསུམ་ལ་གསོལ་བ་འདེབས། ། རྟེན་འབྲེལ་ཟབ་མོ་ཇི་ བཞིན་གཟིགས་པ་ཡི། ། ཤིང་རྟ་ཆེན་པོའི་སྲོལ་འཛིན་མར་མེ་མཛད། ། ལམ་བཟང་གསལ་མཛད་སྟོན་པ་རིན་པོ་ཆེ། ། འཛམ་གླིང་རྒྱན་གྱུར་གཉིས་ ལ་གསོལ་བ་འདེབས༑ རྒྱལ་བའི་གདུང་འཚོབ་བཤེས་གཉེན་པོ་ཏོ་བ། ། རྣམ་དཔྱོད་འགྲན་ཟླ་བྲལ་བའི་ཤ་ར་བ། ། བྱང་ཆུབ་ཐུགས་ཀྱི་བཀའ་བབས་ འཆད་ཁ་བ། ། འགྲོ་བའི་རེ་སྐོང་གསུམ་ལ་གསོལ་བ་འདེབས་༑ ༑ལུང་ རྟོགས་མངའ་བདག་བྱང་སེམས་སྤྱིལ་བུ་པ། ། དྲི་མེད་ལུང་གི་དབང་ཕྱུག་ མཁས་པའི་མཆོག ། ཁམས་གསུམ་འགྲོ་བའི་མགོན་པོ་རིན་པོ་ཆེ། ། གནས་ བརྟན་ཆེན་པོ་གསུམ་ལ་གསོལ་བ་འདེབས༑ ༑རྣམ་དག་ཚུལ་ཁྲིམས་ངད་ ལྡང་ཟངས་ཆེན་པ། ། འདུལ་བ་འབུམ་སྡེའི་མངའ་བདག་མཚོ་སྣ་པ། །

ས་མངོན་རྒྱ་མཚོའི་མཐར་སོན་མོན་གྲ་པ། ། འགྲོ་བའི་འདྲེན་པ་གསུམ་ལ་ སོལ་བ་འདེབས་༑ ༑ ཟབ་ཅིང་རྒྱ་ཆེའི་ཆོས་ལ་མངའ་བརྙེས་པ། ། སྐལ་ ན་འགྲོ་བ་ཀུན་གྱི་སྐྱབས་སུ་གྱུར། ། འཕྲིན་ལས་བཟང་པོས་བསྟན་པ་རྒྱས་ ཛད་པ། ། དཔལ་ལྡན་བླ་མའི་ཞབས་ལ་གསོལ་བ་འདེབས། ། གྲུབ་པའི་ བང་ཕྱུག་ཆེན་པོ་ཚུལ་ཁྲིམས་འབར། ། བཤེས་གཉེན་ཚུལ་བཞིན་བསྟེན་ ཛད་གཞོན་ནུ་འོད། ། ཐེག་མཆོག་ལམ་གྱིས་རྒྱུད་སྦྱངས་དགྱེར་སྒོམ་ བས། ། རྒྱལ་བའི་སྲས་པོ་གསུམ་ལ་གསོལ་བ་འདེབས༑ ༑རྨད་བྱུང་ཡོན་ ན་མཛོད་འཛིན་སངས་རྒྱས་དབོན། ། དམ་པས་བྱིན་བརླབས་ནམ་མཁའ་ པ་པོ་དང་། ། འཇིག་རྟེན་ཆོས་བརྒྱད་སྤངས་པའི་སེང་གེ་བཟང་། ། རྒྱལ་ པ་བཟང་པོའི་ཞབས་ལ་གསོལ་བ་འདེབས། ། བྱང་ཆུབ་ཐུགས་ཀྱིས་འགྲོ་ ད་བུ་བཞིན་གཟིགས༑ ༑ལྷག་པའི་ལྷ་ཡིས་རྗེས་བཟུང་བྱིན་གྱིས་ བས། ། སྙིགས་དུས་འགྲོ་བ་འདྲེན་པའི་བཤེས་གཉེན་མཆོག ། ནམ་ པའ་རྒྱལ་མཚན་ཞབས་ལ་གསོལ་བ་འདེབས། ། བློ་གྲོས་ཁང་བཟང་བསླབ་

མ་ནོར་གྱིས་ཕྱུག ། བཟང་པོའི་འཕྲིན་ལས་གདུལ་བྱའི་དཔལ་དུ་ ༑ ༑གྲགས་པའི་འོད་ཟེར་འཛམ་གླིང་ཕྱོགས་བཅུར་འཕྲོ། ། བློ་བཟང་ །ས་པའི་ཞབས་ལ་གསོལ་བ་འདེབས། ། རི་བོ་དགེ་ལྡན་གྱི་བླ་བརྒྱུད་

། ཐར་འདོད་ཐར་པར་འཇུག་པའི་སྒོ་གཅིག་པུ། ། དངོས་སྟོབས་རིག་ ་རིགས་པའི་གཞུང་གི་དོན། ། ལེགས་པར་རྟོགས་ནས་གཞན་ལ་ལེགས་ མས་པའི། ། རྒྱལ་ཚབ་ཆོས་རྗེའི་ཞབས་ལ་གསོལ་བ་འདེབས། ། འདུལ་

བ་འཛིན་པ་ཐུབ་པའི་རྒྱལ་ཚབ་ཅེས། ། སྙན་པའི་གྲགས་པས་ས་སྟེང་ཀུན་ ཁྱབ་ཅིང་། ཕྱོགས་དུས་ཀུན་ཏུ་བསྟན་པའི་རྒྱལ་མཚན་འཛིན། ། གྲགས་པ་ རྒྱལ་མཚན་ཞབས་ལ་གསོལ་བ་འདེབས། ། གཞུང་ལུགས་རལ་པ་འཐུག་པོས་ གཉའ་ཁེབས་ཤིང་། ། རིགས་པའི་གད་རྒྱངས་ས་གསུམ་འདར་བྱེད་པ༑ ༑ འཇིགས་མེད་སྨྲ་བའི་སེང་གེ་དགེ་ལེགས་དཔལ། ། མཁས་གྲུབ་ཆོས་རྗེའི་ ཞབས་ལ་གསོལ་བ་འདེབས། ། དགེ་ལེགས་གཏིང་མཐའ་ཡངས་པའི་ཆུ་ གཏེར་ལས། ། མཁྱེན་བརྩེའི་འོད་འབར་དབང་གི་རྒྱལ་པོ་ལྟར། ། མཁས་ མང་རྒྱལ་མཚན་གཙུག་ན་མངོན་མཐོ་བ། ། མི་ཕམ་བླ་མའི་ཞབས་ལ་གསོལ་ བ་འདེབས། ། དྲི་མེད་བློ་གྲོས་རིགས་པའི་འབྲུག་སྒྲ་ཅན། ། རྒྱ་མཚོའི་གོས་ ཅན་མཐའ་དག་འདར་བྱེད་པ། ། ཆོས་ཆར་ཟབ་མོས་ཐུབ་བསྟན་དཔྱིད་ཀྱི་ དཔལ། ། སྐྱོང་མཛད་བླ་མ་མཆོག་ལ་གསོལ་བ་འདེབས། ། ཐུབ་བསྟན་ གསེར་གྱི་དྲ་བས་གདོང་མཛེས་ཤིང་། ། བསླབ་གསུམ་དག་པའི་ལུས་སྟོབས་ རབ་ཏུ་རྒྱས། ། བདག་ཏུ་འཛིན་པའི་འཁྲི་ཤིང་ཀུན་གཅོད་པ། ། བསྡུ་ བཞིས་འགྲོ་དོན་མཛད་ལ་གསོལ་བ་འདེབས། ། མདོ་རྒྱུད་ཀུན་ལ་བློ་ གྲོས་ ཐོགས་མེད་རྒྱས། ། རི་དབང་ལྟ་བུར་བརྟན་པའི་ཐུགས་རྗེ་ཅན། ། ལུས་ ཅན་ཐེག་པ་མཆོག་ལ་འགོད་མཛད་པའི། ། དཔལ་ལྡན་བླ་མའི་ཞབས་ལ་ གསོལ་བ་འདེབས། ། སྨོན་ལམ་བརྒྱ་ཕྲག་དུ་མས་བསྐྲུན་པའི་སྐུ། ། དཔལ་ ལྡན་མཁས་པའི་ཡིད་རབ་འཕྲོག་པའི་གསུང་། ། ལེགས་པའི་བློ་གྲོས་ཆོས་ ཀུན་གསལ་བའི་ཐུགས། ། རབ་འབྱམས་ཆོས་རྗེའི་ཞབས་ལ་གསོལ་བ་
