ས་ལུགས་ཆོས་སྐྱོང་གཏོར་བསྔོ་བསྡུས་པ་བཞུགས་སོ། །

མཆོད་གཏོར་རྣམས། ཨོཾ་བཛྲ་ཨ་མྲྀ་ཏ་ཀུཎྜ་ལཱི་ཧ་ན་ཧ་ན་ཧཱུྃ་ཧཱུྃ་ཧཱུྃ་ཕཊ་སྭཱ་ཧཱ། ཨོཾ་སྭ་བྷཱ་ཝ་ཤུདྡྷཿ (སོགས་ཀྱིས་སྦྱང་། )

སྟོང་པའི་ངང་ལས་ཨ་ལས་ཐོད་པའི་སྣོད། །

ཡངས་ཤིང་རྒྱ་ཆེ་སྟོང་གསུམ་མཉམ་པའི་ནང་།

ཤ་ལྔ་བདུད་རྩི་ཡེ་ཤེས་ལྔའི་ངོ་བོར། །

འོད་ཞུ་བདུད་རྩིའི་རྒྱ་མཚོ་ཆེན་པོར་གྱུར། ཨོཾ་ཨཱཿཧཱུྃ་ཧ་ཧོ་ཧྲཱིཿ (ལན་གསུམ་གྱི་བྱིན་གྱིས་བརླབ།) ཧཱུྃ། །

འོག་མིན་བདེ་ཆེན་དག་པའི་ཕོ་བྲང་ནས། །

རྣམ་སྣང་ཐུགས་ལས་སྤྲུལ་པའི་མཐུ་བོ་ཆེ། །

བསྟན་སྲུང་ཀུན་གྱི་གཙོ་བོ་རྡོ་རྗེ་གུར། །

དཔལ་ལྡན་མགོན་པོ་འདིར་བྱོན་མཆོད་གཏོར་བཞེས། །

ཡོངས་འདུའི་ཚལ་དང་གཤིན་རྗེའི་ཕོ་བྲང་དང་། །

འཛམ་གླིང་དེ་ཝཱི་ཀོ་ཊའི་གནས་མཆོག་ནས། །

འདོད་ཁམས་གཙོ་མོ་ནམ་གྲུ་རེ་མ་ཏི། །

དཔལ་ལྡན་ལྷ་མོ་འདིར་བྱོན་མཆོད་གཏོར་བཞེས། །

སྣང་སྲིད་བྷ་ག་དབྱིངས་ཀྱི་དཀྱིལ་འཁོར་ནས། །

འཁོར་འདས་ཀུན་གྱི་བདག་མོ་དབྱིངས་ཕྱུག་ཡུམ། །

སྔགས་སྲུང་དྲགས་མོ་མ་མོ་མཁའ་འགྲོའི་གཙོ། །

ཡུམ་ཆེན་རལ་གཅིག་འདིར་བྱོན་མཆོད༴ །

བསིལ་བའི་ཚལ་དང་ཧ་ཧ་རྒོད་པ་དང་། །

སིངྒ་གླིང་དང་ཏི་སེ་གངས་རི་དང་། །

དར་ལུང་གནས་དང་ཁའུའི་བྲག་རྫོང་ནས། །

ཞིང་སྐྱོང་དབང་པོ་འདིར༴ །

དུར་ཁྲོད་བརྒྱད་དང་ལྷོ་ཕྱོགས་རི་སུལ་དང་། །

རྡོ་རྗེ་གདན་དང་དཔལ་གྱི་བསམ་ཡས་དང་། །

ན་ལ་ཙེ་དང་དཔལ་ལྡན་ས་སྐྱ་ནས། །

ལས་མགོན་ཕོ་མོ་འདིར་བྱོན༴ །

བྱང་ཤར་མ་རུ་རྩེ་ཡི་དུར་ཁྲོད་དང་། །

རྒྱ་གར་བང་སོ་དམར་པོའི་བྲག་རི་དང་། །

དར་ལུང་བྲག་ཧྲམ་ལ་སོགས་གནས་མཆོག་ནས། །

གནོད་སྦྱིན་ལྕམ་དྲལ་འདིར༴ ཨོཾ་ཤྲཱི་མ་ཧཱ་ཀཱ་ལཱ་ཡ། མ་ཧཱ་ཀཱ་ལི་དེ་ཝཱི་ཨེ་ཀ་ཛཱ་ཊྀ་དེ་ཝཱི་ཨིནྡྲ་སཱི། བཛྲ་ཀཱ་ལ་ཡཀྵ་ཡེ། ཀཱ་ལི་ཡཀྵ་སཱི། ཡཀྵ་པུ་ཏྲ་ཡེ། རུ་ཏྲ་བྷ་ཏྲ་ཡེ། རུ་ལུ་རཀྑ་སཱི། ས་པ་རི་ཝཱ་ར་ན་མཿསརྦ་ཏ་ཐཱ་ག་ཏེ་བྷྱོ་བི་ཤྭ་མུ་ཥེ་བྷྱཿསརྦ་ཐཱ་ཁཾ་ཨུངྒ་ཏེ་སྥ་ར་ཎ་ཨི་མྃ་ག་ག་ན་ཁཾ་གྲིཧྞ་དམ་བ་ལིང་དེ་སྭཱ་ཧཱ། ཨོཾ་མ་ཧཱ་ཀཱ་ལ་ཀཱ་ལ་བི་ཀ་ལ་ར་ཏྲི་ཏ། ཌོམྦྷི་ནཱི། ཙཎྜ་ལཱི། རཀྑ་སཱི། སིངྒ་ལཱི། དེ་ཝཱི་ས་པ་རི་ཝཱ་ར་ན་མཿསརྦ༴ སོགས། ཨོཾ་གྷྲྀ་རཱ་ཛ་གྷྀ་རཱ་ཛ། ཀུ་མ་ཀུ་མ་ཁུམ་ཐིས་སྭཱ་ཧཱ། ཤྲཱི་ཤྨ་ཤ་ན་ཨ་དྷི་པ་ཏི་མ་ཧཱ་པི་ཤཱ་ཙི་བ་ལིངྟ་ཁ་ཁ་ཁཱ་ཧི་ཁཱ་ཧི། ཨོཾ་རཀྨོ་པྲཱ་ཎ་ཨཱཏྨ་ས་པ་རི་ཝཱ་ར་སརྦ་བིགྷྣན་ཤ་ཏྲཱུྃ་མཱ་ར་ཡ་ཨི་དམ་བྷ་ལིང་ཏ་གྲིཧྣ་པ་ཡ་ཁ་ཁ་ཁཱ་ཧི་ཁཱ་ཧི། ཨོཾ་ཤྲཱི་དྷརྨ་པཱ་ལ་ཨརྒྷཾ༴ ནས་ཤབྡ་ཨཱཿཧཱུྃ། ཨོཾ་ཤྲཱི་དྷརྨ་པཱ་ལ་ཨོཾ་ཨཱཿཧཱུྃ། (གིས་ནང་མཆོད་ཕུལ།) དཔལ་རྡོ་རྗེ་ནག་པོ་ཆེན་པོ་ལྕམ་དྲལ་འཁོར་དང་བཅས་པ་ཁྱེད་རྣམས་མཆོད་སྦྱིན་གྱི་གཏོར་མ་རྒྱ་ཆེན་པོ་འདི་བཞེས་ལ་སངས་རྒྱས་ཀྱི་བསྟན་པ་སྲུངས། དཀོན་མཆོག་གི་དབུ་འཕང་བསྟོད། དགེ་འདུན་གྱི་སྡེ་སྐྱོངས། རྣལ་འབྱོར་པ་བདག་ཅག་དཔོན་སློབ་ཡོན་མཆོད་འཁོར་དང་བཅས་པའི་བར་ཆད་མི་མཐུན་པའི་ཕྱོགས་ཐམས་ཅད་ཞི་བ་དང་། བསམ་སྦྱོར་ཆོས་མཐུན་ཡིད་བཞིན་དུ་འགྲུབ་ཅིང་། བྱང་ཆུབ་མ་ཐོབ་ཀྱི་བར་དུ་བསྲུང་བསྐྱབ་སྦ་བའི་ཕྲིན་ལས་སྟོང་གྲོགས་རྒྱ་ཆེན་པོ་མཛད་དུ་གསོལ། (ཞེས་ཕྲིན་ལས་བཅོལ།)

གསོལ་ལོ་མཆོད་དོ་རྒྱལ་བའི་བསྟན་སྲུང་ཚོགས། །

བསྒྲུབ་བོ་རྟེན་ནོ་བླ་མའི་བཀའ་སྲུང་ཆེ། །

འབོད་དོ་བསྐུལ་ལོ་རྣལ་འབྱོར་དགྲ་ལྷའི་ཚོགས། །

རིངས་པའི་ཚུལ་གྱིས་འདིར་བྱོན་མཆོད་གཏོར་བཞེས། །

ཤ་ཁྲག་དམར་གྱིས་བརྒྱན་པའི་གཏོར་མས་མཆོད། །

ཛ་གད་སྨན་ཕུད་རཀྟའི་བཏུང་བས་མཆོད། །

རྔ་ཆེན་རྐང་གླིང་རོལ་མོའི་སྒྲ་ཡིས་མཆོད། །

དར་ནག་འཕན་ཆེན་སྤྲིན་ལྟར་གཏིབས་པས་མཆོད། །

ཡིད་འཕྲོག་སྤྱན་གཟིགས་ནམ་མཁའ་མཉམ་པས་མཆོད། །

རབ་བརྗིད་སྙན་པའི་དབྱངས་ཀྱི་ང་རོས་མཆོད། །

ཕྱི་ནང་གསང་བའི་དམ་རྫས་རྒྱ་མཚོས་མཆོད། །

བདེ་སྟོང་དབྱེར་མེད་ཡེ་ཤེས་རོལ་པས་མཆོད། །

སངས་རྒྱས་བསྟན་པ་གཉེན་པོ་ཁྱེད་ཀྱིས་སྲུངས། །

དཀོན་མཆོག་དབུ་འཕང་གཉེན་པོ་ཁྱེད་ཀྱིས་བསྟོད། །

དཔལ་ལྡན་བླ་མའི་ཕྲིན་ལས་ཁྱེད་ཀྱིས་སྤེལ། །

རྣལ་འབྱོར་བཅོལ་བའི་འདོད་དོན་ཁྱེད་ཀྱིས་སྒྲུབས། །

མཆོད་པའི་ཡོ་བྱད་མ་འབྱོར་ཉམས་པ་སོགས། །

ནོངས་པ་གང་མཆིས་ཆོས་སྐྱོང་སྲུང་མར་བཤགས། །

མཆོག་ཐུན་དངོས་གྲུབ་མ་ལུས་རྩོལ་བ་དང་། །

རྟེན་ལ་བཞུགས་སམ་རང་གནས་གཤེགས་སུ་གསོལ། །

སྔོན་ཚེ་རྡོ་རྗེ་འཆང་གི་སྤྱན་སྔ་རུ་སོགས། །(ཅེས་བརྗོད་པར་བྱའོ། །ཞེས་པའང་ཆོས་ཀྱི་བློ་གྲོས་པས་སོ།། །།)
