སངས་རྒྱས་བྱེ་བ་ཕྲག་བརྒྱ་རྩ་བཞིས་མགྲིན་གཅིག་ཏུ་གསུངས་པ་ཡེ་ཤེས་ཏ་ལའི་གཟུངས།

༄༅། ། སངས་རྒྱས་བྱེ་བ་ཕྲག་བརྒྱ་རྩ་བཞིས་མགྲིན་གཅིག་ཏུ་གསུངས་པ་ཡེ་ཤེས་ཏ་ལའི་གཟུངས། སྱཱདྱ་ཐེ་དན་ཙཀྵུཿ ཤུ་རུ་ད། ཙཀྵུཿ བྷ་བྷ། དྷུ་ལ་མེ་ཐཱ། ཀ་ལ་ཏ་ཐ། ཨི་ཏ་ཏི་ཏ་ཐཾ་ས། སུ་རཊྚསུ་རཊྚ། སུ་ཏ་ཐ་ཡ། ཨི་ཏ་ཐཾ་ས། བེ་ལཱ་བེ་ལཱ། པཱ་ཎི། ཙ་རུ་མཱུརྟི་ནི། ཨཱ་ར་ནི་ཨཱ་ར་ནི། ཀཱ་ལ་པཱ་ནི། ཀཱ་ལ་པཱ་ནི། ཏུ་རུ་དུ་སི། ཏུ་རུ་དུ་སི༑དྷ་སུ་ཏི་དྷ་སུ་ཏི། དྷི་རི་དྷི་རི། དྷུ་རུ་དྷུ་རུ། དྷུ་རུ་དྷུ་རུ། ཀཱ་ལ་ཀཱ་ལ། ས་ཏེ་ཐཱ་ས། གཱི་ལ་གཱི་ལ། གཱི་ལ་པ་ཡ། གཱི་ལ་པ་ཡ། དྷུ་སུ་དྷུ་སུ། སུཾ་དྷུ་སུཾ་དྷུ། སིངྠཱ་སུ། ཨེཏྠཱ་སུ། ཨེ་ཐཱ་ས། ཨེ་ཐཱ་ས་པཱ་ནི། ཡུ་ཅུ་རེ་དྷ་སུ། དྷ་རེ། ཀ་ར་ཀ་ར། ཀི་རི་ཀི་ལི། ཀུ་རུ་ཀུ་རུ། ཀུརྨ་ཀུརྨ། ཀུརྨ་པཱ་ཎི། ཀུརྨ་པཱ་ནི། ཀེ་ལུ་ཀེ་ལུ། ཀེ་ལ་པཱ་ནི། ཀེ་ལ་པཱ་ཏི། ཀངྐཱ་རི་ཀངྐཱ་རི། ལ་རུ་བུདྡྷེ ༑ དྷུ་རུ་དྷེ། དྷུ་རུ་དྷེ། མཧཱ་དྷུ་རུ་དྷེ། ཀ་ར་ཀ་ར། ཀི་རི་ཀི་རི། པི་ཧུ་སི་པི་ཧུ་སི། དྷ་སུ་དྷ་སུ། ཧ་སུ་པ་ཎི་སྭཱཧཱ། ། འདི་ལན་ཉི་ཤུ་རྩ་གཅིག་བརྗོད་ན་སེམས་ཅན་དམྱལ་བ་པ་ཡང་ཐར་པ་ཐོབ་པར་སོམ་ཉི་མི་དགོས་སོ། །
