སོ་སོར་ཐར་པའི་མདོ། 

༄༅། ། རྒྱ་གར་སྐད་དུ། པྲ་ཏི་མོཀྵ་སུཏྲ། བོད་སྐད་དུ། སོ་སོར་ཐར་པའི་མདོ། ཐམས་ཅད་མཁྱེན་པ་ལ་ཕྱག་འཚལ་ལོ། ། སྙན་པའི་བ་དན་ འཇིག་རྟེན་གསུམ་དུ་གྲགས། ། དམ་པའི་ཆོས་སྒྲ་སེངྒེའི་སྒྲ་སྒྲོགས་པ། ། ཐམས་ཅད་མཁྱེན་པ་དཀོན་མཆོག་མཛོད་བརྙེས་པ། ། ཞབས་ལ་ཚངས་ དབང་གཙུག་གི་ནོར་བུས་གཏུགས། ། སྡུག་བསྔལ་རྒྱ་མཚོ་གཏིང་མཐའ་མེད་ བརྒལ་བ། ། འགྲོ་བའི་གཙོ་ལ་སྤྱི་བོས་ཕྱག་འཚལ་ཏེ། ། ཐམས་ཅད་མཁྱེན་ པའི་བསླབ་གཞི་དཀོན་མཆོག་སྣོད། ། འཕགས་པའི་ཚོགས་ཀྱི་དབུས་སུ་དབྱེ་ བར་བྱ། ། སངས་རྒྱས་འདུལ་བ་གངས་ཆེན་མཚོ། ། གཏིང་མཐའ་མེད་པ་ ཐམས་ཅད་ཀྱི། ། གནས་པའི་སྙིང་དང་སྙིང་པོ་ནི། ། སོ་སོར་ཐར་པ་འདི་ ཡིན་ནོ། ། འདི་ནི་དམ་ཆོས་རྒྱལ་པོ་ཡི། ། ཆོས་ཀུན་གྱི་ནི་འདྲེན་པ་མཆོག་༑ འདི་ནི་དགེ་སློང་ཚོང་ཚོགས་ཀྱི། ། བསླབ་ཟོང་ཚོང་ཁང་ཆེན་པོ་ཡིན། ། ཚུལ་ཁྲིམས་འཆལ་བས་ཟིན་རྣམས་ཀྱི། ། གདུག་པ་རྣམ་གསལ་སྨན་འདི་ཡིན། འདི་ནི་ལང་ཚོ་རྣམ་འཁྲུལ་བའི། ། རྐང་གཉིས་གཞོན་ནུའི་ལྕགས་ཀྱུ་ཡིན། མཚོ་བས་ཟབ་པའི་འཁོར་བ་ལས། ། སྒྲོལ་བའི་རྒལ་ཐབས་འདི་ཡིན་ཏེ། ། འདི་ནི་བཟང་འགྲོར་འགྲོ་བ་ཡི། ། ངེས་པའི་ཆུ་ལོན་ཟམ་པ་ཡིན། ། འདི་ནི་ཉོན་མོངས་ཕམ་བྱེད་ལམ། ། རྒྱལ་པོ་ཡི་ནི་འདྲེན་པ་མཆོག ། འདི་ནི་ཐར་ པའི་གྲོང་འཇུག་པའི། ། ཐེམ་སྐས་གཞི་དང་འདྲ་བར་གནས། ། ང་ནི་མྱ་ ངན་འདས་གྱུར་ན། ། འདི་ནི་ཁྱོད་ཀྱི་སྟོན་པའོ་ཞེས། ། རང་བྱུང་ཉིད་ཀྱིས་ གུས་བཅས་པར། ། ནན་ཏན་དགེ་སློང་ཚོགས་འདུན་བསྟོད། ། སངས་རྒྱས་ ཞེས་བྱའི་སྒྲ་འདི་ཡང་། ། འཇིག་རྟེན་དག་ན་རབ་ཏུ་དཀོན༑ ༑ མི་ཉིད་ རྙེད་པར་ཤིན་ཏུ་དཀའ། ། རབ་ཏུ་བྱུང་བའང་ཤིན་ཏུ་དཀོན། ། དེ་བཞིན་ རབ་ཏུ་བྱུང་རྣམས་ཀྱི། ། ཚུལ་ཁྲིམས་ཕུན་སུམ་ཚོགས་རབ་དཀོན། ། ཚུལ་ ཁྲིམས་ཡོངས་སུ་དག་གྱུར་ཀྱང་། ། གྲོགས་བཟང་ཤིན་ཏུ་རྙེད་པར་ དཀའ། ། འཇིག་རྟེན་སངས་རྒྱས་འབྱུང་བ་དང་། ། མི་དང་རབ་ཏུ་བྱུང་བ་ དང་། ། ཚུལ་ཁྲིམས་ཕུན་སུམ་ཚོགས་པ་དང་། ། གྲོགས་བཟང་རྙེད་དཀའ་ རྙེད་གྱུར་ནས། ། མཁས་པ་བདག་ལ་ལེགས་འདོད་ཅིང་། ། དེ་དག་འབྲས་ བཅས་བྱེད་འདོད་པས། ། སྡོམ་བརྩོན་རྣམས་ཀྱི་སོ་སོ་ཐར། །མཉན་པའི་ ཕྱིར་ནི་འབད་པར་བྱ། ། དགེ་སློང་དབང་འདུལ་ཀུན་དབང་པོ། ། སངས་ རྒྱས་མི་འགྱུར་རབ་བྱུང་བ། ། ངེས་པར་ཐར་པ་འདོད་རྣམས་ཀྱི། ། སོ་སོར་ ཐར་པ་རྟག་ཏུ་བསྲུང་། ། བསྐལ་པ་བྱེ་བ་རྣམས་སུ་ཡང་། ། སོ་སོར་ཐར་པ་ ཐོས་པ་དང་། ། གཟུང་དང་འཛིན་པ་རྙེད་དཀའ་སྟེ། ། སྒྲུབ་པའང་ཤིན་ཏུ་ རྙེད་པར་དཀའ། ། སངས་རྒྱས་རྣམས་ནི་འབྱུང་བ་བདེ། ། ཆོས་བསྟན་པ་ ཡང་བདེ་བ་ཡིན། ། དགེ་འདུན་མཐུན་པ་བདེ་བ་སྟེ། ། མཐུན་པ་རྣམས་ཀྱི་ དཀའ་ཐུབ་བདེ། ། འཕགས་པ་རྣམས་ནི་མཐོང་བ་བདེ། ། དམ་པ་དག་དང་འགྲོགས་པ་བདེ། ། བྱིས་པ་རྣམས་ནི་མ་མཐོང་ན། ། རྟག་ཏུ་བདེ་བ་ ཉིད་དུ་འགྱུར། ། ཚུལ་ཁྲིམས་ལྡན་པ་མཐོང་བ་བདེ། ། མང་དུ་ཐོས་པ་ མཐོང་བ་བདེ། ། ཡང་སྲིད་རྣམ་པར་གྲོལ་བ་ཡི། ། དགྲ་བཅོམ་པ་དག་ མཐོང་བ་བདེ། ། འཇུག་ངོགས་བདེ་བའི་ཆུ་ཀླུང་བདེ། ། ཆོས་ལ་གོམས་ པའི་སྐྱེ་བོ་བདེ༑ ༑ཤེས་རབ་ཐོབ་པར་གྱུར་པ་བདེ། ། ངའོ་ང་རྒྱལ་ཟད་ པར་བདེ༑ ༑ངེས་པར་བྱས་ཤིང་དབང་པོ་ཐུལ་བ་དག ། དགོན་པ་ཞི་བ་ རྣམས་སུ་རྒས་འགྱུར་དང་། ། མང་དུ་ཐོས་པ་ནགས་ཀྱི་ནང་དག་ཏུ་༑ ༑ ལང་ཚོ་ཡོལ་བ་རྣམས་ཀྱི་གནས་པ་བདེ། ། ཤཱཀྱ་སེངྒེ་དེ་ལ་ནི། ། སོར་མོ་བཅུ་ ཡི་ཐལ་སྦྱར་ཏེ། ། སོ་སོ་ཐར་པ་གདོན་པར་བྱ། ། གདུལ་བའི་དོན་དུ་ང་ལས་ ཉོན། ། ཐོས་ནས་དྲང་སྲོང་ཆེན་པོ་ཡིས། ཇི་སྐད་གསུངས་བཞིན་སྒྲུབ་བྱ་ ཞིང་། ། ཁ་ན་མ་ཐོ་ཕྲ་རྣམས་ལ། ། བསྟུན་ཅིང་བྱེད་པ་ཉིད་དུ་གྱིས། ། རྟག་ཏུ་འབད་པས་འདའ་བ་ཡི། ། སེམས་རྟ་ཁ་གླན་དཀའ་བ་ལ། ། མཐུན་ པའི་གཟེར་གནོན་བརྒྱབ་པ་ཡི། ། སྲབ་ནི་སོ་སོ་ཐར་འདི་ཡིན། ། ཆེ་བ་གང་ དག་ངག་ཙམ་གྱིས། ། ལྡོག་ཅིང་མཚམས་ལས་མི་འདའ་བ། ། དེ་དག་མི་རྟ་ བཟང་པོ་སྟེ། ། ཉོན་མོངས་གཡུལ་ལས་ངེས་རྒྱལ་འགྱུར། ། སུ་ལ་སྲབ་འདི་ མེད་པ་དང་། ནམ་དུའང་འདོད་པར་མི་འགྱུར་བ། ། དེ་དག་ཉོན་མོངས་ གཡུལ་གྱིས་དཀྲུགས། ། འཇོག་བྲལ་རྣམ་པར་འཁྱམས་པར་འགྱུར། ། མི་ ཚངས་སྤྱོད་དང་རྐུ་བ་དང་། ། མི་ནི་བསད་པར་བྱ་བ་དང་། ། རྫུན་དུ་སྨྲ་ དང་བཅས་པ་ཡི། ། ཆོས་བཞི་འདིར་ནི་གསུངས་པ་ཡིན། ། ཁུ་བ་འཛིན་དང་འཁྲིག་ཚིག་བསྙེན་བཀུར་སྨྱན། ཁང་པ་ ཁང་ཆེན་དང་ནི་གཞི་མེད་ དང་། ། བག་ཙམ་དགེ་འདུན་དབྱེན་དང་དེ་རྗེས་ཕྱོགས༑ ༑ཁྱིམ་སུན་ འབྱིན་དང་བཀའ་བློ་མི་བདེ་བའོ། ། དབེན་པ་སྐྱབས་ཡོད་འདུག་པའོ། ། འཆང་དང་འབྲལ་བ་འཇོག་པ་དང་། ། འཁྲུར་འཇུག་པ་དང་ལེན་པ་ དང་། ། སློང་དང་སྟོད་གཡོག་སྨད་གཡོག་ བཅས། ། རིན་ཐང་སོ་སོར་ བསྐུར་བའོ། སྲིན་བལ་འབའ་ཞིག་ཆ་གཉིས་དང་། ། དྲུག་དང་མཐོ་གང་ ལམ་དང་ནི། ། འཁྲུ་བ་དང་ནི་གསེར་དངུལ་དང་། ། མངོན་ཚན་ཅན་དང་ ཉོ་ཚོང་ངོ་། ། ལྷུང་བཟེད་གཉིས་དང་ཐ་ག་གཉིས། ། བྱིན་འཕྲོག་སྟོན་ཟླ་ཐ་ ཆུང་དང་། ༑ དགོན་པ་བ་དང་རས་ཆེན་དང་། ། བསྔོས་པ་དང་ནི་གསོག་ འཇོག་གོ ། ཤེས་བཞིན་དང་ནི་ས་བོན་དང་། ། མ་བསྐོས་པ་དང་ཡང་ཡང་ དང་། ། ཆུ་དང་ཁྱིམ་དང་བསམ་བཞིན་དང་། ། མགྲོན་མང་རྐུན་མ་ཆོས་ སྟོན་ནོ། ། རྫུན་སྐྱོན་དགེ་སློང་ཕྲ་མ་དང་། ། སྐྱོ་སྔོགས་བྱེད་དང་སྟོན་པ་དང་། འདོན་དང་གནས་ངན་ལེན་དང་ཆོས། ། བཤེས་ངོར་བྱེད་ དང་ཁྱད་དུ་ གསོད། ། ས་བོན་འཕྱ་བ་བསྒོ་བ་དང་༑ ༑ཁྲི་དང་འདིངས་དང་བསྐྱོད་པ་ དང་། ། ཕྱིས་ནོན་འབྱུང་བ་འདེབས་པ་དང་། ། རིམ་པ་གཉིས་སུ་བརྩིག་ པའོ། ། མ་བསྐོས་ཉི་མ་ནུབ་པ་དང་། ། ཟས་དང་ཆོས་གོས་གཉིས་དག་ དང་། ། དོན་མཐུན་གྲུ་དང་དབེན་པ་གཉིས། ། དགེ་སློང་མ་ཡིས་བཅུག་ པའོ། ། ཡང་ཡང་དང་ནི་འདུག་གནས་གཅིག ། ཕྱེ་དང་བཅའ་དང་སྟོབ་པ་ དང་། ། འདུས་དང་དུས་མིན་གསོག་འཇོག་དང་། ། ཁ་ནས་མིད་དང་ བསོད་པ་ཉིད། ། སྲོག་ཆགས་དང་བཅས་ཉལ་སར་འདུག ། འགྲེང་དང་ གཅེར་བུར་དམག་དང་ནི། ། ཞག་གཉིས་བཤམ་དཀྲུག་འགྲོ་བ་དང་ ༑ ༑ བརྡེག་དང་གཟས་དང་གནས་ངན་ལེན། ། བདེ་དང་མེ་དང་འདུན་པ་ དང་། ། བསྙེན་པར་མ་རྫོགས་ཆོས་དང་སྨྲ། ། དགེ་ཚུལ་ཁ་དོག་བསྒྱུར་བ་ དང་། ། རིན་པོ་ཆེ་དང་ཚོ་བའི་དུས། ། དུད་འགྲོ་འགྱོད་པ་སོར་མོ་ དང་། ། རྩེ་དང་ལྷན་གཅིག་དངངས་བྱེད་དང་། ། སྦེད་དང་གདེངས་མེད་ གཞི་མེད་དང་། ། སྐྱེས་པ་མེད་པར་ལམ་འགྲོ་བའོ༑ ༑རྐུ་དང་ཉི་ཤུ་མ་ལོན་ དང་། ། རྐོ་དང་མགྲོན་དང་བསླབ་པ་དང་ ༑ ༑ འཐབ་དང་མི་སྨྲ་འགྲོ་བ་ དང་། ། མི་གུས་ཆང་འཐུང་དུས་མིན་པའོ། ། ཟས་བཅས་སྐྱ་རེངས་ད་སྡོད་ དང་། ། ཁབ་རལ་དང་ནི་ཁྲི་རྐང་དང་། ། གདལ་དང་འདིང་དང་གཡན་པ་ དང་། ། རས་ཆེན་བདེ་གཤེགས་ཆོས་གོས་སོ། ། གྲོང་དང་ཁྱིམ་གཞན་ཉིད་ དང་ནི། བསླབ་ པ་ རྣམས་ དང་ དགོན་ པ་ ནི། ། སངས་ རྒྱས་ ཕན་ པར་ གསུངས་པ་ཡིས། ། སོ་སོར་བཤགས་པར་བྱ་བར་གསུངས། ། ཤམ་ཐབ་ལ་ ནི་རྣམ་བདུན་དང་། ། སྟོད་གཡོགས་ལ་ནི་རྣམ་གསུམ་དང་། ། ཤིན་ཏུ་ བསྡམས་ལ་སོགས་པ་ལྔ། ། མགོ་གཡོགས་ལ་སོགས་རྣམ་པ་ལྔ། ། མཆོང་བ་ ལ་སོགས་རྣམ་པ་ལྔ། ལུས་ལ་སོགས་པ་རྣམ་པ་ལྔ། འདུག་པར་བྱ་བར་དགུ་ དག་དང་། ། བྱིན་ལེན་བྱ་བར་བརྒྱད་རྣམས་སོ། ། ཟས་ལ་ལེགས་པར་བྱ་བ་ དྲུག ། ཙུག་ཙུག་ལ་སོགས་རྣམ་པ་ལྔ། ། འབྲུ་ནས་ཐ་དད་བྱེད་པ་ལྔ། ། ལག་པ་བལྡག་ལ་སོགས་པ་ལྔ། ། འཕྱ་ལ་སོགས་པ་རྣམ་པ་བཞི། ། ལྷུང་ བཟེད་ལ་ཡང་རྣམ་པ་བཅུ། ། འགྲེང་བར་བྱེད་ལ་ སོགས་པ་ལྔ། ། མགོ་ གཡོགས་ལ་སོགས་རྣམ་པ་ལྔ། ། དོ་ཀེར་ཅན་ལ་སོགས་པ་ལྔ། ། གླང་ཆེན་ལ་ སོགས་བཞོན་པ་ལྔ། ། ལག་ན་འཁར་བ་ལ་སོགས་དྲུག ། ན་བ་རྣམ་པ་བཞི་ རྣམས་སོ། ། མངོན་སུམ་དྲན་པ་མ་མྱོས་དང་། ། དེ་བཞིན་གང་མང་ངོ་བོ་ ཉིད། ། རྩྭ་རྣམས་བཀྲམ་པ་ལྟ་བུ་དང་། ། མཁས་བླང་བར་ཡང་བྱ་བའོ། ། བཟོད་པ་དཀའ་ཐུབ་དམ་པ་བཟོད་པ་ནི། ། མྱ་ངན་འདས་པ་མཆོག་ཅེས་ སངས་རྒྱས་གསུངས། ། རབ་ཏུ་བྱུང་བ་གཞན་ལ་གནོད་པ་དང་། ། གཞན་ ལ་འཚེ་བ་དགེ་སྦྱོང་མ་ཡིན་ནོ། ། མིག་ལྡན་འགྲོ་བ་ཡོད་པ་ཡིས། ། ཉམ་ང་བ་ དག་ཇི་བཞིན་དུ། ། མཁས་པས་འཚོ་བའི་འཇིག་རྟེན་འདིར། ། སྡིག་པ་ དག་ནི་ཡོངས་སུ་སྤོང་། ། བསྐུར་བའང་མི་གདབ་གནོད་མི་བྱ། ། སོ་སོར་ ཐར་པ་བསྡམས་པར་བྱ༑ ༑ཟས་ཀྱི་ཚོད་ཀྱང་རིག་པར་བྱ། ། བས་མཐའི་ གནས་སུ་གནས་པར་བྱ། ། ལྷག་པའི་སེམས་ལ་ཡང་དག་སྦྱོར། ། འདི་ནི་ སངས་རྒྱས་བསྟན་པ་ཡིན། ། ཇི་ལྟར་བུང་བ་མེ་ཏོག་ལས། ། ཁ་དོག་དྲི་ལ་མི་ གནོད་པར། ། ཁུ་བ་གཞིབས་ནས་འཕུར་བ་ལྟར། ། དེ་བཞིན་ཐུབ་པ་གྲོང་ དུ་རྒྱུ། ། བདག་གི་རིག་དང་མི་རིག་ལ། ། བརྟག་པར་བྱ་སྟེ་གཞན་རྣམས་ ཀྱི། ། མི་མཐུན་པ་དང་གཞན་དག་གིས། །བྱས་དང་མ་བྱས་རྣམས་ལ་ མིན། ། ལྷག་པའི་སེམས་ལ་བག་བྱ་སྟེ། ། ཐུབ་པའི་བསླབ་གཞི་རྣམས་ལ་ བསླབ། ། ཉེར་ཞི་རྟག་ཏུ་དྲན་ལྡན་པའི། ། སྐྱོབ་པ་མྱ་ངན་མེད་པ་ཡིན། ། སྦྱིན་པས་བསོད་ནམས་རབ་ཏུ་འཕེལ། ། ལེགས་བསྡམས་དགྲ་ སོགས་ མི་ འགྱུར་རོ། །
