༄༅། །ན་མ་ཨཱརྻ་ཏཱ་རཱ་ཡ། རྗེ་བཙུན་འཕགས་མ་སྒྲོལ་མ་ཉིད་ཀྱིས་རྗེས་སུ་གནང་བ་མཛད་པ་ལ་བརྟེན་ནས་སློབ་དཔོན་ངག་གི་དབང་ཕྱུག་གྲགས་པས་འཆི་བ་སླུ་བ་བསྡུས་པའི་རྟོག་པ་ནས་འདི་ཕྱུང་བར་མཛད། མན་ངག་རྣམས་སྒྲོལ་མ་ཉིད་ཀྱིས་གསུངས་སྐད། དེ་ལ་བླ་མ་རྡོ་རྗེ་གདན་པས་གསན་ནས། སྒྲུབ་ཐབས་ཀྱང་མང་དུ་མཛད། དེ་ལ་བླ་མ་བ་རི་ལོ་ཙཱ་བས་གསན། སྒྲུབ་ཐབས་ཀྱང་མང་པོ་བོད་དུ་བསྒྱུར། དེ་ལ་བླ་མ་གླེ་ནག་པས་གསན་སྒྲུབ་པ་ལོ་དྲུག་མཛད་པས་ཞལ་མངོན་སུམ་དུ་གཟིགས་ཏེ་ཆོས་གསུངས་སྐད། དེ་ལ་སྒྲེ་པ་རིན་པོ་ཆེས་གསན་ཏེ། དང་པོ་ཐོས་བསམ་མཛད་པའི་དུས་སུ་ལ་སྟོད་སྟག་རི་བ་ཡིན་ཟེར་བའི་རྣལ་འབྱོར་པ་གཅིག་ལ་ལག་རིས་བསྟན་པས། ཇོ་བོ་བཙུན་པ་ཁྱོད་ལ་ཆོས་ཀྱི་ལས་འཕྲོ་གཤའ་མ་གཅིག་འདུག་པ་ལ་ཚེ་ལ་ལོངས་མི་འདུག་སྟེ། ལོ་གསུམ་ལས་མི་ཐུབ་པར་འདུག་ཟེར་བ་དང༌། སྔར་ནས་ཁོང་རང་གི་རྨི་ལམ་དུ་ལྟས་ངན་ཡང་ཡང་བྱུང་བ་གཉིས་འཛོམ་པས། ཚེ་འདི་ལ་ཐུགས་གཤའ་མ་རང་བྱུང་སྟེ་ལོ་གཅིག་ཐུབ་བམ་ཇི་ཙམ་ཐུབ་ཐུབ་ཀྱི་ཚེ་ལྷག་མ་འདི་ལ་ཆོས་ལག་ལེན་གཤའ་མར་ཐེབས་པ་གཅིག་བྱ། དེ་ལ་ཐོག་མར་གདམས་ངག་ཁྱད་པར་ཅན་སྟོན་པའི་དགེ་བའི་བཤེས་གཉེན་གཅིག་བཙལ་དགོས་དགོངས་ནས་གསན་འཕྲོ་བཅད་ནས་དབུས་སུ་བྱོན་པའི་ལམ་ཁར་གཞུའི་མཁར་གོང་དུ་བླ་མ་གླེ་ནག་པ་དང་མཇལ་ནས་ལོ་རྒྱུས་བྱས་པས། བྱང་ཆུབ་སྒྲུབ་པ་ལ་ལམ་གྱི་བར་ཆད་སེལ་བའི་ཐབས་ཁྱད་པར་ཅན་ང་ལ་ཡོད་ཀྱི། སྟོན་པ་ཁྱོད་ཐུགས་བདེ་འབོལ་གྱིས་ཐོང་ཞིག་གསུངས་ནས་འདིའི་སྒྲུབ་ཐབས་གདམས་ངག་དང་བཅས་པ་མ་ལུས་པ་གནང༌། ད་ཁྱོད་ཀྱིས་གཞན་དུ་མ་འགྲོ་བར་འདི་རང་དུ་སྡོད་ལ་སྒོམས་ཤིག་གསུངས་ནས། ལུང་པ་དེ་རང་གི་ལུང་ལག་མཁར་ལུང་བྱ་བ་ཅིག་གི་ཕུར་སྒྲུབ་པ་རྩེ་གཅིག་ཏུ་མཛད་པས་ཟླ་བ་བཅུ་གཅིག་ལོན་ཙ་ན་རྗེ་བཙུན་མའི་ཞལ་མངོན་སུམ་དུ་གཟིགས་ནས་ཁྱོད་ཀྱིས་ལོ་དྲུག་ཅུ་ཙམ་ཐུབ།

སེམས་ཅན་ལ་ཕན་ཐོགས་པ་འོངས་གསུངས། དེ་ནས་ཕྱིས་དགུང་ལོ་དྲུག་ཅུ་ལོན་པའི་དུས་སུ་རྗེ་བཙུན་མ་ལ་གསོལ་བ་བཏབ་པས། ངའི་སྐུ་གཅིག་བཞེངས་དང་ལོ་བཅུ་ཙམ་སྡོད་པར་འདུག་གི་གསུངས་ནས་རས་བྲིས་གཅིག་བཞེངས་སྐད། དགུང་ལོ་བདུན་ཅུ་བཞེས་པའི་དུས་སུ་གསོལ་བ་བཏབ་པས་ཡང་ངའི་སྐུ་གཅིག་བཞེངས་དང་ལོ་བཅུ་ཐུབ་པར་འདུག་གསུངས་ནས། ལུགས་མ་གཅིག་བཞེངས་སྐད། སྐུ་དེ་གཉིས་བྱིན་རླབས་ཆེ་བར་གྲགས་ཏེ། རས་བྲིས་དེ་ད་ལྟ་ར་སྒྲེང་ན་བཞུགས་སྐད། དགུང་ལོ་བརྒྱད་ཅུ་བཞེས་པའི་དུས་སུ་རྗེ་བཙུན་མ་ལ་གསོལ་བ་བཏབ་པས། ད་དུང་ང་ཡི་སྐུ་གཅིག་བཞེངས་ན། ལོ་མང་དུ་ཐུབ་པར་འདུག་གསུངས་པ་ལ། ཁོང་རང་བཞུགས་པའི་མདུན་གྱི་གྱང་ལོགས་ལ་རི་མོ་ཅིག་བཞེངས་སྐད། དེ་ནས་ལོ་བཅོ་ལྔའི་བར་དུ་བཞུགས་ཏེ། ཡོངས་སུ་རྫོགས་པ་དགུ་ལོ་དགུ་བཅུ་རྩ་ལྔར་བཞུགས་སྐད་དོ། །འདིའི་བརྒྱུད་པ་ནི། རྗེ་བཙུན་མ། ངག་དབང་གྲགས་པ། གསེར་གླིང་པ། ཇོ་བོ། འབྲོམ་སྟོན་པ། སྤྱན་སྔ་བ། གླེ་ནག་པ། སྒྲེ་པ། ཞུ་སྟོན། ཨ་སེང༌། རཏྣ་དྷྭ་ཛ། ཡང་སྤྱན་སྔ་བ། དྭགས་པོ་ལྷ་རྗེ། ཞང་སུམ་ཐོག་པ། སྐ་ཚལ་པ། ལྷ་པཎྜི་ཏ།

བླ་མ་ལྷ་པ། རཏྣ་བྷ་དྲ། རིན་ཆེན་བཟང་པོ། རཏྣ་དྷྭ་ཛའོ། ལྷོ་བྲག་པ། །སྤྱན་སྔ་བའི་ཞལ་ནས་འདིའི་བསྙེན་པ་འབུམ་ཚོ་གཅིག་བསྐྱར་ན་བར་ཆད་གང་གིས་ཀྱང་མི་ཚུགས་གསུངས། འདི་ཕྲིན་ལས་ཀྱི་ལྷ་མོ་ཡིན་པས་འགྲུབ་སླ་བ་ཡིན་གསུངས་སོ། །དགེ་བཤེས་གླེ་ནག་པའི་ཞལ་ནས། འདི་ཟློས་ཤེས་ན་མགོ་སུག་བྲལ་ཡང་འགལ་བ་མེད། རོ་ཁུང་བྲུས་ཀྱང་འགལ་བ་མེད། ཤ་དུམ་བུ་དུམ་བུར་འཐོར་གྱིན་སོང་ཡང་འགལ་བ་མེད་གསུངས་སོ། །འཆི་བ་སླུ་བའི་སྒྲོལ་མའི་ལོ་རྒྱུས་རྫོགས་སོ།། ༈ །རྒྱ་གར་སྐད་དུ། སི་ཏཱ་ཏཱ་རཱ་སཱ་དྷ་ནཾ་ནཱ་མ། བོད་སྐད་དུ། སྒྲོལ་མ་དཀར་མོའི་སྒྲུབ་ཐབས་ཞེས་བྱ་བ། འཕགས་མ་སྒྲོལ་མ་ལ་ཕྱག་འཚལ་ལོ། །ཆེད་དུ་ཟླ་བ་དྲུག་གི་བར། །སྒྲོལ་མའི་འཁོར་ལོ་བསྒོམས་བྱས་པས། །དངོས་གྲུབ་ཐམས་ཅད་ཐོབ་འགྱུར་ཞིང༌། །འཆི་བའི་སྐྱོན་གྱིས་མི་གོས་སོ། །ངག་གི་དབང་ཕྱུག་སྣང་བྱས་པའི། །འཆི་བདག་སླུ་བ་བསྡུས་པ་ལས། །ཕྱུངས་ནས་སྒྲོལ་མ་དཀར་མོ་འདི། །བསྒོམས་པས་འཇིགས་པ་ཉམས་བྱེད་པ། །བཞི་པའི་སྡེ་ཚན་དང་པོ་ཨོཾ། །བཅས་པ་ནམ་མཁར་འོད་ཟེར་འཕྲོས། །པད་དཀར་ཟླ་བའི་སྟེང་གནས་པའི། །གསུམ་ལ་སྐྱབས་འགྲོ་ཡང་བརྗོད་བྱ། །རང་བཞིན་སྒྲོ་བཏགས་གཞི་མེད་དུ། །སྐད་ཅིག་སྟོང་པ་ཉིད་དུ་ལྟ། །དེ་ནས་སྔོན་གྱི་སྨོན་ལམ་དབང༌། །བྱང་ཆུབ་སེམས་ཀྱི་ཡི་གེ་བསྒོམ། །པདྨ་དཀར་པོའི་དབུས་གནས་པའི། །ཟླ་བའི་གདན་གྱི་སྟེང་དུ་ནི། །ས་བོན་ཡི་གེའི་འཕྲོ་འདུ་ལས། །རྡོ་རྗེའི་སྐྱིལ་མོ་ཀྲུང་བཅས་པའི། །མཆོག་སྦྱིན་པ་དང་ཨུཏྤལ་འཛིན། །སྟོན་ཀའི་ཟླ་བའི་འོད་རྣམ་ཅན། །ཟླ་བའི་རྒྱབ་ལ་མཉམ་བརྟེན་པའོ། །རྒྱན་རྣམས་ཐམས་ཅད་ཡོངས་སུ་རྫོགས། །བཅུ་དྲུག་ལོ་ལོན་ལུས་ཅན་མ། །སངས་རྒྱས་ཀུན་ཀྱང་དེ་ཡི་སྲས། །ཡུམ་ནི་འདོད་པའི་གཟུགས་འཛིན་མ། །སྒྲོལ་མ་དེ་ཡི་སྙིང་གར་བསྒོམ། །འཁོར་ལོ་འོད་ཟེར་དཀར་པོ་ཅན། །རེའུ་མིག་བརྒྱད་ལ་ཡི་གེ་ནི། །བརྒྱད་ཀྱིས་ཡོངས་སུ་བསྐོར་བའོ། །ཨོཾ་ཧཱ་གསལ་བྱེད་དབུས་གནས་པ། །བསྒྲུབ་བྱའི་མིང་ནི་ལྟེ་བར་བྱ། །ཟླ་བ་དྲུག་ཏུ་ཚངས་ཡིད་ཀྱིས། །ངེས་པ་བརྟན་པོ་དག་ཏུ་བྱ། །ཡི་གེ་བཅུ་པའི་སྔགས་འདི་ནི། །མི་སྐྱོའི་ཡིད་ཀྱིས་བཟླ་བར་བྱ། །ཨོཾ་གྱི་ཡི་གེ་དང་པོར་སྦྱིན། །ཕྱིས་ནི་ཏཱ་རེ་རབ་ཏུ་སྦྱིན། །ཏུཏྟཱ་རེ་དང་ཏུ་རེ་ཕྱི། །སྭཱ་ཧཱ་མཐའ་རུ་ལས་ཀུན་བྱེད། །ཚངས་དབང་ཁྱབ་འཇུག་ཉི་ཟླ་དང༌། །དྲག་པོ་ཕྱོགས་སྐྱོང་འདོད་ལྷས་ཀྱང༌། །བ་སྤུའི་རྩེ་མོའང་བསྐྱོད་མི་ནུས། །འཆི་བདག་ལས་རྒྱལ་སྒྲོལ་མ་བཞིན། །སྐྲ་དཀར་གཉེར་མ་རྒས་ནད་དང༌། །དབུལ་བ་ཉམས་སོགས་མདོར་བསྡུ་ན། །སེང་གེ་ལ་སོགས་འཇིགས་ཆེན་བརྒྱད། །དུ་མ་རབ་ཏུ་འཇོམས་པར་བྱེད། །སྐུལ་བ་མེད་པར་གོས་བཏུང་བ། །ཁ་ཟས་རིན་ཆེན་ཚོགས་ཀྱང་འདུ། །རལ་གྲི་དང་ནི་མིག་སྨན་དང༌། །རྐང་པ་བུམ་པ་བཟང་པོ་འགྲུབ།

ནད་དང་འཆི་བ་རྣམས་ཉམས་པའོ། སྒྲོལ་མ་དཀར་མོའི་སྒྲུབ་པས་ནི། །ཚིགས་བཅད་མཁས་དང་སྙན་ངག་དང༌། །བློ་རྣོ་ཤེས་རབ་དྲི་མེད་འགྱུར། །གཞན་ཡང་དངོས་གྲུབ་ཅི་འདོད་པ། །འཁོར་ལོ་དེ་ལས་འབྱུང་བར་འགྱུར། །གང་ཞིག་བྱུང་བའི་དགེ་བསགས་པས། །དེས་ནི་མྱུར་དུ་སྒྲོལ་མར་ཤོག །སྒྲོལ་མ་དཀར་མོའི་སྒྲུབ་ཐབས་སློབ་དཔོན་ཆེན་པོ་ངག་གི་དབང་ཕྱུག་གྲགས་པས་མཛད་པ་རྫོགས་སོ། །རྒྱ་གར་གྱི་མཁན་པོ་པཎྜི་ཏ་དོན་ཡོད་རྡོ་རྗེ་དང༌། བོད་ཀྱི་ལོ་ཙཱ་བ་ཁམས་པ་བ་རིས་བསྒྱུར་བའོ།། ༈ །སྒྲོལ་དཀར་གྱི་སྒྲུབ་ཐབས་བུམ་ཆོག་གཏོར་མ། སྲུང་འཁོར་གྱི་ཡི་གེ་བཞུགས་སོ།། ༈ །འཕགས་མ་སྒྲོལ་མ་ལ་ཕྱག་འཚལ་ལོ། །བྱང་ཆུབ་སྒྲུབ་པའི་གང་ཟག་འཆི་ལྟས་ཀྱི་མཚན་མ་ཚང་ན་འཕགས་མ་སྒྲོལ་མའི་སྒོ་ནས་འཆི་བ་སླུ་བའི་གདམས་ངག་ནི། སྐྱབས་འགྲོ་སེམས་བསྐྱེད་གཞན་དང་མཐུན། ཁྱད་པར་དུ་བདག་གིས་སེམས་ཅན་ཐམས་ཅད་ཀྱི་ཚེའི་བར་ཆད་ཞི་བར་བྱ་བའི་ཕྱིར། སྒྲོལ་མ་དཀར་མོ་བསྒོམ་སྙམ་དུ་སེམས་བསྐྱེད། དེ་ནས་རང་ཐ་མལ་པའི་སྙིང་གར་པད་ཟླའི་སྟེང་དུ་ཏཱཾ་ཡིག་དཀར་པོ་ཞིག་བསམ། དེ་ནས། སྭ་བྷཱ་ཝས་སྐད་ཅིག་གིས་ཆོས་ཐམས་ཅད་གཞི་མེད་སྟོང་པ་ཉིད་དུ་བསྒོམས་པའི་ངང་ནས་པཾ་ལས་པདྨ། ཨ་ལས་ཟླ་བའི་སྟེང་དུ་ཏཱྃ་ཡིག་དཀར་པོ་འོད་འཕྲོ་བ་ཞིག་བསམ། དེའི་འོད་ཟེར་གྱིས་སེམས་ཅན་གྱི་དོན་བྱས་ཚུར་འདུས་ནས་འོད་དུ་ཞུ་བ་ལས་བདག་ཉིད་འཕགས་མ་སྒྲོལ་མ་སྐུ་མདོག་དཀར་མོ་ཞལ་གཅིག་ཕྱག་གཉིས་མ། པདྨ་དང་ཟླ་བའི་གདན་ལ་རྡོ་རྗེའི་སྐྱིལ་ཀྲུང་གིས་བཞུགས་པ་འོད་ཟེར་ཀུན་ཏུ་འཕྲོ་བ་ཟླ་བའི་རྒྱབ་ཡོལ་ཅན། གཡས་སྐྱབས་སྦྱིན། གཡོན་ཨུཏྤལ་བསྣམས་པ། བཅུ་དྲུག་ལོན་པའི་ལང་ཚོ་དང་ལྡན་པ། དབུ་སྐྲ་ལྟག་པར་བཅིངས་པ། རིན་པོ་ཆེའི་རྒྱན་ཐམས་ཅད་ཀྱིས་བརྒྱན་པར་བསྒོམ་པར་བྱའོ། །དེའི་ཐུགས་ཀར་འཁོར་ལོ་རྩིབས་བརྒྱད་པ་ལ་དབུས་སུ་ཏཱྃ་ཡིག་བསམ་པ་ལས་འོད་ཟེར་འཕྲོས་པས་ཡེ་ཤེས་པ་སྤྱན་དྲངས་རང་ལ་བསྟིམ།

གནས་གསུམ་ཡི་གེས་བརྒྱན། དབང་བསྐུར་སྣང་བ་མཐའ་ཡས་ཀྱིས་དབུར་བརྒྱན། དེ་ནས་ཏཱྃ་ལ་ཡི་གེ་བཅུ་པས་བསྐོར་བ་ལ་དམིགས་ནས་བཟླས་པ་བྱ་བའམ་ཡང་ན་འཁོར་ལོའི་ལྟེ་བར་ཨོཾ་དང་ཧཱ་གཉིས་བསམ། རྩིབས་བརྒྱད་ལ་ཏཱ་ལ་སོགས་པ་བརྒྱད་བསམ། ཨོཾ་ཏཱ་རེ་ཏུཏྟཱ་རེ་ཏུ་རེ་བདག་གི་ཚེ་འཕེལ་བར་གྱུར་ཅིག་སྭཱ་ཧཱ། འཁོར་ལོ་གཡས་སུ་འཁོར་བ་ལ་ཤེས་པ་གཏད་དེ་ཡིད་ཀྱིས་བཟླས་པ་བྱའོ། །འདོད་པའི་དོན་ལ་གསོལ་བ་བཏབ། དགེ་བ་བྱང་ཆུབ་ཏུ་བསྔོ། རྫོགས་རིམ་དམིགས་མེད་དུ་བསྐྱལ། དེ་ནས་སྒྱུ་མ་ལྟ་བུར་ལངས་ལ་བྱ་བ་གཞན་ལ་འཇུག དེ་ལྟ་བུ་ཞག་གམ་ལོའི་བར་དུ་མཚན་མའི་རྟགས་མ་བྱུང་གི་བར་བྱའོ། །མཚན་མ་ངན་ངེས་ན་བྱ་བ་ཡིན་གསུངས། གཏོར་མ་གཏོང་བར་འདོད་ན་རིན་པོ་ཆེའི་སྣོད་དུ་དཀར་གསུམ་གྱིས་བརྒྱན་པའི་གཏོར་མ་བཤམས་ལ། དེའི་ནང་ཟླ་བའི་སྟེང་ཏཱྃ་ཡིག་དཀར་པོ་བསམ་པ་ལས་འོད་ཟེར་འཕྲོས་ནས་སྣོད་ལ་ཕོག་པས་རིན་པོ་ཆེར་གྱུར། བཅུད་ལ་ཕོག་པས་སྐྱོན་དག་པར་བསམ། ཏཱྃ་ཞུ་ནས་ཐིམ་པས་བདུད་རྩིར་གྱུར། དེ་ནས་མགྲོན་སྤྱན་དྲངས་པ་ནི་གཞན་དང་མཐུན། ཏཱྃ་ལས་འོད་འཕྲོས་སེམས་ཅན་ཐམས་ཅད་མདུན་དུ་སྤྱན་དྲངས། ཡང་འོད་ཀྱིས་སྡིག་སྒྲིབ་སྦྱངས་ནས་སྒྲོལ་མར་གྱུར་བའི་ལྗགས་རྡོ་རྗེའི་སྦུ་གུས་གསོལ་བར་བསམས་ལ་གསོལ་བ་བཏབ་པ་ལ་སོགས་པ་བྱ་བར་གསུངས་སོ། །བུམ་ཆོག་བྱེད་པར་འདོད་ན། བུམ་པ་བཅའ་བ་གཞན་དང་མཐུན། རང་གི་མདུན་དུ་པདྨ་འདབ་བརྒྱད་ཀྱི་དབུས་སུ་བཞག མཆོད་པ་ལ་སོགས་པ་བཤམ། བྷྲཱུྃ་ལས་བུམ་པ་གཞལ་ཡས་ཁང་དུ་བསྐྱེད། ནང་དུ་པད་ཟླའི་སྟེང་དུ་ཆོ་ག་གསུམ་གྱིས་སྒྲོལ་མ་བསྒོམ། དེའི་ཐུགས་ཀར་སྔགས་ཀྱི་འོད་ཟེར་ཐིམ་པས་བ་སྤུའི་བུ་ག་ནས་བདུད་རྩིའི་རྒྱུན་བྱུང་བས་བུམ་པ་ཁེངས་པར་བསམ་ཞིང་བསྙེན་པ་བརྒྱ་སྟོང་ལ་སོགས་པར་བྱའོ། །དེ་ནས་བུམ་པའི་ལྷ་ལ་གསོལ་བ་གདབ། བསྟོད་པ་བཀྲ་ཤིས་ལ་སོགས་པ་བྱས་ལ། ལྷ་བདུད་རྩིར་ཞུ་བར་བསམས་ནས་རང་གཞན་ལ་ཁྲུས་བྱ།

ཡི་གེ་བཅུ་པའི་ཤམ་བུ་ལ་ཞི་རྒྱས་དབང་དྲག་སྐབས་དང་སྦྱར་ལ་བརྗོད། གཟུངས་ནར་མ་ལ་སོགས་པར་མར་དང་བདུད་རྩི་སྦགས་ལ་བྱ། སྲུང་འཁོར་བསྒོམ་པར་འདོད་ན་མི་དམིགས་པའི་ངང་ལས་འཁོར་ལོ་དཀར་པོ་རྩིབས་བརྒྱད་པ་ཡངས་ཤིང་རྒྱ་ཆེ་བ་ཞིག་བསམ། དེའི་དབུས་སུ་པདྨ་འདབ་མ་བཞི་པའི་དབུས་སུ་རང་རྫུས་སྐྱེས་ཀྱི་སྒྲོལ་མ་དཀར་མོ་གཞུང་བཞིན་དུ་བསྒོམ། གཞན་སྲུང་ན་ཡང་རང་ལ་བསྲུང་བྱས་བསྐོར་བར་བསྒོམ། ཕྱི་རོལ་དུ་འཁོར་ལོ་གཡས་བསྐོར་དུ་འཁོར་བ་ལ་དམིགས་ནས་བཟླས་པ་བྱ་གསུངས་སོ།། ༈ །སྒྲོལ་དཀར་གྱི་བྲིས་པའི་སྲུང་འཁོར་གྱི་ཡི་གེ་བཞུགས་སོ། །ན་མོ་གུ་རུ། རྗེ་བཙུན་སྒྲོལ་མ་དཀར་མོ་ཡི། །སྲུང་བའི་འཁོར་ལོའི་གདམས་ངག་ནི། །གྲོ་ག་ལ་སོགས་གང་རུང་ལ། །རྩིབས་བརྒྱད་རྡོ་རྗེའི་རྭ་བར་བཅས། །ཕྱི་རོལ་ཨུཏྤ་ལས་བསྐོར་བྲི། །ལྟེ་བར་ཨོཾ་དང་ཧཱ་ཡི་བར། །རཀྵ་རཀྵ་ཀུ་རུ་ཞེས། །གང་བསྲུང་མིང་ནི་སྤེལ་བར་བྱ། །རྩིབས་བརྒྱད་ཏཱ་རེ་ལ་སོགས་བརྒྱད། །བུམ་པའི་མཎྜལ་སྟེང་དུ་བཞག །དེ་ནས་རྣལ་འབྱོར་རང་ཉིད་ནི། །དང་པོ་བྱམས་པ་སྙིང་རྗེ་བསྒོམ། །དེ་ནས་རང་ཉིད་སྐད་ཅིག་གིས། །སྒྲོལ་མ་དཀར་མོར་གྱུར་པར་བསམ། །དེ་ནས་སྟོང་པ་ཉིད་ངང་ལས། །ཆུ་བོ་ཆེན་པོའི་སྟེང་དུ་ནི། །ཙཀྲ་དེ་ཉིད་བཞུགས་པར་བསྒོམ། །དཀོན་མཆོག་པཱུ་ཛ་ལ་སོགས་བཤམ། །སྐད་ཅིག་གིས་ནི་ཙཀྲ་ཡི། །ཡིག་འབྲུ་རྣམས་ལ་རྗེ་བཙུན་མ། །སྒྲོལ་མ་མངོན་དུ་གྱུར་པར་བསྒོམ། །བདག་དང་དེ་ལས་འོད་འཕྲོས་པས། །འོག་མིན་གནས་ནས་ཡེ་ཤེས་མ། །སྤྱན་དྲངས་བསྟིམ་ལ་མཆོད་བསྟོད་བྱ། །བཟླས་པ་ཁྲི་ཕྲག་ལྔའམ་ནི། །ཡང་ན་འབུམ་ཕྲག་གཅིག་ཏུ་བྱ། །བར་ཆད་ཞི་དང་ཚེ་རིང་དང༌། །གཞན་ཡང་གང་འོས་གསོལ་བ་གདབ། །དངོས་གྲུབ་ཞལ་བཞེས་གྲུབ་པ་དང༌། །ཇོ་མོ་དེ་རྣམས་ཡིག་འབྲུ་ཡི། །རང་བཞིན་ཉིད་དུ་བསམ་བྱས་ལ། །དེ་ནས་ལེགས་པར་བལྟབས་བྱས་ཤིང༌། །གང་བསྲུང་ལུས་ལ་བཏགས་པར་བྱ། །དེ་ལ་ཡོན་ཏན་དཔག་ཏུ་མེད། །།།།

༈ །སྒྲོལ་དཀར་གྱི་བར་ཆད་ཞི་བར་བྱེད་པའི་ཡི་གེ་བཞུགས་སོ། །བར་ཆད་ཞི་བར་བྱེད་པ་ལྟས་ངན་གཡང་དུ་ལེན་པའི་མན་ངག་ནི། རང་སྒྲོལ་མ་དཀར་མོར་བསམས་ལ། སྙིང་གའི་འོད་ཟེར་གྱིས་རང་ལ་གནོད་བྱེད་མདུན་དུ་བཀུག་ནས་སྒྲོལ་མ་དཀར་མོར་བསྒོམས་ལ། དེ་ལ་ཡེ་ཤེས་པ་སྤྱན་དྲངས་ནས་གཉིས་སུ་མེད་པར་བསྟིམ། རང་གི་ཐུགས་ཀ་ནས་སྤྲོས་པའི་མཆོད་པ་སྣ་ལྔ་དབུལ། གང་འདོད་པའི་གསོལ་བ་གདབ། བཛྲ་མུཿས་གཤེགས་སུ་གསོལ། དེ་ལྟར་བྱས་ན་གནོད་བྱེད་བར་ཆད་གང་གིས་ཀྱང་མི་ཚུགས་གསུངས་སོ། །ཆག་ཆེ་ན་འཁོར་ལོའི་གུར་ཁང་གི་གདམས་ངག་ནི། འཁོར་ལོ་རྩིབས་སྤྲད་གུར་ཁང་ཞྭ་ཐེབ་སྤྲད་པ་ལྟ་བུའི་ནང་གི་ལྟེ་བ་ལ། རང་ཉིད་སྒྲོལ་མར་བསྐྱེད། ལྟེ་བར་ཨོཾ་དང་ཧཱ་བསམ། རྩིབས་བརྒྱད་ལ་ཡི་གེ་བརྒྱད། དེ་ནས་ཡི་གེ་བཅུ་པ་བཟླའོ། །ཆག་ཆེ་བ་ཐམས་ཅད་བཟློག་གོ །གཞན་ཡང་འཇིགས་པ་བསྲུང་བ་ལ་སོགས་པ་ཐམས་ཅད་མིང་སྤེལ་བསྒྱུར་ལ་བྱ་གསུངས་སོ། །།།། ༈ །སྒྲོལ་མའི་བྱད་འགྲོལ་གྱི་ཡི་གེ་སློབ་དཔོན་པདྨ་ནག་པོས་མཛད་པ་བཞུགས་སོ། །ན་མསྟཱ་རཱ་ཡི། ངག་གི་བྱད་འགྲོལ་བ་ནི། རྗེ་བཙུན་མ་མདུན་གྱི་ནམ་མཁར་སྤྱན་དྲངས་ལ། མཆོད་པ་རྣམ་ལྔ་འབུལ། བདུན་རྣམ་དག་བྱས་ལ། བདག་གི་ངག་བཅིངས་པ་དང་མནན་པ་ལས་དགྲོལ་དུ་གསོལ། ཞེས་གསོལ་བ་གདབ་བོ། །དེ་ནས་རང་གི་སྙིང་གར་རྃ་ལས་མེའི་རང་བཞིན་ཨེ་དམར་པོ་གཅིག་བསམ། མགྲིན་པར་ན་ནག་པོ་གཅིག་བསམ། ལྕེ་རྩེར་སྟྭྃ་ཡིག་ནག་པོ་གཅིག་བསམ། སྤྱན་དྲངས་པའི་ཡི་དམ་ལྷའི་ཐུགས་ཀའི་ས་བོན་ལས་འོད་ཟེར་འཕྲོས་པས་རང་གི་སྙིང་གའི་རྃ་ཡིག་ལ་ཕོག་པས་དམར་ལམ་གྱིས་སོང་བས། ན་ནག་པོ་དང་སྟྭྃ་གཉིས་ཐུལ་གྱིས་ཚིག་པར་བསམ། དེ་ནས་ཡི་དམ་དེ་རང་ལ་ཐིམ་པར་བསམས་ལ། ཨོཾ་ཛི་ཧྭཱ་སཾ་རཀྵ་ནི་ཧཱུྃ། སྔགས་བརྒྱ་རྩ་བརྒྱད་བཟླའོ། །ཨོཾ་ཏཱ་རེ་ཏཱ་ར་ཡེ་བྷ་ག་ཝ་ཏཱི་ཛི་ཧྭཱ་བྷ་ག་ཝཱན་བྷནྡྷ་མོ་ཙ་ནི་སྭཱ་ཧཱ། བརྒྱ་རྩ་བརྒྱད་བཟླས་པས་ལྕེ་རྡོ་རྗེ་རྩེ་ལྔ་པ་དཀར་པོ་འབར་བ་མཐུ་དང་ལྡན་པར་བསམ་མོ། །ངག་གི་བྱད་འགྲོལ་བ་རྒྱ་གར་གྱི་མཁན་པོ་པདྨ་ནག་པོས་མཛད་པ།

སྐྱབས་འགྲོ་སེམས་བསྐྱེད་སྤྱི་འགྲོ་ལྟར་བྱས་ལ། རང་སྐད་ཅིག་གིས་རྗེ་བཙུན་མར་གསལ་བའི་སྙིང་གར་ཟླ་བའི་དཀྱིལ་འཁོར་གྱི་སྟེང་དུ་ཏཱྃ་ཡིག་དཀར་པོ་དེ་ལས་འོད་ཟེར་འཕྲོས་པས། རང་བཞིན་གྱི་གནས་ནས་བཅོམ་ལྡན་འདས་མ་ལ་བླ་མ་འོད་དཔག་མེད་ཀྱིས་དབུར་བརྒྱན་པ་མདུན་གྱི་ནམ་མཁར་ཨོཾ་བཛྲ་ས་མཱ་ཛཿ ཅེས་པས་སྤྱན་དྲང༌། ན་མོ་གུ་རུ་བྷྱཿ ནམ་ཨཱརྻ་ཏཱ་རེ་ས་པ་རི་ཝཱ་རེ་བྷྱཿ ཅེས་པས་ཕྱག་བྱ། ཨཱརྻ་ཏཱ་རེ་ས་པ་རི་ཝཱ་ར་ཨརྒྷཾ། པཱ་དྱཾ། པུཥྤེ། དྷཱུ་པེ། ཨཱ་ལོ་ཀེ། གནྡྷེ། ནཻ་ཝི་དྱེ། ཤབྡ་ཨཱཿཧཱུྃ། ཞེས་པས་མཆོད། དཀོན་མཆོག་གསུམ་ལ་བདག་སྐྱབས་མཆི། །སོགས་ལན་གསུམ་གྱིས་སྡོམ་པ་བཟུང༌། སེམས་ཅན་ཐམས་ཅད་བདེ་བ་དང་བདེ་བའི་རྒྱུ་དང་ལྡན་པར་གྱུར་ཅིག ཅེས་སོགས་ཚད་མེད་བཞི་བསྒོམ། ཨོཾ་སྭ་བྷཱ་ཝ་ཤུདྡྷཿསརྦ་དྷརྨཱཿསྭ་བྷཱ་ཝ་ཤུདྡྷོ྅ཧཾ། སྟོང་བར་གྱུར། སྟོང་པའི་ངང་ལས་པྃ་ལས་པདྨ་དཀར་པོ་འདབ་མ་བརྒྱད་པའི་ལྟེ་བར་ཨ་ལས་ཟླ་བའི་དཀྱིལ་འཁོར་གྱི་སྟེང་དུ། རང་གི་རྣམ་པར་ཤེས་པའི་ངོ་བོ་ཏཱྃ་ཡིག་དཀར་པོ་དེ་ལས་འོད་ཟེར་འཕྲོས། འཕགས་པ་མཆོད། སེམས་ཅན་གྱི་དོན་བྱས། ཚུར་འདུས་ཡོངས་སུ་གྱུར་པ་ལས། རང་ཉིད་རྗེ་བཙུན་མ་སྒྲོལ་མ་ཡིད་བཞིན་འཁོར་ལོ་སྐུ་མདོག་སྟོན་ཀའི་ཟླ་བ་ལྟར་དཀར་ཞིང༌། ནོར་བུ་ཆུ་ཤེལ་དྲི་མ་མེད་པ་ལྟར་དྭངས་ཤིང་འོད་འཕྲོ་བ། ཞལ་གཅིག་ཕྱག་གཉིས་སྤྱན་གསུམ་དང་ལྡན་པ་ལང་ཚོ་བཅུ་དྲུག་ལོན་པའི་ན་ཚོད་ཅན། ཕྱག་གཡས་མཆོག་སྦྱིན་གྱི་ཕྱག་རྒྱ་མཛད་པ། གཡོན་པས་ཨུཏྤལ་དཀར་པོའི་ཡུ་བ་མཐེབ་སྲིན་གྱིས་ཐུགས་ཀར་བཟུང་བའི་འདབ་མ་སྙན་གྱི་ཐད་ཀར་དུས་གསུམ་གྱིས་སངས་རྒྱས་མཚོན་པའི་བརྡར་རྩ་བ་གཅིག་ལ་རྩེ་མོ་གསུམ་དུ་གྱེས་པའི་དབུས་ཀྱི་མེ་ཏོག་ཁ་བྱེ་བ། གཡས་འབྲས་བུ། གཡོན་གང་བུའི་རྣམ་པར་ཡོད་པ། དབུ་སྐྲ་མཐོན་མཐིང་རལ་པའི་ཟུར་ཕུད་ལྟག་པར་བཅིངས་ཤིང༌། ལན་བུ་རིང་དུ་འཕྱངས་པ། རིན་པོ་ཆེའི་རྒྱན་སྣ་ཚོགས་ཀྱིས་བརྒྱན་པ།

དར་སྣ་ཚོགས་ཀྱི་སྟོད་གཡོགས་དང༌། དར་དམར་པོའི་ཤམ་ཐབས་ཅན། ཞབས་རྡོ་རྗེའི་སྐྱིལ་ཀྲུང་གིས་བཞུགས་པ། ཕྱག་ཞབས་ཀྱི་མཐིལ་རྣམས་ལ་སྤྱན་རེ་རེ་སྟེ་ཡེ་ཤེས་ཀྱི་སྤྱན་བདུན་མངའ་བ། ཟླ་བའི་རྒྱབ་ཡོལ་དང་འོད་ཀོར་ཅན། དེའི་སྤྱི་བོར་ཨོཾ། མགྲིན་པར་ཨཱཿ ཐུགས་ཀར་ཧཱུྃ། ཐུགས་ཀའི་ཧཱུྃ་ལས་འོད་ཟེར་འཕྲོས་པས། རང་བཞིན་གྱི་གནས་ནས་རྗེ་བཙུན་མ་སྒྲོལ་མ་དཀར་མོ་ལ་སངས་རྒྱས་དང་བྱང་ཆུབ་སེམས་དཔའ་ཚོགས་ཀྱིས་བསྐོར་བ་ཨོཾ་བཛྲ་ས་མཱ་ཛཿ ཞེས་པས་སྤྱན་དྲང༌། ཛཿཧཱུྃ་བཾ་ཧོཿ གཉིས་སུ་མེད་པར་ཐིམ། སླར་ཡང་ཐུགས་ཀའི་ཧཱུྃ་ལས་འོད་འཕྲོས། དབང་གི་ལྷ་དེ་བཞིན་གཤེགས་པ་རིགས་ལྔ་འཁོར་དང་བཅས་པ་ཨོཾ་བཛྲ་ས་མཱ་ཛཿ ཞེས་པས་སྤྱན་དྲང༌། དེ་བཞིན་གཤེགས་པ་ཐམས་ཅད་ཀྱིས་བདག་ལ་མངོན་པར་དབང་བསྐུར་དུ་གསོལ། ཞེས་གསོལ་བ་བཏབ་པས། ཨོཾ་སརྦ་ཏ་ཐཱ་ག་ཏ་ཨ་བྷི་ཥེ་ཀ་ཏ་ས་མ་ཡ་ཤྲཱི་ཧཱུྃ། ཞེས་གསུང་ཞིང་བུམ་པའི་ཆུས་སྤྱི་བོ་ནས་དབང་བསྐུར། སྐུ་གང༌། དྲི་མ་དག ཆུའི་ལྷག་མ་ཡར་ལུད་པ་ལས་འོད་དཔག་མེད་ཀྱིས་དབུར་བརྒྱན་པར་བསམ། ཨོཾ་ཨཱརྻ་ཏཱ་རེ་ས་པ་རི་ཝཱ་ཨརྒྷཾ་ནས་ཤབྡའི་བར་གྱིས་མཆོད། འཁོར་བ་ལས་སྒྲོལ་ཏཱ་རེ་མ། །ཏུཏྟཱ་རེ་ཡིས་འཇིགས་བརྒྱད་སྒྲོལ། །ཏུ་རེས་ན་བ་རྣམས་ལས་སྒྲོལ། །སྒྲོལ་མ་ཡུམ་ལ་ཕྱག་འཚལ་ལོ། །ཞེས་པས་བསྟོད། ལྷའི་སྐུ་སྣང་སྟོང་དབྱེར་མེད་ལ་སེམས་གཏད་དེ། ཇི་ཙམ་གནས་ཀྱི་བར་དུ་མཉམ་པར་བཞག དེ་ནས་བཟླས་པ་བྱ་བར་འདོད་པས། རང་གི་སྙིང་གར་ཟླ་བའི་རང་བཞིན་གྱི་འཁོར་ལོ་དཀར་པོ་རྩིབས་བརྒྱད་པ་མུ་ཁྱུད་ལྔ་དང་ལྡན་པའི་ལྟེ་བར་ཏཱྃ་ཡིག་དཀར་པོ། དེའི་མཐའ་སྐོར་དུ། མ་མ་ཨཱ་ཡུཿཔུཎྱེ་ཛྙཱ་ན་པུཥྚིཾ་ཀུ་རུ་ཞེས་པས་བསྐོར་བ། མདུན་དུ་ཨོཾ། རྒྱབ་ཏུ་ཧཱ། རྩིབས་བརྒྱད་ལ་མདུན་ནས་བརྩམས་ཏེ་གཡས་སྐོར་དུ། ཏཱ། རེ། ཏུཏ། ཏཱ། རེ། ཏུ། རེ། སྭཱ། ཞེས་པའི་ཡིག་འབྲུ་རེ་རེ་ཡོད་པ། དེའི་ཕྱི་རོལ་གྱི་མུ་ཁྱུད་ནང་མ་ལ།

ཨ་ཨཱ། ཨི་ཨཱི། ཨུ་ཨཱུ། རྀ་རཱྀ། ལྀ་ལཱྀ། ཨེ་ཨཻ། ཨོ་ཨཽ། ཨཾ་ཨཿ ཞེས་པ་གཡོན་སྐོར་དུ་ཡོད་པ། དེའི་ཕྱི་རོལ་མུ་ཁྱུད་ལ། ཀ་ཁ་ག་གྷ་ང༌། ཙ་ཚ་ཛ་ཛྷ་ཉ། ཊ་ཋ་ཌ་ཌྷ་ཎ། ཏ་ཐ་ད་དྷ་ན། པ་ཕ་བ་བྷ་མ། ཡ་ར་ལ་ཝ། ཤ་ཥ་ས་ཧ་ཀྵཿ། ཞེས་པ་གཡས་སྐོར་དུ་ཡོད་པ། དེའི་ཕྱི་རོལ་གྱི་མུ་ཁྱུད་ལ། ཡེ་དྷརྨཱཿཧེ་ཏུ་པྲ་བྷཱ་ཝ་ཧེ་ཏུནྟེ་ཥནྟ་ཐཱ་ག་ཏོ་ཧྱ་བ་དཏཱ་ཏེ་ཥཱཉྩ་ཡོ་ནི་རོ་དྷ་ཨེ་ཝཾ་བཱ་དཱི་མ་ཧཱ་ཤྲ་མ་ཎཿ། ཞེས་པ་གཡས་སྐོར་དུ་ཡོད་པ། དེའི་ཕྱི་རོལ་གྱི་མུ་ཁྱུད་ལ་རྡོ་རྗེ་སྔོན་པོའི་ཕྲེང་བས་བསྐོར་བ། དེའི་ཕྱི་རོལ་དུ་འོད་ཟེར་སྣ་ལྔའི་ཕྲེང་བ་གཡས་སྐོར་དུ་ཡོད་པར་བསམ། དེ་ནས་སྤྱི་བོའི་བླ་མ་འོད་དཔག་མེད་ཡོངས་སུ་གྱུར་པ་ལས། མགོན་པོ་ཚེ་དང་ཡེ་ཤེས་དཔག་ཏུ་མེད་པ་སྐུ་མདོག་དཀར་ལ་དམར་བའི་མདངས་ཆགས་པ། ཞལ་གཅིག་ཕྱག་གཉིས་ཀྱིས་བུམ་པ་འཆི་མེད་བདུད་རྩེའི་ཆུ་རྒྱུན་གྱིས་གང་བ་མཉམ་གཞག་གི་སྟེང་ན་འཛིན་པ། དར་དང་རིན་པོ་ཆེའི་རྒྱན་གྱིས་བརྒྱན་པ། ཞབས་རྡོ་རྗེའི་སྐྱིལ་མོ་ཀྲུང་གིས་བཞུགས་པ། དེའི་ཐུགས་ཀའི་ཧྲཱིཿཡིག་ལས་འོད་ཟེར་ཕྱོགས་བཅུར་འཕྲོས་པས་སྣོད་བཅུད་འཁོར་འདས་ཀྱིས་བསྡུས་པའི་ཚེ་དང་བསོད་ནམས་ཐམས་ཅད་འོད་ཟེར་གྱི་རྣམ་པར་བསྡུས་ཏེ་བུམ་པའི་ནང་དུ་སོང་བས་འཆི་མེད་བདུད་རྩིའི་ཆུ་རྒྱུན་དུ་གྱུར། དེ་ཉིད་གྱེན་དུ་འཕྱུར་ཞིང་ཁ་ནས་ལུད་པས་རང་གི་སྤྱི་བོ་ཚངས་པའི་བུ་ག་ནས་ཞུགས། ལུས་ཀྱི་ནང་ཐམས་ཅད་གང་བས། ནད་གདོན་སྡིག་སྒྲིབ་དུས་མ་ཡིན་པའི་འཆི་བའི་འཇིགས་པ་ཐམས་ཅད་དག འཆི་མེད་ཚེ་ཡི་དངོས་གྲུབ་ཐོབ་པར་བསམ། ཐུགས་ཀའི་འཁོར་ལོ་སྔགས་ཕྲེང་དང་བཅས་པ་ལས་ཀྱང་འཆི་མེད་བདུད་རྩིའི་ཟིལ་དངར་གྱི་རྒྱུན་དང༌། འོད་ཟེར་དཀར་པོ་སྔགས་ཀྱི་སྒྲ་སྒྲོགས་པའི་ཚོགས་དཔག་ཏུ་མེད་པ་འཕྲོས། རང་གི་ལུས་ཀྱི་སྤྱི་གཙུག་ནས་རྐང་མཐིལ་གྱི་བར་ཐམས་ཅད་མེར་གྱིས་ཁེངས་པས་ནད་གདོན་སྡིག་སྒྲིབ་དུས་མིན་འཆི་བའི་འཇིགས་པ་ཐམས་ཅད་བྱང་ཞིང་དག

འཆི་མེད་ཚེ་ཡི་དངོས་གྲུབ་ཐོབ། ལུས་ཀྱི་བ་སྤུའི་བུ་ག་ཐམས་ཅད་ཀྱང་བདུད་རྩིའི་ཟིལ་པ་ཕྲོམ་མེར་ཡོད་པར་བསམ། ཡང་སྙིང་གའི་སྔགས་ཕྲེང་ཏཱྃ་ཡིག་དང་བཅས་པ་ལས་འོད་ཟེར་དཀར་པོ་ཟླ་བ་སྟོང་དུས་གཅིག་ཏུ་ཤར་བ་ལྟ་བུ་འཕྲོས། ལུས་ཀྱི་ནང་ཐམས་ཅད་གང༌། ནད་གདོན་སྡིག་སྒྲིབ་ཐམས་ཅད་བྱང་ཞིང་དག བ་སྤུའི་བུ་ག་ནས་ཕྱིར་འཕྲོས། ལུས་ཀྱི་ཕྱི་རོལ་འདོམ་གང་ཙམ་དུ་འོད་དཀར་པོའི་གུར་ཁང་དུ་གྱུར། ཞི་བའི་ཕྲིན་ལས་ཐམས་ཅད་འགྲུབ་པར་བསམ། ཡང་སྙིང་གའི་སྔགས་ཕྲེང་ལས་འོད་ཟེར་སེར་པོ་གསེར་ཞུན་མ་ལྟ་བུ་འཕྲོས། ལུས་ཀྱི་ནང་ཐམས་ཅད་གང༌། ཚེ་དང་བསོད་ནམས་ཡེ་ཤེས་ཐམས་ཅད་རྒྱས་པར་བྱས། བ་སྤུའི་བུ་ག་ནས་ཕྱིར་འཕྲོས། འོད་དཀར་པོའི་གུར་ཁང་གི་ཕྱི་རོལ་འདོམ་གང་ཙམ་དུ་འོད་སེར་པོའི་གུར་ཁང་དུ་གྱུར། རྒྱས་པའི་ཕྲིན་ལས་ཐམས་ཅད་འགྲུབ་པར་བསམ། ཡང་སྙིང་གའི་སྔགས་ཕྲེང་ལས་འོད་ཟེར་དམར་པོ་བྱེ་རུའི་མདོག་ལྟ་བུ་འཕྲོས། ལུས་ཀྱི་ནང་ཐམས་ཅད་གང༌། བརྟན་གཡོ་སྣོད་བཅུད་ཐམས་ཅད་དབང་དུ་བསྡུས། བ་སྤུའི་བུ་ག་ནས་ཕྱིར་འཕྲོས། འོད་སེར་པོའི་གུར་ཁང་གི་ཕྱི་རོལ་འདོམ་གང་ཙམ་དུ་འོད་དམར་པོའི་གུར་ཁང་དུ་གྱུར། དབང་གི་ཕྲིན་ལས་ཐམས་ཅད་འགྲུབ་པར་བསམ། ཡང་སྙིང་གའི་སྔགས་ཕྲེང་ལས་འོད་ཟེར་མཐིང་ག་འཕྲོས། ལུས་ཀྱི་ནང་ཐམས་ཅད་གང༌། མི་མཐུན་པའི་ཕྱོགས་ཐམས་ཅད་ཚར་བཅད། བ་སྤུའི་བུ་ག་ནས་ཕྱིར་འཕྲོས། འོད་དམར་པོའི་གུར་ཁང་གི་ཕྱི་རོལ་འདོམ་གང་ཙམ་དུ་འོད་མཐིང་གའི་གུར་ཁང་དུ་གྱུར། དྲག་པོའི་ཕྲིན་ལས་ཐམས་ཅད་འགྲུབ་པར་བསམ། ཡང་སྙིང་གའི་སྔགས་ཕྲེང་ལས་འོད་ཟེར་ལྗང་གུ་མརྒད་ཞུན་མའི་མདོག་ལྟ་བུ་འཕྲོས། ལུས་ཀྱི་ནང་ཐམས་ཅད་གང༌། མཆོག་དང་ཐུན་མོང་གི་དངོས་གྲུབ་ཐམས་ཅད་གཅིག་ཆར་དུ་འགྲུབ་པར་བྱས། བ་སྤུའི་བུ་ག་ནས་ཕྱིར་འཕྲོས། འོད་མཐིང་གའི་གུར་ཁང་གི་ཕྱི་རོལ་འདོམ་གང་ཙམ་དུ་འོད་ལྗང་གུའི་གུར་ཁང་དུ་གྱུར།

ཕྲིན་ལས་ཐམས་ཅད་འགྲུབ་པར་བསམ། ཡང་སྙིང་གའི་སྔགས་ཕྲེང་ལས་འོད་ཟེར་མཆིན་ཁ་འཕྲོས། ལུས་ཀྱི་ནང་ཐམས་ཅད་གང༌། དངོས་གྲུབ་ཐམས་ཅད་བརྟན་པར་བྱས། བ་སྤུའི་བུ་ག་ནས་ཕྱིར་འོད་ལྗང་གུའི་གུར་ཁང་གི་ཕྱི་རོལ་འདོམ་གང་ཙམ་དུ་འོད་མཆིན་ཁའི་གུར་ཁང་དུ་གྱུར། བྱིན་རླབས་དང་ཕྲིན་ལས་ཐམས་ཅད་བརྟན་པར་བསམ། འོད་ཀྱི་གུར་ཁང་དྲུག་པོ་ཐམས་ཅད་དབྱིབས་སྒོ་ངའི་རྣམ་པ་ལྟ་བུ་དུམ་བུ་གཅིག་ཏུ་གྱུར་པ་སྲ་ཞིང་མཁྲེགས་པ་བསྐལ་པའི་རླུང་གིས་ཀྱང་གཞིག་པར་མི་ནུས་པ། བར་མཚམས་ཐམས་ཅད་ཨུཏྤལ་སྔོན་པོ་མེ་ཏོག་ཁ་བྱེ་མ་ཐག་པ་གསར་ཞིང་མཉེན་ལྕུག་དང་ལྡན་པས་བུན་ལོང་གིས་གཏམས་པར་བསམ། བྱེ་བྲག་ཏུ་བགེགས་ལ་སོགས་པའི་གནོད་པ་སྲུང་བར་འདོད་པས་འོད་དཀར་པོའི་གུར་ཁང་གི་ནང་རང་ཉིད་ཀྱི་ཕྱི་རོལ་ཏུ་འཁོར་ལོ་དཀར་པོ་རྩིབས་བརྒྱད་པ་སྦུབས་ཅན་ཞྭ་ཐེབ་ཁ་སྦྱར་བ་ལྟ་བུ་གཡས་སྐོར་དུ་འཁོར་བ་མེ་དཔུང་འབར་ཞིང་འཁྲུགས་པས་གདོན་བགེགས་ཐམས་ཅད་བྱ་སྒྲོ་མགར་གྱི་སོ་མལ་དུ་བབས་པ་ལྟར་གཞོབ་ཐུལ་ཐུལ་ཚིག་པར་བསམ། དེ་ལྟར་རང་ཉིད་རྗེ་བཙུན་མ་གསལ་བ་ནས་བཟུང༌། ཕྱིའི་འོད་ཟེར་མཆིན་ཁའི་གུར་ཁང་གི་བར་དུ་སེམས་འཛིན་ཅིང་སྣང་སྟོང་དབྱེར་མེད་ཀྱི་ངང་དུ་མཉམ་པར་འཇོག དེ་བཞིན་དུ་ཕྱི་ནས་རིམ་བཞིན་ཐུགས་ཀའི་ཏཱྃ་ཡིག་གི་བར་དུ་སེམས་འཛིན་ཅིང་སྣང་སྟོང་གི་ངང་དུ་མཉམ་པར་འཇོག དེས་སེམས་དུབ་ན་གོང་བཞིན་སྙིང་གའི་འཁོར་ལོ་སྔགས་ཕྲེང་དང་བཅས་པ་ལས་བདུད་རྩི་བབས་ཅིང་འོད་ཟེར་གྱི་སྤྲོ་བསྡུའི་ངང་ནས། ཨོཾ་ཏཱ་རེ་ཏུཏྟཱ་རེ་ཏུ་རེ་མ་མ་ཨཱ་ཡུར་པུཎྱ་ཛྙཱ་ན་པུཥྚིཾ་ཀུ་རུ་སྭཱ་ཧཱ། ཞེས་ཅི་ནུས་བཟླས་པའི་རྗེས་སུ། སྔར་ལྟར་མཆོད་བསྟོད་བྱ། ཡི་གེ་བརྒྱ་པ་ཚར་གསུམ་གྱིས་བྱིན་རླབས་བརྟན་ཞིང་ལྷག་ཆད་ཁ་བསྐང༌། དགེ་བ་འདི་ཡིས་མྱུར་དུ་བདག །ཡིད་བཞིན་འཁོར་ལོ་འགྲུབ་གྱུར་ནས། །ཞེས་སོགས་དང༌། ལོ་བརྒྱ་ཐུབ་ཅིང་སོགས་ཀྱི་སྨོན་ལམ་དང་བཀྲ་ཤིས་བརྗོད་དོ། །དེ་ལྟར་སྒྲོལ་དཀར་ཡིད་བཞིན་འཁོར་ལོ་ལ།

ཇོ་བོ་རྗེ་ནས་བརྒྱུད་པ། གཉན་ལོ་ཙཱ་བ་ནས་བརྒྱུད་པ། ཁ་ཆེའི་པཎྜི་ཏ་ཤཱཀྱ་ཤྲཱི་ནས་བརྒྱུད་པ་སྟེ། བོད་ཡུལ་དུ་བརྒྱུད་པའི་སྲོལ་མི་འདྲ་བ་བཞི་ཙམ་བྱོན་པའི་མན་ངག་གི་སྙིང་པོ་བསྡུས་པ། གཙོ་ཆེར་བ་རི་བའི་སློབ་བརྒྱུད་བླ་མ་གླེ་ནག་པས་དགེ་བའི་བཤེས་གཉེན་སྒྲེ་པ་ལ་གནང་བའི་ཉམས་ལེན་ངོ་མཚར་བའི་གཏམ་རྒྱུད་དང་བཅས་པ་འདི་ནི། ཚེ་སྒྲུབ་ཀྱི་མན་ངག་ལ་ཟབ་པའི་མཐར་ཐུག་པར་གྲགས་སོ། །ཞེས་པ་འདི་ནི་བསྟན་པའི་སྦྱིན་བདག་ཆེན་པོ་ཞལ་ངོ་རྣམས་ཀྱི་རྒྱུན་གྱི་ཉམས་བཞེས་སུ། རྒྱལ་ཁམས་མྱུལ་བའི་བནྡེ་བློ་གསལ་གྱིས་ཕུན་ཚོགས་རྣམ་རྒྱལ་རབ་བརྟན་རྩེའི་གཟིམས་ཆུང་བདུད་འདུལ་བདེ་ཆེན་གླིང་དུ་ལེགས་པར་སྦྱར་བ་མངྒ་ལཾ།། ཡི་གེ་པ་ནི་རྗེ་དེ་ཉིད་ཀྱི་སྐུ་དྲིན་ཁོ་ནས་འཚོ་བའི་བྱ་བྲལ་བ་མཁྱེན་བརྩེའི་དབང་པོས་བགྱིས་པའོ། །ཨི་དཾ་ཤཱ་ས་ནཉྩ་སརྦ་ས་ཏྭཱ་ཧི་ཏོ་བྷ་བནྟུ།། ༈ །སྒྲོལ་དཀར་ཡིད་བཞིན་འཁོར་ལོའི་རྗེས་གནང་གི་ཡི་གེ་བནྡེ་མཁྱེན་བརྩེས་བགྱིས་པ་བཞུགས་སོ། །སྭ་སྟི། བདག་བསྐྱེད། མདུན་བསྐྱེད། བུམ་བཟླས་རྣམས་གྲུབ་རྗེས། མཎྜལ། གང་གི་དྲིན་གྱིས་བདེ་ཆེན་ཉིད། །ནས། རྡོ་རྗེ་ཅན་ཞབས་པད་ལ་འདུད། །འཁོར་བ་ལས་སྒྲོལ་ཏཱ་རེ་མ། །ཏུཏྟཱ་རེ་ཡིས་འཇིགས་བརྒྱད་སྒྲོལ། །ཏུ་རེས་ན་བ་རྣམས་ལས་སྒྲོལ། །སྒྲོལ་མ་ཡུམ་ལ་ཕྱག་འཚལ་ལོ། །ལན་གསུམ་གྱིས་གསོལ་བ་གདབ། དཀོན་མཆོག་གསུམ་ལ་བདག་སྐྱབས་མཆི། །སོགས་གསུམ་གྱི་རྒྱུན་བཤགས། བླ་མ་རིན་པོ་ཆེས་བདག་གི་དུས་མ་ཡིན་པར་འཆི་བའི་འཇིགས་པ་ཐམས་ཅད་བསལ་ནས། རྗེ་བཙུན་བཅོམ་ལྡན་འདས་མ་སྒྲོལ་མ་དཀར་མོ་ཡིད་བཞིན་འཁོར་ལོ་ཚེ་སྦྱིན་མའི་རྗེས་སུ་གནང་བ་མ་ལུས་པ་སྩལ་དུ་གསོལ། ལན་གསུམ་གྱིས་ཁྱད་པར་གྱི་གསོལ་བ་བཏབ། སློབ་དཔོན་གྱི་ཐུགས་ཀ་ནས་འོད་འཕྲོས། མདུན་བསྐྱེད་ཀྱི་ཐུགས་ཀར་ཕོག དེའི་ཐུགས་ཀ་ནས་ཀྱང་འོད་འཕྲོས་སློབ་མའི་ལུས་ལ་ཕོག་པས་སྡིག་སྒྲིབ་ནད་གདོན་དུས་མ་ཡིན་པར་འཆི་བའི་འཇིགས་པ་ཐམས་ཅད་སྦྱངས།

དེར་མ་ཟད་རྣམ་སྨིན་གདོས་བཅས་ཤ་ཁྲག་གི་ལུས་སྟོང་པར་སྦྱངས། སྭ་བྷཱ་ཝ་སོགས་བརྗོད། སྟོང་པའི་ངང་ལས་པྃ་ལས་པདྨ་དཀར་པོ་འདབ་མ་བརྒྱད་པའི་ལྟེ་བར་ཨ་ལས་ཟླ་བའི་སྟེང་དུ་སློབ་མའི་རྣམ་ཤེས་ཏཱྃ་ཡིག་དཀར་པོ་ལས་མངོན་རྟོགས་ལྟར་སྒྲོལ་མའི་ཐུགས་ཀའི་འཁོར་ལོ་དང་བཅས་པ་བསྐྱེད། གནས་གསུམ་དུ་འབྲུ་གསུམ་བསམ། སློབ་དཔོན་མདུན་བསྐྱེད་དང་བཅས་པའི་ཐུགས་ཀའི་འོད་ཀྱིས་སངས་རྒྱས་བྱང་སེམས་ཐམས་ཅད་དང༌། ཁྱད་པར་རྗེ་བཙུན་མའི་སྐུ་གསུང་ཐུགས་ཀྱི་བྱིན་རླབས་ཐམས་ཅད་འཁོར་ལོ་དཀར་པོ་རྩིབས་བརྒྱད་པ་ལྟེ་བ་ཡིག་འབྲུ་དང་བཅས་པ་སྔགས་ཀྱི་རང་སྒྲ་སྒྲོག་ཅིང་འཆི་མེད་ཀྱི་བདུད་རྩིའི་ཆར་འབེབས་བཞིན་པའི་རྣམ་པས་ནམ་མཁའ་གང་བར་བྱོན་པ་དང༌། མདུན་བསྐྱེད་དང་སློབ་དཔོན་ལས་ཀྱང་དེ་འདྲ་བའི་འཁོར་ལོ་དཔག་ཏུ་མེད་པ་ཆད། སློབ་མའི་བ་སྤུའི་བུ་ག་ཚུན་ཆད་ནས་ཀྱང་ཐོགས་མེད་དུ་ཞུགས། ཐུགས་ཀའི་འཁོར་ལོ་ལ་ཐིམ་པས་དེ་ལས་ཀྱང་འཆི་མེད་བདུད་རྩིའི་རྒྱུན་དཔག་ཏུ་མེད་པ་སྐྱ་ཆི་ལི་ལི་བབས། ལུས་ཐམས་ཅད་འཆི་མེད་ཡེ་ཤེས་ཀྱི་བདུད་རྩིས་ཁེངས། དུས་མ་ཡིན་པའི་འཆི་བའི་འཇིགས་པ་ཐམས་ཅད་སངས་ཀྱིས་དག ཚེ་དང་བསོད་ནམས་ཐམས་ཅད་རྒྱས་པར་བསམས་ལ། ཡི་གེ་བཅུ་པའི་མཐར། ཨཱ་བེ་ཤ་ཡ་ཨ་ཨཱཿ ཛཿཧཱུྃ་བཾ་ཧོཿ ཞེས་ཅི་མང་བརྗོད་ཅིང་རྡོར་དྲིལ་གསོར་དཀྲོལ་བྱ། ཡང་རང་གི་སྙིང་ག་ནས་འོད་འཕྲོས། དབང་གི་ལྷ་སྒྲོལ་མ་འཇིགས་པ་བརྒྱད་ལས་སྐྱོབ་པའི་ལྷ་ཚོགས་ཨོཾ་བཛྲ་ས་མཱ་ཛཿ ཁྱེད་རྣམས་ཀྱིས་འདི་ལ་དབང་བསྐུར་དུ་གསོལ། ཇི་ལྟར་བལྟམས་པ་སོགས། ཨོཾ་སརྦ་ཏ་ཐཱ་ག་ཏ་ཨ་བྷི་ཥེ་ཀ་ཏེ་ས་མ་ཡ་ཤྲཱི་ཧཱུྃ་དང༌། ཡི་གེ་བཅུ་པ་བརྗོད་ཅིང་བུམ་པས་དབང་བསྐུར། སྐུ་གང་དྲི་མ་དག་ཆུ་ལྷག་མ་ཡར་ལུད་པས་འོད་དཔག་མེད་ཀྱིས་དབུར་བརྒྱན། ཛཿཧཱུྃ་བཾ་ཧོཿ དབང་གི་ལྷ་རྣམས་སློབ་མ་ལ་ཐིམ། ཕུན་སུམ་ཚོགས་པ་མངའ་བ་སོགས་ཤློ་ཀ་གསུམ་གྱིས་ཤིས་པ་བརྗོད།

མཆོད་པ་ལྔས་མཆོད་པ་ནི་ལྷ་བསྒོམ་པའི་རྗེས་གནང་ངོ༌། །སློབ་དཔོན་གྱི་ཐུགས་ཀའི་འཁོར་ལོ་ལས་སྔགས་ཕྲེང་ཆད། ཞལ་ནས་འཐོན། སློབ་མའི་ཞལ་ནས་ཞུགས། ཐུགས་ཀའི་ཡིག་འབྲུ་ལ་ཐིམ་པར་བསམས་ལ། ཡི་གེ་བཅུ་པ་ལ་སྤེལ་ཚིག་ཅན་བརྒྱ་རྩ་བརྒྱད་ཀྱི་རྗེས་ཟློས་བྱ། བཅོམ་ལྡན་བདག་ལ་སོགས་དང༌། བཅོམ་ལྡན་འདི་ལ་སོགས་བྱ་བ་ནི་སྔགས་བཟླ་བའི་རྗེས་གནང་ངོ༌། །དུས་འདི་ནས་བཟུང་སྟེ་བྱང་ཆུབ་མ་ཐོབ་བར་དུ་ཡི་དམ་དུ་བཟུང་སྙམ་པས། སློབ་དཔོན་དགོངས་སུ་གསོལ། བདག་མིང་འདི་ཞེས་བགྱི་བས་དུས་འདི་ནས་བཟུང་སྟེ་བྱང་ཆུབ་མ་ཐོབ་ཀྱི་བར་དུ་རྗེ་བཙུན་མ་ཡིད་བཞིན་འཁོར་ལོ་ཡི་དམ་གྱི་ལྷར་བཟུང་བར་བགྱིའོ། །ལན་གསུམ། བཅོམ་ལྡན་འདས་མའི་བྱིན་རླབས་ནི། །འདི་ལ་འགོ་བར་མཛད་དུ་གསོལ། །སྔགས་ཀྱི་དངོས་གྲུབ་སྩལ་བ་དང༌། །བརྟུལ་ཞུགས་བརྟན་པར་མཛད་དུ་གསོལ། །རྗེ་བཙུན་ཡིད་བཞིན་འཁོར་ལོའི་སྐུ་གསུང་ཐུགས་ཀྱི་བྱིན་རླབས་ཐམས་ཅད་རྡོ་རྗེའི་སློབ་མ་འདི་རྣམས་ཀྱི་རྒྱུད་ལ་འཇུག་ཅིང་བརྟན་པར་གྱུར་ཅིག ཛཿཧཱུྃ་བཾ་ཧོཿ ཞེས་མེ་ཏོག་གཏོར། མཇུག་དམ་ཚིག་བསྲུང་བར་ཁས་བླང༌། ཡོན་འབུལ་ཆོ་ག་སྤྱི་དང་འདྲ། འདིའི་བདག་མདུན་བུམ་གསུམ་བསྐྱེད་ཚུལ་ལ་ཁྱད་པར་མེད་ཅིང༌། མངོན་རྟོགས་པ་བླ་མས་མཛད་པའི་རྒྱས་པ་དེ་སོར་བཞག་དང༌། བསྡུས་པ་དེ་ཡིན་ན་མདུན་བུམ་གཉིས་ལ་གནས་གསུམ་བྱིན་རླབས། ཡེ་ཤེས་དགུག་གཞུག དབང་བསྐུར། འོད་དཔག་མེད་ཀྱིས་རྒྱས་བཏབ་པ་ངེས་པར་བྱ་དགོས། བདག་བསྐྱེད་ལའང་བྱས་ན་ལེགས། གཏོར་མ་ནི་མདུན་བསྐྱེད་ཀྱི་མཆོད་བསྟོད་གྲུབ་མཚམས། དཀར་ཆ་བཞི་ནས་འབྱུང་བ་ཇི་ལྟ་བས་འབུལ། གཞན་མཆོད་བསྟོད་ཐམས་ཅད་རྒྱས་བསྡུས་སྐབས་སུ་གང་བབས་ཀྱིས་འགྲུབ། བརྒྱ་ཚར་སོགས་མཛད་ན། ཐོག་མ་དེ་ལ་འདི་ཁ་སོར་བཞག གཉིས་པ་མན་ལ་མཎྜལ་ས་གཞི་སྤོས་བྱུགས། གསོལ་གདབ་བསྟོད་པ་གཉིས་ཚར་གཅིག་གི་རྗེས་སུ་ཐུན་མོང་མ་ཡིན་པའི་གསོལ་གདབ་སྦྱར།

རྒྱུན་བཤགས་མི་དགོས། བཟླས་ལུང་ལ་སྔགས་གསུམ་གསུམ་མཛད། གཞན་དཀྱུས་བཞིན་མཛད་པ་ཕྱག་ལེན་ཡིན་ནོ།། ༈ །སྒྲོལ་དཀར་རྗེས་གནང་ཐུན་མོང་མ་ཡིན་པའི་ཟིན་བྲིས་དགེ། སྭ་སྟི། གོང་གི་རྗེས་གནང་དེ་ཐུན་མོང་བ་ཡིན། ཐུན་མོང་མ་ཡིན་པ་ཁྲིད་ཀྱི་སྔོན་འགྲོ་སོགས་ཀྱི་ཚེ། སྤྱི་བཤད་ཇི་ལྟར་རིགས་པའི་རྗེས་འདིའི་ལོ་རྒྱུས་སྦྱར། མཎྜལ་ནས། དམ་ཚིག་པའི་འཁོར་ལོ་བསྐྱེད་པའི་བར་འདྲ། དེ་ལྟར་གསལ་བའི་ཕྱི་རོལ་དུ་འཁོར་ལོ་དཀར་པོ་སོགས་ནས་བུན་ལོང་གིས་གཏམས་པའི་མོས་པ་མཛོད། བར་གྱིས་འཁོར་ལོ་དང་འོད་ཟེར་དྲུག་སྦྲག་གི་དམིགས་པ་སྒྲུབ་ཐབས་ལྟར་ཕོག གནས་གསུམ་བྱིན་རླབས་ནས། མཆོད་པ་ལྔས་མཆོད་པའི་བར་འདྲ། སྐུའི་རྗེས་གནང་ངོ༌། །སློབ་དཔོན་གྱི་ཐུགས་ཀ་ནས་ཡིག་འབྲུ་ལ་ཐིམ་པའི་མོས་པ་མཛོད། སླར་ཡང་ཁྱེད་རང་རྣམས་ཀྱི་སྤྱི་བོའི་བླ་མ་འོད་དཔག་མེད་ཡོངས་སུ་གྱུར་པ་ལས་མགོན་པོ་ཚེ་དཔག་མེད་ནས། ཞབས་རྡོ་རྗེའི་སྐྱིལ་ཀྲུང་གིས་བཞུགས་པ། དེའི་ཐུགས་ཀའི་ཧྲཱིཿཡིག་ནས། ཟིལ་པ་ཕྲམ་མེར་ཡོད་པའི་མོས་པ་དང་བཅས་སྔགས་འདིའི་རྗེས་ཟློས་བརྒྱ་རྩ་བརྒྱད་མཛོད། ཅེས་ཚེ་བཅུད་སྡུད་པའི་དམིགས་པ་སྒྲུབ་ཐབས་ལྟར་ཕོག་ལ། རྩ་སྔགས་སྤེལ་ཚིག་བཅས་བརྒྱ་རྩ། དབྱངས་གསལ་རྟེན་སྙིང་རྣམས་གསུམ་རེ། མཐར་བཅོམ་ལྡན་བདག་ལ་དང༌། བཅོམ་ལྡན་འདི་ལ་དང༌། དམ་བཅའ་བའི་ལུང་དཀྱུས་ལྟར། གསུང་གི་རྗེས་གནང་ངོ༌། །བྱིན་གྱིས་བརླབ་པའི་ལུང་གི་སླད་དུ། ཐུགས་ཀའི་ཏཱྃ་ཡིག་ལ་སེམས་གཏད་པས། དེ་ལས་འོད་ཟེར་འཕྲོས། སྣོད་བཅུད་ཐམས་ཅད་ལ་ཁྱབ་པས། དེ་ཐམས་ཅད་འོད་དུ་ཞུ་ནས་འོད་མཆིན་ཁའི་གུར་ཁང་ལ་ཐིམ་ནས། སྟོང་པ་ཉིད་ཀྱི་ངང་དུ་གྱུར་པའི་མོས་པ་དང་བཅས་ཅུང་ཟད་རེ་མཉམ་པར་འཇོག་པར་ཞུ། ཐུགས་ཀྱི་རྗེས་གནང་སྟེ། སོ་སོའི་རྟོག་པ་བསྐྱེད་པ་སྤྱི་ལྟར་སྦྱར་རོ། །དེ་རྣམས་བརྟན་པར་བྱ་བའི་སླད་དུ། བཅོམ་ལྡན་འདས་མའི་ནས། ཛཿཧཱུྃ་བཾ་ཧོཿ

ཞེས་མེ་ཏོག་དོར་བའི་བར་དཀྱུས་བཞིན་ནོ། །ཞེས་པའང་རྗེ་བླ་མའི་གསན་ཐོར་སྙན་བརྒྱུད་ཞལ་ཤེས་ཟིན་བྲིས་སུ་གནང་བ་བཞིན་བྲིས་པའོ། །ཤུབྷཾ།། ༈ །སྒྲོལ་དཀར་ཡིད་བཞིན་འཁོར་ལོའི་ཟབ་ཁྲིད་ཀྱི་དམིགས་རིམ་ཟིན་བྲིས་བཞུགས་སོ། །ཨོཾ་སྭ་སྟི། སྐྱབས་ཡུལ་ཞིང་ཁམས་ཡོངས་སུ་དག་པའི་དབུས་སུ། ཡེ་ཤེས་ཀྱི་སྣང་བ་ལས་གྲུབ་པའི་དཔག་བསམ་ཤིང་གི་སྡོང་པོའི་སྟེང་པདྨ་འདབ་མ་བཞི་པའི་དབུས་སུ་རྩ་བའི་བླ་མ་འོད་དཔག་མེད་ཀྱི་རྣམ་པ་ཅན་ལ་བརྒྱུད་པའི་བླ་མས་བསྐོར་བ། མདུན་གྱི་འདབ་མ་ལ་སྒྲོལ་དཀར་སོགས་ཡི་དམ། གཡས་སུ་སངས་རྒྱས་ཐུབ་དབང་སོགས་སྤྲུལ་སྐུའི་རྣམ་པ། རྒྱབ་ཏུ་གསུང་རབ་པུསྟི་བརྩེགས་པ། གཡོན་དུ་ཉེ་བའི་སྲས་བརྒྱད་སོགས་དགེ་འདུན། གཞན་ཡང་སྐྱབས་ཡུལ་རྣམས་ཀྱིས་དཔག་བསམ་ཤིང་གི་སྡོང་པོ་ཡལ་ག ལོ་མ། མེ་ཏོག་གི་ཁྲོད་ཐམས་ཅད་གང་བ་ལ་དམིགས་ནས་བདག་གཞན་ཐམས་ཅད་ལུས་ངག་ཡིད་གསུམ་འཇུག་པ་གཅིག་ཏུ་དྲིལ་ནས་སྐྱབས་སུ་འགྲོ་བ་དང༌། སེམས་བསྐྱེད་པ་སྤྱི་ལྟར་བྱས་ནས། རང་སྐད་ཅིག་གིས་རྗེ་བཙུན་མར་གསལ་བའི་སྙིང་གར་ཟླ་བའི་དཀྱིལ་འཁོར་གྱི་སྟེང་དུ་ཞེས་པ་ནས། སྤྱི་བོར་བླ་མ་འོད་དཔག་མེད་ཡོངས་སུ་གྱུར་པ་ལས། མགོན་པོ་ཚེ་དང་ཡེ་ཤེས་དཔག་ཏུ་མེད་པ་སྐུ་མདོག་དཀར་ལ་དམར་བའི་མདངས་ཆགས་པ། ཞལ་གཅིག་ཕྱག་གཉིས་ཀྱིས་གསེར་གྱི་བུམ་པ་འཆི་མེད་བདུད་རྩིའི་ཆུ་རྒྱུན་གྱིས་གང་བ་མཉམ་གཞག་གི་སྟེང་ན་འཛིན་པ། དར་དང་རིན་པོ་ཆེའི་རྒྱན་གྱི་བརྒྱན་པ། ཞབས་རྡོ་རྗེའི་སྐྱིལ་ཀྲུང་གིས་བཞུགས་པ། ཞེས་པའི་བར་བརྗོད། དེ་ནས་རང་ཉིད་ཀྱི་ཕྱི་རོལ་དུ་འཁོར་ལོ་དཀར་པོ་རྩིབས་བརྒྱད་པ་སྦུབས་ཅན་ཞྭ་ཐེབ་ཁ་སྦྱར་བ་ལྟ་བུ་གཡས་སྐོར་དུ་འཁོར་བ་མེ་དཔུང་འབར་ཞིང་འཁྲུགས་པས་གདོན་བགེགས་ཐམས་ཅད་བྱ་སྒྲོ་མགར་གྱི་སོལ་མལ་དུ་བབས་པ་ལྟར་གཞོབ་ཐུལ་ཐུལ་ཚིག་པར་བསམ། དེ་ནས་སྙིང་གའི་ཏཱྃ་ཡིག་སྔགས་ཕྲེང་དང་བཅས་པ་ལས་འོད་ཟེར་དཀར་པོ་ཟླ་བ་སྟོང་དུས་གཅིག་ཏུ་ཤར་བ་ལྟ་བུ་འཕྲོས།

ཞེས་པ་ནས། བར་མཚམས་ཐམས་ཅད་ཨུཏྤ་ལ་སྔོན་པོ་མེ་ཏོག་ཁ་བྱེ་མ་ཐག་པ་གསར་ཞིང་མཉེན་ལྕུག་དང་ལྡན་པས་བུན་ལོང་གིས་གཏམས་པར་བསམ། ཞེས་པའི་བར་གྱི་འོད་ཟེར་དྲུག་སྦྲག་བསྒོམས་ལ། རང་ཉིད་རྗེ་བཙུན་མར་གསལ་བ་ནས་བཟུང༌། འོད་ཟེར་མཆིན་ཁའི་གུར་ཁང་གི་བར་དུ་སེམས་འཛིན་ཅིང་སྣང་སྟོང་དབྱེར་མེད་ཀྱི་ངང་དུ་མཉམ་པར་འཇོག དེ་བཞིན་དུ་ཕྱི་ནས་རིམ་བཞིན་ཐུགས་ཀའི་ཏཱྃ་ཡིག་གི་བར་དུ་སེམས་འཛིན་ཅིང་སྣང་སྟོང་དབྱེར་མེད་ཀྱི་ངང་དུ་མཉམ་པར་འཇོག དེ་ནས་སྤྱི་བོའི་ཚེ་དཔག་མེད་ཀྱི་ཐུགས་ཀའི་ཧྲཱིཿཡིག་ལས་འོད་ཟེར་ཕྱོགས་བཅུར་འཕྲོས་པས་སྣོད་བཅུད་འཁོར་འདས་ཀྱིས་བསྡུས་པའི་ཚེ་དང་བསོད་ནམས་ཐམས་ཅད་འོད་ཟེར་གྱི་རྣམ་པར་བསྡུས་ཏེ་བུམ་པའི་ནང་དུ་སོང་བས་འཆི་མེད་ཀྱི་བདུད་རྩིའི་ཆུ་རྒྱུན་དུ་གྱུར། དེ་ཉིད་གྱེན་དུ་འཕྱུར་ཞིང་ཁ་ནས་ལུད་པས་རང་གི་སྤྱི་བོ་ཚངས་པའི་བུ་ག་ནས་ཞུགས། ལུས་ཀྱི་ནང་ཐམས་ཅད་གང་བས་ནད་གདོན་སྡིག་སྒྲིབ་དུས་མ་ཡིན་པའི་འཆི་བའི་འཇིགས་པ་ཐམས་ཅད་དག་ཅིང་འཆི་མེད་ཚེའི་དངོས་གྲུབ་ཐོབ་པར་བསམ་ལ། རླུང་བུམ་པ་ཅན་ཉེར་གཅིག་གམ། བདུན་ནམ། གསུམ་གྱི་བར་དུ་འཛིན། དེ་ནས་ཐུགས་ཀའི་འཁོར་ལོ་སྔགས་ཕྲེང་དང་བཅས་པ་ལས་ཀྱང་འཆི་མེད་བདུད་རྩིའི་ཟིལ་དངར་གྱི་རྒྱུན་དང༌། འོད་ཟེར་དཀར་པོ་སྔགས་ཀྱི་སྒྲ་སྒྲོགས་པའི་ཚོགས་དཔག་ཏུ་མེད་པ་འཕྲོས། རང་གི་ལུས་ཀྱི་སྤྱི་གཙུག་ནས་རྐང་མཐིལ་གྱི་བར་ཐམས་ཅད་མེར་གྱིས་གང་བས། ནད་གདོན་སྡིག་སྒྲིབ་དུས་མ་ཡིན་པའི་འཆི་བའི་འཇིགས་པ་ཐམས་ཅད་བྱང་ཞིང་དག འཆི་མེད་ཚེའི་དངོས་གྲུབ་ཐོབ། ལུས་ཀྱི་བ་སྤུའི་བུ་ག་ཐམས་ཅད་ཀྱང་བདུད་རྩིའི་ཟིལ་པ་ཕྲམ་མེར་ཡོད་པར་བསམས་ལ། སྔགས་སྤེལ་ཚིག་ཅན་བཟླ། དམིགས་པ་ནམ་མི་གསལ་བའི་དུས་སྙིང་གའི་འཁོར་ལོ་སྔགས་ཕྲེང་དང་བཅས་པ་ལས་བདུད་རྩི་འབེབས་ཤིང༌། འོད་ཟེར་གྱི་སྤྲོ་བསྡུའི་ཚིག་གོང་བཞིན་བརྗོད་ལ།

སྔགས་དང་དམིགས་པ་སྤྲད་དེ་ཡེ་མ་ཡེངས་པར་བཟླས་པའི་རྗེས་སུ་སྔར་ལྟར་མཆོད་ཅིང་བསྟོད། ཨཱ་ལི་ཀཱ་ལི་ཡེ་དྷརྨཱ་ཡི་གེ་བརྒྱ་པ་རྣམས་ཚར་གསུམ་རེས་བྱིན་རླབས་བརྟན་ཞིང་ལྷག་ཆད་ཁ་བསྐང༌། དེ་ནས་ཐུགས་ཀའི་ཏཱྃ་ཡིག་ལས་འོད་ཟེར་འཕྲོས། སྣོད་བཅུད་རྣམས་ལ་ཕོག་པས་དེ་རྣམས་འོད་དུ་ཞུ་ནས་འོད་ཟེར་མཆིན་ཁའི་གུར་ཁང་ལ་ཐིམ། དེ་ནས་རིམ་བཞིན་བསྡུས་ནས་ཏཱྃ་ཡིག་གི་ནཱ་ད་ཡང་མི་དམིགས་ཏེ་འོད་གསལ་གྱི་ངང་ལ་མཉམ་པར་བཞག་གོ །དེ་ནས་ཆུ་ལས་ཆུ་བུར་གྱི་ཚུལ་གྱིས་རང་ཉིད་སྐད་ཅིག་གིས་རྗེ་བཙུན་མ་ཐུན་མཚམས་གྱི་ལྷར་གསལ། དགེ་བ་འདི་ཡིས་མྱུར་དུ་བདག །སོགས་དང༌། ལོ་བརྒྱ་ཐུབ་ཅིང༌། སོགས་ཀྱིས་སྨོན་ལམ་དང་བཀྲ་ཤིས་བརྗོད། སྣང་གྲགས་རིག་གསུམ་རྗེ་བཙུན་མའི་གསང་བ་གསུམ་གྱི་ངོ་བོར་ཤེས་པར་བྱས་ནས་སྤྱོད་ལམ་ལ་འཇུག་པར་བྱའོ། །འདི་ནི་རང་གི་གོ་ཡུལ་ཙམ་དུ་མ་བཞག་པར་རྗེ་བླ་མ་ཉིད་ཀྱི་སྤྱན་སྔར་ཕབ་སྟེ་དཔྱིས་ཕྱིན་པར་བྱས་སོ། །ཞེས་གྲུབ་མཆོག་ཚར་པ་ཆོས་རྗེའི་གསུང་ལས་རྙེད་པའི་གསང་བ་རྨད་དུ་བྱུང་བའོ། །སརྦ་མངྒ་ལཾ།། ༈ །རྗེ་བཙུན་ཡིད་བཞིན་འཁོར་ལོའི་སྦྱིན་སྲེག་གི་ཆོ་གའི་ངག་འདོན་བཀླགས་ཆོག་ཏུ་བཀོད་པ་ཚེ་དང་བསོད་ནམས་ཉི་མ་བདུན་འབར་མ་ཞེས་བྱ་བ་བཞུགས་སོ། །ཨོཾ་སྭ་སྟི་པྲ་ཛཱ་བྷྱཿ། རྗེ་བཙུན་མ་ཡིད་བཞིན་འཁོར་ལོ་ཚེ་སྦྱིན་མའི་སྦྱིན་སྲེག་གི་ཆོ་ག་ཚེ་དང་བསོད་ནམས་ཀྱི་ཉི་མ་བདུན་འབར་མ་ཞེས་བྱ་བ། མཐའ་ཡས་ཚེ་དང་ཡེ་ཤེས་རབ་འབར་བའི། །དཔལ་མོ་གང་ཞིག་ཞབས་ཟུང་དྲན་ཙམ་གྱིས། །གྲུབ་གཉིས་ཡིད་ཀྱི་ཤིང་རྟ་འགེངས་པའི་གཉེན། །སྒྲོལ་མའི་དྲིན་གྱིས་ཁྱོད་ལ་ཤིས་གྱུར་ཅིག །རབ་འབྱམས་རྒྱུད་དུ་རྒྱལ་བས་ཆེས་བསྔགས་པ། །རབ་འབྱམས་ཕྲིན་ལས་སྒྲུབ་བྱེད་ལྷ་མོ་དེའི། །རབ་འབྱམས་སྔགས་ཀྱི་ནུས་བ་འབྱིན་པའི་ཐབས། །རབ་འབྱམས་དགེ་ལེགས་སྟེར་བྱེད་འདིར་བཤད་བྱ། །དེ་ལ་རྗེ་བཙུན་མ་འདི་ཉིད་ཀྱི་རྗེས་སུ་གནང་བ་ཐོབ་ཅིང་བསྙེན་པ་སྔོན་དུ་སོང་བའི་རྣལ་འབྱོར་པས་བསྙེན་པའི་ཁ་སྐོང་བའམ།

རང་གཞན་གྱི་མི་མཐུན་པའི་ཕྱོགས་ཐམས་ཅད་ཞི་བའི་སླད་དུ་སྦྱིན་སྲེག་གི་ཆོ་ག་ཉམས་སུ་ལེན་པར་འདོད་པས། གནས་མཚན་ཉིད་དང་ལྡན་པར། རབ་བརྩིགས་པའམ་བྲུས་པ། བསྡུ་ན་བྲིས་པའི་ཐབ་བསྐྱེད་ཆོག་ནས་འབྱུང་བ་ལྟ་བུའི་དབུས་སུ་པདྨ་འདབ་བརྒྱད་ཀྱི་ལྟེ་བར་ཟླ་བ་ལ་ཏཱྃ་ཡིག་གམ། ཨུཏྤ་ལའམ་ཚོམ་བུ་དཀར་པོ་ཆེ་བ་ཞིག་བཀོད། སྲོག་ཆགས་མེད་པའི་ལྕི་བའམ། འོ་མ་ཅན་གྱི་ཤིང་ཁྲུ་ཕྱེད་ཙམ་དུ་ཟླུམ་པོར་བརྩིགས་པའི་ཤར་ལྷོར་ཟན་གྱི་ཏིང་ལོ་མེ་ཚང་དང་བཅས་པ་བཤམ། རང་གི་གཡས་སུ་སྟེགས་གཙང་མའི་སྟེང་དུ་རིམ་པ་བཞིན་ཡམ་ཤིང་སོར་བཅུ་གཉིས་ཀྱི་ཚད་

འཇམ་དཔལ་རྩ་རྒྱུད་ལས། མྱ་ངན་མེད་ཅིང་ཞི་དོན་དུ། །རློན་པ་ཉིད་ནི་རབ་ཏུ་བསྔགས། །ཞེས་དང༌། རྒྱ་གཞུང་ཕལ་ཆེར་དུ་ཞི་རྒྱས་ལ་རློན་པ་ཁོ་ན་གསུངས་པས་རྡོ་རྗེ་འཆང་སྒོ་རུམ་པའི་ཞལ་སྔ་ནས་རློན་པ་འབའ་ཞིག་མཛད། རྡོ་རྗེ་རིན་ཆེན་ནས་རློན་པ་བསྲེག་ཚེ་མེ་པྲ་ལེགས་པས་དགའ་གསུངས། སྐམ་པོ་མཛད་པ་མཁན་ཆེན་ཐམས་ཅད་མཁྱེན་པས་སྐམ་པོ་མཛད་པས་གང་བདེ་བྱའོ ། །མཐེ་བོང་ལས་མི་སྦོམ་ཞིང༌། མཐེའུ་ཆུང་ལས་མི་ཕྲ་བ་རྩ་རྩེ་མར་དང་བུ་རམ་གྱིས་བྱུགས་པ། མར། ཏིལ། ཟས་མཆོག འབྲས། སོ་བ། ནས། སྲན་མ། གྲོ། དཱུརྦ། ཀུ་ཤ་རྣམས་བཤམ། བསྲེགས་རྫས་འདི་རྣམས་རྒྱས་པའི་དབང་དུ་བྱས་ཀྱི་མ་འབྱོར་ན་ཡམ་ཤིང༌། མར། ཏིལ་གསུམ་མེད་ཐབས་མེད་གཞན་གང་འབྱོར་གྱིས་ཆོག་པར་བཤད། གཡོན་དུ་ཆུ་གཉིས་སྔོན་དུ་འགྲོ་བའི་ཉེར་སྤྱོད་ཚར་གཉིས་འཇིག་རྟེན་དང་འཇིག་རྟེན་ལས་འདས་པའི་གཏོར་མ་གཉིས་དང་གཏོར་མ་ཆ་གསུམ་རྣམས་ལག་ལེན་ལྟར་བཤམ། མདུན་དུ་རྡོ་རྗེ་དྲིལ་བུའི་བར་དུ་དགང་གཟར་བླུགས་གཟར་ཁ་སྦྱར་བ། བསང་ཆུའི་དུང་ཆོས་འཐོར་ཡབ་དང་བཅས་པ་བཀོད། མེ་ཡོལ་གངས་ཏི་སེའི་མཚོན་བྱེད་སོ་ཕག་དཀར་པོ་ལ་ཝྃ་ཡིག་བྲིས་པ་དང༌། ཐབ་ཀྱི་མཐར་དཀྲམ་རྒྱུའི་ཀུ་ཤ་ཆུན་པོ་བརྒྱད། མེ་སྦར་བའི་ལྕེ་མདེ། མེ་གསོ་བའི་བུད་ཤིང་སྐམ་པོ་སོགས་ཉེར་མཁོ་མ་ཚང་བ་མེད་པར་འདུ་བྱས་ཏེ། སྔ་དྲོ་དང་པོའམ། སྲོད་ཀྱི་ཐུན་ལ་བབས་པ་ན། སྟན་བདེ་བ་ལ་ཁ་ཤར་དུ་བལྟས་ཏེ། སྐྱིལ་ཀྲུང་ངམ་ཙོག་བུར་འདུག་ལ། ཐོག་མར་བསང་ཆུ་ལ་དམིགས་ཏེ། ཨོཾ་བཛྲ་ཡཀྵ་ཧཱུྃ། གསུམ་དང༌། རྡོ་རྗེ་གནོད་སྦྱིན་གྱི་ཕྱག་རྒྱས་བགེགས་བསྐྲད། ཨོཾ་བཛྲ་ཡཀྵ་ཧཱུྃ། བདུན་ནམ་ཉེར་གཅིག་དང་ཧཱུྃ་རིང་པོ་བརྗོད་པས་བྱིན་གྱིས་བརླབས་དེ་ནས་གཏོར་ཞིང༌། ཡཀྵས་བསང༌། སྭ་བྷཱ་ཝས་སྦྱང༌། སྟོང་པའི་ངང་ལས་ཨོཾ་ལས་རིན་པོ་ཆེ་ལས་གྲུབ་པའི་སྣོད་ཡངས་ཤིང་རྒྱ་ཆེ་བ་རྣམས་ཀྱི་ནང་དུ་ཨོཾ་ཞུ་བ་ལས་བྱུང་བའི་མཆོད་ཡོན།

ཞབས་བསིལ། མེ་ཏོག བདུག་སྤོས། སྣང་གསལ། དྲི་ཆབ། ཞལ་ཟས། གཞན་ཡང་རོལ་མོ་དང་འཕན་གདུགས་ལ་སོགས་པ་མཆོད་རྫས་བསམ་གྱིས་མི་ཁྱབ་པ་དྭངས་ཤིང་ཐོགས་པ་མེད་པ་ནམ་མཁའི་ཁམས་ཐམས་ཅད་གང་བ་འདི་ཉིད་བཅོམ་ལྡན་འདས་མ་ཡིད་བཞིན་འཁོར་ལོའི་ལྷ་ཚོགས་སངས་རྒྱས་དང་བྱང་ཆུབ་སེམས་དཔའ་ཐམས་ཅད་ཀྱི་སྤྱན་ལམ་དུ་འབྱུང་ཞིང་རྒྱས་པར་གྱུར་ཅིག སོ་སོའི་ཕྱག་རྒྱ་བཅས་ལ། ཨོཾ་ཨརྒྷ་སྭཱ་ཧཱ། པཱ་དྱཾ་སྭཱ་ཧཱ། ཨོཾ་བཛྲ་པུཥྤེ་ཨཱཿཧཱུྃ། ནས་ཤབྡ་ཨཱཿཧཱུྃ། རོལ་ཆེན་བྱ། ན་མཿསརྦ་བུདྡྷ་བོ་དྷི་ས་ཏྭཱ་ནཱཾ། ཨོཾ་སརྦ་བིཏ་པཱུ་ར་པཱུ་ར་སུ་ར་སུ་ར་ཨཱ་ཝརྟ་ཡ་ཨཱ་ཝརྟ་ཡ་ཧོཿ བཛྲ་སྥ་ར་ཎ་ཁཾ། ཞེས་ལན་གསུམ་བརྗོད། དེ་ནས་སྐྱབས་འགྲོ་ནས་བཟུང་སྟེ་བཟླས་པ་བསྡུ་བའི་བར་བདག་བསྐྱེད་སྔོན་དུ་བཏང་ལ་དཔལ་འབར་སྦར་ལ་ལག་པ་ཐལ་མོ་སྦྱར་བའི་བར་དུ་བཟུང་སྟེ་ཐབ་ལ་གཡས་སུ་བསྐོར་ཞིང༌། ཨོཾ་ཏཱ་རེ་ཏུཏྟཱ་རེ་ཏུ་རེ་སྭཱ་ཧཱ། ལན་གསུམ། ཨོཾ་བཛྲ་ཛྭ་ལ་རཾ། ཞེས་པས་མེ་སྦར། སྦྱིན་སྲེག་མང་པོ་བྱེད་ན་ཏིང་ལོ་ཙམ་ཀ་སྦར་ལ། འཇིག་རྟེན་པ་རྫོགས་མཚམས་ལྟ་བུ་གང་འོས་ཐམས་ཅད་སྦར་ཞིང་མེ་པྲ་ནི་འདི་དང་གོང་འོག་གཉིས་ཀར་བརྟག ཨོཾ་བཛྲ་ཛྭ་ལ་རཾ། ཞེས་བརྗོད་ཅིང་དགང་བླུགས་རྐྱང་པ་ལན་གསུམ་གྱིས་མེ་གསོ། བསང་ཆུ་འཐོར་ཞིང་རྡོ་རྗེ་གནོད་སྦྱིན་གྱི་ཕྱག་རྒྱ་དང༌། ཨོཾ་བཛྲ་ཡཀྵ་ཧཱུྃ། ལན་གསུམ་གྱིས་ཐབ་དང་ཡོ་བྱད་བསང༌། ཨོཾ་ཨོཾ་སྭཱ་ཧཱ། ཨོཾ་ཨཱཿསྭཱ་ཧཱ། ཨོཾ་ཀུ་རུ་ཀུ་རུ་སྭཱ་ཧཱ། ཐབ་དང་ཡོ་བྱད་ཐམས་ཅད་སྟོང་པར་གྱུར། སྟོང་པའི་ངང་ལས་པྃ་ལས་པདྨ་རྃ་ལས་ཉི་མའི་སྟེང་དུ་ཧཱུྃ་ལས་ཐབ་ཁུང་དཀར་པོ་དྭངས་ཤིང་ཐོགས་པ་མེད་པ་རྒྱར་ཁྲུ་གང༌། ཟབས་སུ་ཁྲུ་ཕྱེད་པ་ཁ་ཁྱེར་དང་མུ་རན་ལ་རྡོ་རྗེའི་ཕྲེང་བས་བསྐོར་བ། ཕྱི་རོལ་གྱི་གྲྭ་བཞིར་ཟླ་ཕྱེད་རྡོ་རྗེས་མཚན་པ་དབྱིབས་ཟླུམ་པོ་མཚན་ཉིད་ཕུན་སུམ་ཚོགས་པ་དང་ལྡན་པར་གྱུར། ཨོཾ་སྭ་བྷཱ་ཝ་སོགས། མཆོད་རྫས་དང་བསྲེག་རྫས་ཐམས་ཅད་སྟོང་པའི་ངང་ལས་རིན་པོ་ཆེའི་སྣོད་ཡངས་ཤིང་རྒྱ་ཆེ་བའི་ནང་དུ།

ལྷ་རྫས་ལས་གྲུབ་པའི་མཆོད་རྫས་དྭངས་ཤིང་ཐོགས་པ་མེད་པ་ས་དང་ནམ་མཁའ་བར་སྣང་ཐམས་ཅད་ཁྱབ་པ་ཀུན་ཏུ་བཟང་པོའི་རྣམ་འཕྲུལ་ལས་བྱུང་བའི་མཆོད་པའི་སྤྲིན་ཕུང་ལྟ་བུ་བསམ་གྱིས་མི་ཁྱབ་པར་གྱུར། སོ་སོའི་ཕྱག་རྒྱ་དང༌། ཨོཾ་ཨརྒྷཾ་སྭཱ་ཧཱ། ནས། ཤབྡ་ཨཱཿཧཱུྃ། མཆོད་ཡོན་གྱི་ཕྱག་རྒྱ་བྱས་ལ། ཨོཾ་བོ་དྷི་བྲྀཀྵ་ཡེ་སྭཱ་ཧཱ། ཨོཾ་ཨགྣ་ཡེ་སྭཱ་ཧཱ། ཨོཾ་སརྦ་པཱ་པཾ་ད་ཧ་ན་བཛྲ་ཡ་སྭཱ་ཧཱ། ཨོཾ་སརྦ་སཾ་པ་དེ་སྭཱ་ཧཱ། ཨོཾ་བཛྲ་པུཥྚ་ཡེ་སྭཱ་ཧཱ། ཨོཾ་བཛྲ་བཱི་ཛཱ་ཡཻ་སྭཱ་ཧཱ། ཨོཾ་མ་ཧཱ་བེ་གཱ་ཡ་སྭཱ་ཧཱ། ཨོཾ་མ་ཧཱ་བ་ལཱ་ཡ་སྭཱ་ཧཱ། ཨོཾ་བཛྲ་གྷསྨ་རཱ་ཡ་སྭཱ་ཧཱ། ཨོཾ་བཛྲ་ཨཱ་ཡུ་ཥེ་སྭཱ་ཧཱ། ཨོཾ་ཨ་པྲ་ཏི་ཧ་ཏ་བཛྲ་ཡ་སྭཱ་ཧཱ། ཞེས་དང༌། དྷརྨཱ་ར་ཎི་ཏ། དང་རོལ་ཆེན་བྱ། ཡང་ཡཀྵས་བསང༌། སྭ་བྷཱ་ཝས་སྦྱང༌། སྟོང་པའི་ངང་ལས་ཐབ་ཀྱི་ནང་དུ་པྃ་ལས་པདྨ་རྃ་ལས་ཉི་མའི་སྟེང་དུ་རྃ་ལས་བགྲང་ཕྲེང་རྃ་གྱིས་མཚན་པ་ལས་འོད་འཕྲོས་འདུས་གཅིག་ཏུ་གྱུར་པ་ལས་མེ་ལྷ་སྐུ་མདོག་དཀར་པོ་འོད་ཟེར་དཀར་པོ་འཕྲོ་ཞིང་འདུ་བ་ར་སྐྱེས་ལ་ཞོན་པ་ཞལ་གཅིག་ཕྱག་གཉིས་ཀྱི་ཕྱག་གཡས་བགྲང་ཕྲེང་དང་གཡོན་སྤྱི་བླུགས་འཛིན་པ་གསུས་ཁྱིམ་ཆེ་བ། རལ་པའི་ཐོར་ཚུགས་ཅན་ཨག་ཚོམ་འཕྱང་བ་དྲང་སྲོང་གི་ཚོགས་ཀྱིས་བསྐོར་བའི་དཔྲལ་བར་ཨོཾ། མགྲིན་པར་ཨཱཿ ཐུགས་ཀར་ཧཱུྃ། རང་གི་ཐུགས་ཀའི་ས་བོན་ལས་འོད་འཕྲོས་ཤར་ལྷོ་ནས་མེ་ལྷ་འཁོར་དང་བཅས་པ་བཛྲ་ས་མཱ་ཛཿ དབྱུ་གུ་རྩེ་གསུམ་པའི་ཕྱག་རྒྱ་རྣ་དྲུང་དུ་བཅས་ལ། ཨོཾ་ཚུར་བྱོན་ཚུར་བྱོན་འབྱུང་པོ་ཆེ། །མེ་ལྷའི་རྒྱལ་པོ་དྲང་སྲོང་མཆོག །སྲེག་བླུགས་ཞལ་ཟས་བཞེས་སླད་དུ། །འདི་ཉིད་དུ་ནི་བཞུགས་སུ་གསོལ། །གཟི་བརྗིད་ཆེན་པོ་མེ་ཡི་ལྷ། །འདོད་པ་ཐམས་ཅད་སྒྲུབ་པ་པོ། །ཐུགས་རྗེས་སེམས་ཅན་དོན་མཛད་པ། །འདི་ཉིད་དུ་ནི་བཞུགས་སུ་གསོལ། །ཨོཾ་ཨགྣ་ཡེ་མ་ཧཱ་ཏེ་ཛཿསརྦ་ཀརྨ་པྲ་སཱ་དྷ་ཀ མ་ཧཱ་བྷཱུ་ཏ་དེ་བ་རྀ་ཥི་དྭི་ཛ་ས་ཏྭཱཾ།

གྲྀ་ཧཱི་ཏྭཱ་ཨཱ་ཧུ་ཏིཾ་ཨཱ་ཧ་ར། ཨ་སྨིན྄་སནྣི་ཧི་ཏོ་བྷ་ཝ། ཞེས་པས་སྤྱན་དྲང༌། པདྨ་ཀ་མ་ལ་ཡ་སྟྭཾ། གྱིས་གདན་ལ་བཞུགས་པར་བསམ། ཨོཾ་ཨགྣ་ཡེ་ས་པ་རི་ཝཱ་ར་ཨརྒྷཾ་པྲ་ཏཱིཙྪ་སྭཱ་ཧཱ། ཨོཾ་ཨགྣ་ཡེ་ས་པ་རི་ཝཱ་ར་པཱ་དྱཾ་པྲ་ཏཱིཙྪ་སྭཱ་ཧཱ། ཞེས་པས་མཆོད་ཡོན་དང་ཞབས་བསིལ་ཕུལ་ལ། རྡོ་རྗེ་རྩེ་ལྔ་པའི་ཕྱག་རྒྱ་བཅས་ཏེ། ཨོཾ་བཛྲ་ཨཱ་ན་ལ་མ་ཧཱ་བྷཱུ་ཏ་ཛྭ་ལ་ཡ་སརྦ་བྷསྨ་ཀུ་རུ་སརྦ་དུཥྚཱཾ་ཧཱུྃ་ཕཊ་དྲྀ་ཤྱ་ཛཿཧཱུྃ་བཾ་ཧོཿ། ས་མ་ཡ་སྟྭཾ་གྱིས་དམ་ལ་བཞག ཨོཾ་བཛྲ་ཨཾ་ཀུ་ཤ་ཛཿ ཨོཾ་བཛྲ་པཱ་ཤ་ཧཱུྃ། ཨོཾ་བཛྲ་སྥོ་ཊ་བཾ། ཨོཾ་བཛྲ་ཨཱ་ཝེ་ཤ་ཡ་ཧོ། ཞེས་ཕྱག་རྒྱ་དང་བཅས་པས་དགུག་གཞུག་བཅིང་མཉེས་དབང་དུ་བྱས། སླར་ཡང་འོད་འཕྲོས་དབང་གི་ལྷ་རིགས་ལྔ་འཁོར་དང་བཅས་པ་བཛྲ་ས་མཱ་ཛཿ ཁྱེད་རྣམས་ཀྱིས་འདི་ལ་དབང་བསྐུར་དུ་གསོལ། ཨོཾ་སརྦ་ཏ་ཐཱ་ག་ཏ་ཨ་བྷི་ཥེ་ཀ་ཏ་ས་མ་ཡ་ཤྲཱི་ཡེ་ཧཱུྃ། ཞེས་གསུང་ཞིང་དབང་བསྐུར། སྐུ་གང༌། དྲི་མ་དག ཆུ་ལྷག་མ་ཡར་ལུད་པས་རྣམ་སྣང་གིས་དབུར་བརྒྱན། ཨོཾ་བཛྲ་ཡཀྵ་ཧཱུྃ། ཨོཾ་སྭ་བྷཱ་ཝ་སོགས་སྟོང་པའི་ངང་ལས་ཀུ་ཤ་སྔོ་ལྗང་རྩ་བ་དང་རྩེ་མོ་མ་ཉམས་པ་ལ་སོགས་པའི་མཚན་ཉིད་ཕུན་སུམ་ཚོགས་པ་དང་ལྡན་པ་བདག་ཅག་གི་མི་མཐུན་པའི་ཕྱོགས་ཐམས་ཅད་ཞི་བར་བྱེད་པ་ལ་ཐོགས་པ་མེད་པའི་ནུས་ལྡན་དུ་གྱུར། ཨོཾ་ཀུ་ཤ་འདི་ནི་གཙང་ཞིང་དགེ །ས་ལས་སྐྱེས་པའི་སྙིང་པོ་མཆོག །དཀོན་མཆོག་གསུམ་གྱི་དག་བྱེད་དེ། །ཚངས་པ་ཡི་ནི་དག་བྱེད་ཡིན། །འདིས་བདག་ཅག་གི་བགེགས་ཞི་ཞིང་མི་མཐུན་པའི་ཕྱོགས་ཐམས་ཅད་ཞི་བར་གྱུར་ཅིག ཅེས་བརྗོད་ལ་ཀུ་ཤ་ཆུན་པོ་བརྒྱད། ཨོཾ་བཛྲ་ས་ཏྭ་ཨཱཿཞེས་པས་རྩེ་མོ་གཡས་སུ་བསྟེན་ཏེ་ཐབ་ཀྱི་ཕྱོགས་མཚམས་བརྒྱད་དུ་བསྐོར་ཞིང༌། ཡཀྵ་དང་ཆུས་བསང༌། མེ་ལྷའི་ལྕེ་ཧཱུྃ་ལས་རྡོ་རྗེ་རྩེ་ལྔ་པར་གྱུར་པའི་ཁ་དང༌། གཟར་བུ་གཉིས་ཀྱི་ཁ་ལ་རྃ་བཀོད། ཨོཾ་ཨགྣ་ཡེ་སྭཱ་ཧཱ། ལན་བདུན་གྱིས་བཀང་སྟེ་གཟར་བུ་གཉིས་ཁ་སྦྱར་ལ་ཐབ་ལ་གཡས་སུ་ཟླུམ་པོར་བསྐོར་ཞིང༌།

ཨོཾ་ཨགྣ་ཡེ་ཨཱ་དཱི་བྱ་ཨཱ་དཱི་བྱ་བི་ཤུདྡྷེ་མ་ཧཱ་ཤྲི་ཡེ་ཧ་བྱ་ཀ་བྱ་བཧྣ་ཡ་སྭཱ་ཧཱ། བདག་ཅག་གི་མི་མཐུན་པའི་ཕྱོགས་ཐམས་ཅད་ཞི་བར་མཛོད་ཅིག་ཤཱནྟིཾ་ཀུ་རུ་སྭཱ་ཧཱ། ཞེས་ལན་གསུམ་གྱིས་བྱོན་ཟན་འབུལ། ཨོཾ་ཨགྣ་ཡེ་ས་པ་རི་ཝཱ་ར། ཐམས་ཅད་ཀྱི་མགོར་སྦྱར། ཨོཾ་ཨརྒྷཾ་པྲ་ཏཱིཙྪ་སྭཱ་ཧཱ། ནས། ཤབྡ་པྲ་ཏཱིཙྪ་སྭཱ་ཧཱ། རོལ་ཆེན་དང་བཅས་པས་མཆོད། རྡོ་རྗེ་ཕྱག་འཚལ་པའི་སྟངས་ཀས་བཟུང༌། ཚིགས་བཅད་ཀྱི་མཚམས་སུ་དྲིལ་བུ་གསིལ་ཞིང༌། འཇིག་རྟེན་དབང་ཕྱུག་ཚངས་དབང་མགོན་གྱི་སྲས། །མེ་ལྷའི་རྒྱལ་པོ་ཌཱ་ཀིས་དབང་བསྐུར་བ། །ཤེས་རབ་མཆོག་གིས་ཉོན་མོངས་ཐམས་ཅད་བསྲེགས། །འཛིན་བྱེད་མེ་ལྷ་ཉིད་ལ་རྣམ་པར་འདུད། །ཅེས་པས་བསྟོད། ཡམ་ཤིང་ནས་བཟུང་ཀུ་ཤའི་བར་ལྷ་སྔགས་འབའ་ཞིག་གིས་འབུལ་བའམ། སྤྲོ་ན་ཐོག་མར་ལྷ་རྫས་ལས་སྔགས་གསུམ་སྤེལ་བ་ལན་གཅིག་དང༌། བར་སྐབས་སུ་རྫས་སྔགས་སོ་སོའི་མཇུག་ཏུ་ཐམས་ཅད་ཚང་བ་གཅིག་འབུལ་བ་གང་བདེ་ཡིན་པ་ལས། འདིར་རྒྱས་པ་ལྟར་ན། ཨོཾ་ཨགྣ་ཡེ་སྭཱ་ཧཱ། ཨོཾ་བོ་དྷི་བྲྀཀྵཱ་ཡ་སྭཱ་ཧཱ། གཟི་བརྗིད་ཕུན་སུམ་ཚོགས་པ་ལ་བར་དུ་གཅོད་པ་ཐམས་ཅད་ཤཱནྟིཾ་ཀུ་རུ་སྭཱ་ཧཱ། ཞེས་པ་གཅིག་དང༌། ཨོཾ་བོ་དྷི་བྲྀཀྵ་ཡ་སྭཱ་ཧཱ། བཅུ་དགུ། སླར་གོང་མ་ལྟ་བུ་གཅིག་སྟེ་ཉེར་གཅིག་གམ་བདུན་ལ་སོགས་པ་འབུལ། དེ་བཞིན་འགྲེས་ཤེས་པར་བྱས་ལ། ཨོཾ་ཨགྣ་ཡེ་སྭཱ་ཧཱ། འབྱོར་པ་ཕུན་སུམ་ཚོགས་པ་ལ་བར་དུ་གཅོད་པ་ཐམས་ཅད་ཤཱནྟིཾ་ཀུ་རུ་སྭཱ་ཧཱ། མར། ཨོཾ་སརྦ་པཱ་པཾ་ད་ཧ་ན་བཛྲ་ཡ་སྭཱ་ཧཱ། དགེ་བ་ཕུན་སུམ་ཚོགས་པ་ལ་བར་དུ་གཅོད་པ་ཐམས་ཅད༴ ཏིལ། ཨོཾ་སརྦ་སམྤཱ་དེ་སྭཱ་ཧཱ། བདེ་བ་ཕུན་སུམ་ཚོགས་པ་ལ༴ ཟས་མཆོག ཨོཾ་བཛྲ་པུཥྚ་ཡེ་སྭཱ་ཧཱ། བསོད་ནམས་རྒྱས་པ་ལ༴ འབྲས། ཨོཾ་བཛྲ་བཱི་ཛཱ་ཡ་སྭཱ་ཧཱ། ནོར་དང་འབྲུ་རྒྱས་པ་ལ༴ སོ་བ། ཨོཾ་མ་ཧཱ་བེ་གཱ་ཡ་སྭཱ་ཧཱ། ཤུགས་ཕུན་སུམ་ཚོགས་པ་ལ༴ སྲན་མ། ཨོཾ་མ་ཧཱ་བ་ལཱ་ཡ་སྭཱ་ཧཱ། སྟོབས་ཕུན་སུམ་ཚོགས་པ་ལ༴

ནས། ཨོཾ་བཛྲ་གྷསྨ་རཱ་ཡ་སྭཱ་ཧཱ། ནད་མེད་པ་ཕུན་སུམ་ཚོགས་པ་ལ༴ གྲོ། ཨོཾ་བཛྲ་ཨཱ་ཡུ་ཥེ་སྭཱ་ཧཱ། ཚེ་ཕུན་སུམ་ཚོགས་པ་ལ༴ དཱུརྦ། ཨོཾ་ཨ་པྲ་ཏི་ཧ་ཏ་བཛྲཱ་ཡ་སྭཱ་ཧཱ། སྲུང་བ་ཕུན་སུམ་ཚོགས་པ་ལ་བར་དུ་གཅོད་པ་ཐམས་ཅད་ཤཱནྟིཾ་ཀུ་རུ་སྭཱ་ཧཱ། ཞེས་པས་ཀུ་ཤ་རྣམས་བདུན་ནམ་ཉེར་གཅིག་འབུལ་བའི་མཐར་སྔར་བཞིན་མཆོད། ཨོཾ་ཨརྒྷཾ་སྭཱ་ཧཱ། ཨོཾ་པཱ་དྱཾ་སྭཱ་ཧཱ། ཞེས་པས་མཆོད་ཡོན་དང་ཞབས་བསིལ་ཕུལ། མར་དགང་གཟར་གང་ཐབ་ཏུ་བླུགས་ལ། ཨོཾ་བཛྲ་ཛྭ་ལ་རཾ། མེ་ལྷ་འབར་བའི་རྣམ་པར་གྱུར། ཅེས་བརྗོད་དོ། །འཇིག་རྟེན་པའི་མེ་ལྷ་མཆོད་པའོ། །དེ་ནས་བསང་ཆུ་ཡཀྵས་བསང༌། སྭ་བྷཱ་ཝས་སྦྱང༌། སྟོང་པའི་ངང་ལས་ཐབ་ཀྱི་དབུས་སུ་པྃ་ལས་པདྨ་དཀར་པོ་འདབ་མ་བརྒྱད་པའི་ལྟེ་བར་ཞེས་པ་ནས། བཛྲ་ས་མཱ་ཛཿ། ཞེས་པའི་བར་མཁན་ཆེན་ཐམས་ཅད་མཁྱེན་པའི་ཞལ་སྔ་ནས་མཛད་པའི་སྒྲུབ་ཐབས་ལྟར་བྱས་ལ། རྡོ་རྗེ་དྲིལ་བུ་སྤོས་བཅས་འཛིན་པས། དམིགས་མེད་མཁའ་ལྟར་དག་ཅིང་ཁྱབ་པའི་སྐུ། །ཆོས་སྐུ་ནམ་མཁའ་བཞིན་དུ་དབྱེར་མེད་ཀྱང༌། །ཐབས་དང་ཤེས་རབ་གཟུགས་བརྙན་སྐུར་སྟོན་པའི། །བཅོམ་ལྡན་འཁོར་བཅས་གནས་འདིར་གཤེགས་སུ་གསོལ། །ཞེས་དང༌། པོ་ཏ་ལ་ཡི་གནས་མཆོག་ནས། །སོགས་ཀྱི་མཐར། ཨཱརྻ་ཏཱ་རེ་ས་པ་རི་ཝཱ་ར་ཨེ་ཧྱེ་ཧི། བཛྲ་ས་མཱ་ཛཿ ཛཿཧཱུྃ་བཾ་ཧོཿ ཞེས་པས་སྤྱན་དྲངས་ལ། ཨཱརྻ་ཏཱརེ་ས་པ་རི་ཝཱ་ར༴ སྔོན་དུ་འགྲོ་བས། ཨརྒྷཾ་ནས་ཤབྡ་ཨཱཿཧཱུྃ་གིས་མཆོད། ཨོཾ་བཛྲ་ཨངྐུ་ཤ་ཛཿ། སོགས་ཀྱིས་དགུག་གཞུག་བྱ། སླར་ཡང་འོད་འཕྲོས་ཞེས་པ་ནས། འོད་དཔག་མེད་ཀྱིས་དབུར་བརྒྱན། ཞེས་པའི་བར་སྒྲུབ་ཐབས་བཞིན་བརྗོད། ལྷའི་ཞལ་ཅུང་ཟད་གདངས་པར་བསམ། ཡི་གེ་བཅུ་པ་ལན་གཅིག་དང༌། ཨོཾ་ཨགྣ་ཡེ་སྭཱ་ཧཱ། ལན་བདུན་གྱིས་བླུགས་གཟར་བཀང་བའི་གཟར་བུ་གཉིས་ཁ་སྦྱར་ལ་ཐབ་ལ་གཡས་སུ་བསྐོར་ཞིང༌། ཨོཾ་ཏཱ་རེ་ཏུཏྟཱ་རེ་ཏུ་རེ་སྭཱ་ཧཱ། ཨོཾ་ཨགྣ་ཡེ་ཨཱ་དཱི་བྱ་ཨཱ་དཱི་བྱ་བི་ཤུདྡྷེ་མ་ཧཱ་ཤྲི་ཡེ་ཧ་བྱ་ཀ་བྱ་བཧྣ་ཡ་སྭཱ་ཧཱ།

བདག་ཅག་དཔོན་སློབ་ཡོན་མཆོད་རྣམས་ཀྱི་འགལ་རྐྱེན་མི་དང་མི་མ་ཡིན་པའི་ཉེ་བར་འཚེ་བ་དང༌། དུས་མ་ཡིན་པའི་འཆི་བ་ཐམས་ཅད་ཤཱནྟིཾ་ཀུ་རུ་སྭཱ་ཧཱ། ཞེས་པ་ལན་གསུམ་གྱིས་བྱོན་ཟན་འབུལ། སྤྲོ་ན། དྲི་མེད་གཙང་ཞིང་ཡན་ལག་བརྒྱད་ལྡན་ཆུ། །ཞེས་སོགས་ཚིགས་བཅད་སྔགས་དང་བཅས་པའམ། བསྡུ་ན། ཨཱརྻ་ཏཱ་རེ་ས་པ་རི་ཝཱ་ར་སྔོན་དུ་འགྲོ་བའི། ཨརྒྷཾ་པྲ་ཏཱིཙྪ་སྭཱ་ཧཱ། ནས། ཤབྡ་པྲ་ཏཱིཙྪ་སྭཱ་ཧཱ། བར་རོལ་ཆེན་དང་བཅས་པས་མཆོད། རྡོ་རྗེ་ཕྱག་འཚལ་བའི་སྟངས་ཀས་བཟུང༌། དྲིལ་བུ་ཚིགས་བཅད་ཀྱི་མཚམས་སུ་གསིལ་ཞིང༌། འཁོར་བ་ལས་སྒྲོལ་ཏཱ་རེ་མ། །སོགས་དང༌། སྤྲོ་ན། ཕྱག་འཚལ་ཉེར་གཅིག་དང༌། ལྷ་དང་ལྷ་མིན་ཅོད་པན་གྱིས། །ཞེས་པ་ནས། བཀྲ་ཤིས་དཔལ་འབར་མ་ལ་ཕྱག་འཚལ་ལོ། །ཞེས་པའི་བར་གང་ཤེས་པས་བསྟོད། ཡམ་ཤིང་ནས་ཀུ་ཤའི་བར་བརྒྱའམ་ཉེར་གཅིག་ལ་སོགས་པ་གང་འབྱོར་མེ་ལྷའི་སྐབས་སུ་བཤད་པ་ཇི་ལྟ་བ་བཞིན་ལ། ལྷ་སྔགས་སྤེལ་བས་དབུལ་བའི་མཐར། བསྙེན་པ་ཁ་སྐོང་ཡིན་ན་སྡིག་པ་ཞི་ཕྱིར་ཡིན་པས་ཏིལ་ཁོ་ནར་གྲངས་བཟུང་སྟེ། དེའང་བསྙེན་པ་ཁ་གསོའི་གྲངས་ལྷ་སྔགས་ཁོ་ནར་འཛིན་པ་མུས་ཆེན་འཁོར་ལོ་སྡོམ་པས་བདག་ཆེན་རྡོ་རྗེ་འཆང་ལ་ཆེད་དུ་གདམས་པའི་བཞེད་པ་ཡིན་པས། ཡི་གེ་བཅུ་པའམ། ཨོཾ་ཏཱ་རེ་ཏུཏྟཱ་རེ་ཏུ་རེ་མ་མ་ཨཱ་ཡུཿཔུཎྱེ་ཛྙཱ་ན་པུཥྚིཾ་ཀུ་རུ་སྭཱ་ཧཱ། ཞེས་པའི་སྤེལ་ཚིག་ཅན་གཉིས་བསྙེན་པ་གང་ལ་སོང་བ་ལ་གྲངས་བཟུང་སྟེ། ཐུན་རེ་ལ་ཁྲི་ཚོ་རེ་སོགས་ཅི་ནུས་འབུལ་ལོ། །རིམ་གྲོ་ལྟར་བྱེད་ན་སློབ་དཔོན་ངག་དབང་གྲགས་པས་མཛད་པའི་འཆི་བ་སླུ་བའི་གཞུང་ལས། དང་པོར་རྣམ་པར་སྣང་མཛད་བརྗོད། །དེའི་འོག་ཏཱ་རེ་ལན་གཉིས་བརྗོད། །ཏུཏྟཱ་རེ་གཉིས་ཏུ་རེ་གཅིག །སྭཱ་ཧཱའི་མཐའ་ཅན་འཆི་བ་སླུ། །སྒྲོལ་མ་འབྱུང་བའི་རྒྱུད་ལས་གསུངས། །སྔགས་ནི་ཡི་གེ་འབུམ་པ་རུ། །སྒྲོལ་མའི་སྤྱན་སྔར་བཟླས་ནས་ནི། །ཞོ་དང་སྦྲང་རྩིས་ངར་བཏགས་པའི། །དཱུརྦའི་མྱུ་གུ་ཤར་དང་ནི། །བྱང་དུ་རྩེ་བསྟན་སྦྱིན་སྲེག་བྱ། །སྔོན་གྱི་མཐུ་ལས་བྱུང་བ་ཡི། །འཆི་བ་ངེས་པར་ཟློག་པར་བྱེད། །ཅེས་གསུངས་པས།

དཱུརྦའི་མྱུ་གུ་རྩེ་མོ་མ་ཉམས་པའམ། མ་འབྱོར་ན་དུམ་བུར་བྱས་པ་ཞོ་དང་སྦྲང་རྩི་དང་འབྲས་བསྲེས་པ་འབའ་ཞིག་ལྷ་སྔགས་ཁོ་ནས་བསྲེགས་པའི་སྦྱིན་སྲེག་ཁྲི་ལ་སོགས་པ་བྱས་ན་ཆེད་དུ་བསྔགས་པར་མངོན་ནོ། །སླར་བྱོན་ཟན་གྱི་སྐབས་སུ་བཤད་པ་ཇི་ལྟ་བའི་དགང་བླུགས་ལན་གསུམ་གཤེགས་སྐྱེམས་སུ་དབུལ། ཨོཾ་ཨཱརྻ་ཏཱ་རེ་ས་པ་རི་ཝཱ་ར་སྔོན་དུ་འགྲོ་བའི། ཨརྒྷཾ་ནས། ཤབྡ་པྲ་ཏཱིཙྪ་སྭཱ་ཧཱའི་བར་རོལ་ཆེན་དང་བཅས་པས་མཆོད། འཁོར་བ་ལས་སྒྲོལ་ཏཱ་རེ་མ། །སོགས་ཀྱིས་བསྟོད། གཏོར་མ་ཡཀྵས་བསང༌། སྭ་བྷཱ་ཝས་སྦྱང༌། ཨོཾ་ཨ་ཀཱ་རོ་མུ་ཁཾ་སརྦ་དྷརྨཱ་ཎཱ་ཨཱ་དྱ་ནུཏྤནྣ་ཏྭ་ཏ྄་ཨོཾ་ཨཱཿཧཱུྃ་ཕཊ། ལན་གསུམ་གྱི་བྱིན་གྱིས་བརླབ། ནམ་མཁའ་མཛོད་ཀྱི་ཕྱག་རྒྱའི་མཐར་སེ་གོལ་གཏོགས་པ་དང་བཅས། ན་མཿས་མནྟ་བུདྡྷཱ་ནཱཾ་ཨ་པྲ་ཏི་ཧ་ཏ་ཤཱ་ས་ནཱ་ནཱཾ་ཧེ་ཧེ་བྷ་ག་ཝ་ཏཱི་མ་ཧཱ་ས་ཏྭ་སརྦ་བུདྡྷ་ཨ་ཝ་ལོ་ཀི་ཏེ་མཱ་བི་ལམྦ་མཱ་བི་ལམྦ་ཨི་དཾ་བ་ལིཾ་གྲྀཧྞ་པ་ཡ་གྲྀཧྞཱ་པ་ཡ་ཧཱུྃ་ཧཱུྃ་སརྦ་བི་སཉྩ་རེ་སྭཱ་ཧཱ། ལན་གསུམ་གྱིས་དབུལ། ཨཱརྻ་ཏཱ་རེ་ས་པ་རི་ཝཱ་ར་ཨརྒྷཾ་ནས། ཤབྡ་ཨཱཿཧཱུྃ་གིས་མཆོད། ལྷ་དང་ལྷ་མིན་ཅོད་པན་གྱིས། །སོགས་ནས། སྒྲོལ་མ་ཡུམ་ལ་ཕྱག་འཚལ་ལོ། །ཞེས་པས་བསྟོད། འདོད་པའི་དོན་ལ་གསོལ་བ་དྲག་ཏུ་བཏབ། ཡི་གེ་བརྒྱ་པ་ཚར་གསུམ་དང༌། མ་འབྱོར་པ་དང་ཉམས་པ་དང༌། སོགས་ཀྱིས་བཟོད་པར་གསོལ། །ཨོཾ་ཁྱེད་ཀྱིས་སེམས་ཅན་དོན་ཀུན་མཛད། །སོགས་ཀྱི་མཐར། བཛྲ་མུཿཐབ་ནང་གི་ཡེ་ཤེས་པ་རང་བཞིན་གྱི་གནས་སུ་གཤེགས། ཐབ་དབུས་ནས་ཐལ་བ་ཅུང་ཟད་བླངས་ཏེ་དཔྲལ་བར་བྱུག་ཅིང༌། དམ་ཚིག་པ་རང་ཉིད་ལ་ཛཿཧཱུྃ་བཾ་ཧོཿ ཞེས་བརྗོད་དེ་འདས་པའི་མེ་ལྷ་མཆོད་པའོ། །སླར་ཐབ་ཁུང་དུ་མར་དགང་གཟར་གང་བླུགས་ལ། ཨོཾ་བཛྲ་ཛྭ་ལ་རཾ། མེ་ལྷ་འབར་བའི་རྣམ་པ་ལས་བཞེངས་པར་གྱུར་པར་བསམས་ལ། ཨོཾ་ཨརྒྷཾ་སྭཱ་ཧཱ། པཱ་དྱཾ་སྭཱ་ཧཱ། ཞེས་པས་བསང་གཏོར་ཞབས་བསིལ་ཕུལ།

ཡམ་ཤིང་ནས་ཀུ་ཤའི་བར་བསྲེག་རྫས་རྣམས་ཇི་ལྟར་འབྱོར་པ་ཕུལ་བའི་མཐར་གོང་བྱོན་ཟན་དུ་བཤད་པ་ལྟར་དགང་བླུགས་ལན་གསུམ་གཤེགས་སྐྱེམས་སུ་དབུལ། གོང་ལྟར་མཆོད་བསྟོད་བྱས་ལ། གཏོར་མ་བསང་སྦྱང་བྱིན་རླབས་སྔར་བཞིན་བྱས་ལ། ཨོཾ་ཨགྣ་ཡེ་ས་པ་རི་ཝཱ་ར་ན་མཿསརྦ་ཏ་ཐཱ་ག་ཏེ་བྷྱོ་བི་ཤྭ་མུ་ཁེ་བྷྱཿ སརྦ་ཐཱ་ཁཾ་ཨུདྒ་ཏེ་སྥ་ར་ཎ་ཨི་མཾ་ག་ག་ན་ཁཾ་གྲྀཧྞེ་དཾ་བ་ལིཾ་ཏ་ཡེ་སྭཱ་ཧཱ། ལན་གསུམ་གྱིས་དབུལ། ཨོཾ་ཨགྣ་ཡེ་ས་པ་རི་ཝཱ་ར་ཨརྒྷཾ་ནས་ཤབྡའི་བར་གྱིས་མཆོད། མེ་སྟེ་བསྲེགས་བྱ་ཟ་བའི་ལྷ། །དྲང་སྲོང་རྒྱལ་པོ་གདོན་གྱི་བདག །ཤར་ལྷོ་གདོན་གྱི་ཚོགས་བཅས་ལ། །མཆོད་ཅིང་བསྟོད་དེ་ཕྱག་ཀྱང་འཚལ། །མེ་ལྷའི་རྒྱལ་པོ་འཁོར་བཅས་རྣམས། །མཆོད་སྦྱིན་གཏོར་མ་འདི་བཞེས་ལ། །ཅེས་པ་ནས། ཡིད་ལ་འདོད་པ་ཀུན་འགྲུབ་ཤོག །བར་གྱིས་ཕྲིན་ལས་བཅོལ། འདོད་པའི་དོན་ལ་གསོལ་བ་གདབ། ཡིག་བརྒྱ་ཚར་གསུམ་དང་མ་འབྱོར་པ་སོགས་ཀྱིས་བཟོད་པར་གསོལ། བདག་གི་དོན་དང་གཞན་གྱི་དོན། །བྱིན་ཟ་ཁྱོད་ཀྱིས་བསྒྲུབས་ཀྱིས་གཤེགས། །ཇི་བཞིན་དུས་སུ་ཚུར་བྱོན་ཅིག །བདག་གི་དངོས་གྲུབ་ཐམས་ཅད་སྒྲུབས། །ཨགྣ་ཡེ་མུཿ མེ་ལྷ་རང་བཞིན་གྱི་གནས་སུ་གཤེགས་པར་གྱུར། ཅེས་བརྗོད། གཏོར་མ་ཆ་གསུམ་ངོར་པ་རྡོ་རྗེ་འཆང་གིས་མཛད་པ་ལྟར་བྱའོ། །དེ་ནས་བསྔོ་བ་སྨོན་ལམ་བཀྲ་ཤིས་སོགས་སྐབས་སུ་བབ་ཅིང་འབྲེལ་ཆགས་པ་རྒྱས་པར་མཛད། སྦྱིན་སྲེག་མང་དུ་བྱེད་ན་སླར་ཡང་ཐལ་བ་རྣམས་ཅུང་ཟད་བཅགས་པའི་སྟེང་དུ་ཨོང་བ་བརྩེགས་པ་ནས་བཀྲ་ཤིས་ཀྱི་བར་ཚུལ་སྔ་མ་ཇི་ལྟ་བར་ཉམས་སུ་བླང་བར་བྱའོ། །དེ་ལྟར་རབ་འབྱམས་གྲུབ་པ་སྟོང་གི་མཛོད། །མཐའ་ཡས་བཀྲ་ཤིས་བྱེ་བའི་དཔལ་འབར་བ། །སྦྱིན་སྲེག་ཉི་མའི་འོད་ཟེར་ཤར་བའི་མཐུས། །དགེ་ལེགས་པད་དཀར་རྒྱ་ཆེར་རྒྱས་གྱུར་ཅིག །ཅེས་རྗེ་བཙུན་ཡིད་བཞིན་འཁོར་ལོའི་སྦྱིན་སྲེག་གི་ཆོ་ག་ངག་འདོན་བཀླགས་ཆོག་ཏུ་བཀོད་པ་ཚེ་དང་བསོད་ནམས་ཉི་མ་བདུན་འབར་མ་ཞེས་བྱ་བ་འདི་ཡང༌།

བདེ་བར་གཤེགས་པའི་བསྟན་པ་རིན་པོ་ཆེ་བླ་མར་མཛད་པའི་ས་སྐྱོང་བསྟན་པའི་མཚན་ཅན་གྱིས་བཀའ་ཡིས་བསྐུལ་བའི་ངོར། དཔལ་ཞྭ་ལུའི་གདན་ས་བཅུ་གསུམ་པ་མཁྱེན་བརྩེའི་དབང་ཕྱུག་དཔལ་ལྡན་སྐལ་བ་བཟང་པོའི་སྡེས་ཐམས་ཅད་མཁྱེན་པ་བསོད་ནམས་རྒྱལ་མཚན་གྱི་གསུང་ཇི་ལྟ་བ་ལ་ཕྱག་ལེན་གྱི་རིམ་པ་ཆེར་གསལ་བར་རྔ་ཆེན་གྱི་ལོ་སྣྲོན་ཅན་གྱི་ཟླ་བའི་ཚེས་གཅིག་ལ་དཔལ་ཞྭ་ལུའི་གཙུག་ལག་ཁང་དུ་ལེགས་པར་གྲུབ་པ་འདིས་ཀྱང་རྗེ་བླ་མའི་བསྟན་པ་ཕྱོགས་ཐམས་ཅད་དུ་རྒྱས་ཤིང་བཀྲ་ཤིས་བདེ་ལེགས་མཁའ་ཁྱབ་ཏུ་འབར་ཞིང་བརྟན་པར་གྱུར་ཅིག །སརྦ་མངྒ་ལཾ།། ༈ །སྒྲོལ་དཀར་ཡིད་བཞིན་འཁོར་ལོའི་སྒྲུབ་ཐབས་རྒྱུན་གྱི་ཉམས་ལེན་ཁྱེར་བདེ་བ་བཞུགས་སོ། །སྐྱབས་འགྲོ་སེམས་བསྐྱེད་ཐུན་མོང་དུ། སངས་རྒྱས་ཆོས་དང་ཚོགས་ཀྱི་མཆོག་རྣམས་ལ། །ཞེས་པ་ནས། འགྲོ་ལ་ཕན་ཕྱིར་སངས་རྒྱས་འགྲུབ་པར་ཤོག །ཅེས་པའི་བར་བདུན་ནམ་གསུམ་བྱ། ཨོཾ་སྭ་བྷཱ་ཝ་ཤུདྡྷཿསརྦ་དྷརྨཱཿསྭ་བྷཱ་ཝ་ཤུདྡྷོ྅ཧཾ། སྟོང་པར་གྱུར། སྟོང་པའི་ངང་ལས་པྃ་ལས་པདྨ་དཀར་པོ་འདབ་མ་བརྒྱད་པའི་ལྟེ་བར་ཨ་ལས་ཟླ་བའི་དཀྱིལ་འཁོར་གྱི་སྟེང་དུ། རང་གི་རྣམ་ཤེས་ཀྱི་ངོ་བོ་ཏཱྃ་ཡིག་དཀར་པོ་ལས་འོད་འཕྲོས། འཕགས་པ་མཆོད་དོན་གཉིས་བྱས་ཚུར་འདུས་པ་ལས། རང་ཉིད་རྗེ་བཙུན་བཅོམ་ལྡན་འདས་མ་ཡིད་བཞིན་འཁོར་ལོ་ཚེ་སྦྱིན་མ་སྐུ་མདོག་དཀར་མོ། ཞལ་གཅིག་ཕྱག་གཉིས་མ། ཕྱག་གཡས་མཆོག་སྦྱིན་དང༌། གཡོན་ཐུགས་ཀར་མཐེབ་སྲིན་གྱི་ཕྱག་རྒྱས། དུས་གསུམ་གྱི་སངས་རྒྱས་མཚོན་པའི་ཨུཏྤལ་བསྣམས་པ། ཞབས་རྡོ་རྗེའི་སྐྱིལ་ཀྲུང་གིས་བཞུགས་པ། ཡེ་ཤེས་ཀྱི་སྤྱན་བདུན་དང་ལྡན་པ། དར་དང་རིན་པོ་ཆེས་བརྒྱན་པའི་དཔྲལ་བར་ཨོཾ། མགྲིན་པར་ཨཱཿ ཐུགས་ཀར་ཧཱུྃ། ཐུགས་ཀའི་ཧཱུྃ་ལས་འོད་ཟེར་འཕྲོས་པས་རྗེ་བཙུན་མ་སྒྲོལ་མ་ལ་སངས་རྒྱས་དང་བྱང་ཆུབ་སེམས་དཔའི་ཚོགས་ཀྱིས་བསྐོར་བ་ཨོཾ་བཛྲ་ས་མཱ་ཛཿ ཛཿཧཱུྃ་བཾ་ཧོཿ
