གཏན་གྱི་སྐྱབས་གཅིག་རྗེ་བཙུན་བཅོམ་ལྡན་མར། །

(ཉེར་བཏུད་དེ་མཉེས་བྱེད་པའི་ཆོ་ག་འབྲི། །འདིར་རྗེ་བཙུན་སྒྲོལ་མའི་མཎྜལ་བཞི་པ་བྱ་བར་འདོད་པས་གནས་ཁང་བྱི་དོར་བྱས་པའི་དབུས་སུ་སྟེགས་བུ་ཇི་ལྟར་རིགས་པ་སྟེགས་ཁེབས་སོགས་ཀྱིས་བཀབ་པའི་སྟེང་དུ་རྗེ་བཙུན་མའི་བྲིས་འབུར་གྱི་རྟེན་ཡོད་ན་བཤམ། །དེའི་མདུན་དུ་མཎྜལ་ཚོམ་བུ་བདུན་མ་བཞི་བཤམ། མེད་ན་མཎྜལ་གཅིག་ལ་བསྐྱར་ནས་འབུལ། །གཏོར་མ་ཟླུམ་པོ་བཞི་འམ་གཅིག་པུས་ཀྱང་རུང་། ཡོན་ཆབ།

སྤོས། མར་མེ། ཞལ་ཟས་རྣམས་བཞིའམ་བསྡུ་ན་ཚར་རེ་ལ་ཁ་གསོས་ཀྱང་ཆོག །དེ་ནས་སྟན་བདེ་བ་ལ་འདུག་ནས་ཁ་བཤལ་ཐོག་མར་མཆོད་པ་རྣམས། )ཨོཾ་ཨ་མྲྀ་ཏས་བསང་། སྭ་བྷ་ཝས་སྦྱངས། སྟོང་པའི་ངང་ལས་ཨོཾ་ལས་བྱུང་བའི་རིན་པོ་ཆེ་ལས་གྲུབ་པའི་སྣོད་ཡངས་ཤིང་རྒྱ་ཆེ་བ་རྣམས་ཀྱི་ནང་དུ་ༀ་ཞུ་བ་ལས་བྱུང་བའི་ལྷ་རྫས་ལས་གྲུབ་པའི་མཆོད་རྫས་རྣམས་དྭངས་ཤིང་ཐོགས་པ་མེད་པ་ནམ་མཁའ་གང་བར་གྱུར། ཨོཾ་བཛྲ་ཨནྡྷཾ་ནས། ཤརྒྷཾ་ཨཱཿཧཱུྃ་བར་དང་དྷརྨཱ་ར་ཎི་ཏ་(སོགས་ནས་བརྗོད་ལ་ཏིང་ཤག་དཀྲོལ།) སངས་རྒྱས་ཆོས་ཚོགས་མ་ཚར་གསུམ། རང་གི་སྙིང་ག་ནས་འོད་ཟེར་འཕྲོས་པས་བླ་མ་རྒྱལ་བ་སྲས་དང་བཅས་པ་ཨོཾ་བཛྲ་ས་མ་ཛ༔ སངས་རྒྱས་ཐམས་ཅད་འདུས་པའི་སོགས་བློ་ཀ་གཅིག །མགོན་པོ་ཐུགས་རྗེ་ཆེ་ལྡན་སོགས་བློ་ཀ་གཅིག །དག་པས་འདོད་ཆགས་བྲལ་བའི་སོགས་བློ་ཀ་གཅིག །གྲོལ་ནས་གྲོལ་བའི་སོགས་བློ་ཀ་གཅིག །ཕྱག་བྱར་འོས་པ་སོགས་བློ་ཀ་གཅིག་གིས་ཕྱག་བྱ། ས་གཞི་སྤོས་ཆུས་སོགས་མཇུག་ཏུ། གུ་རུ་ཏྲི་རཏྣ་མཎྜལ་ལ་པཱུ་ཛ་མེ་གྷ་ས་མུ་དྲ་སྤ་ར་ཎ་ས་མ་ཡེ་ཧཱུྃ་ཞེས་པས་མཎྜལ་འབུལ།

མཁྱེན་བརྩེ་ནུས་ལྡན་རྒྱལ་བ་སྲས་བཅས་རྣམས། །

བདག་ལ་བརྩེ་བ་ཆེན་པོས་ཉེར་དགོངས་ནས། །

འགལ་རྐྱེན་བར་ཆད་ཐམས་ཅད་ཞི་བ་དང་། །

བསམ་དོན་ཡིད་བཞིན་འགྲུབ་པར་མཛད་དུ་གསོལ། །(ཞེས་གསོལ་བ་འདེབས་སོ། །དེ་ནས་རྗེ་བཙུན་མ་ལ་མཎྜལ་གསུམ་

འབུལ་བའི་དང་པོ་ནི། )རང་གི་སྙིང་ག་ནས་འོད་ཟེར་འཕྲོས་པས་ལྷོ་ཕྱོགས་པོ་ཏ་ལ་ནས་རྗེ་བཙུན་མ་འཕགས་མ་སྒྲོལ་མའི་ལྷ་ཚོགས་འཁོར་དང་བཅས་པ་ཨོཾ་བཛྲ་ས་མཱ་ཛ༔ པོ་ཏ་ལ་ཡི་གནས་མཆོག་སོགས་བློ་ཀ་གཅིག །ཨོཾ་པདྨ་ཀ་མ་ལ་ཡེ་སྟེ། ལྷ་དང་ལྷ་མིན་སོགས་བློ་ཀ་གཅིག །ཨོཾ་ཨཱརྱ་ཏཱ་རེ་ས་པ་རི་ཝཱ་ར་ཨརྒྷཾ་པཱུ་ཛ་ཨཱཿཧཱུྃ། དེ་བཞིན་དུ་པཱ་དྱཾ་ནས། ཤབྡའི་བར་འདོན་པ་བསྒྱུར། ས་གཞི་སྤོས་ཆུས་སོགས་ནས། གུ་རུ་ཏཱ་རེ་བཛྲ་སོགས་ནས། ཧཱུྃ་གི་བར་བརྗོད། ཕྱག་འཚལ་ཉེར་གཅིག་ཚར་གཉིས་བརྗོད། ཕྱག་འཚལ་བ་དང་མཆོད་ཅིང་བཤགས་པ་དང་སོགས་བློ་ཀ་གཅིག །ཨོཾ་ཏཱ་རེ་ཏུཏྟ་རེ་ཏུ་རེ་སྭཱ་ཧཱ། (ཡི་གེ་བཅུ་པ་ཉེར་གཅིག་བཟླ། ཏཱ་རེ་ཉེར་གཅིག་བགྲང་། དེ་ནས།) རྗེ་བཙུན་བཅོམ་ལྡན་འདས་མ་ཐུགས་རྗེ་ཅན་སོགས་སམ། བསྡུ་ན། འཕགས་མ་སྒྲོལ་མ་བདག་ལ་དགོངས་སུ་གསོལ། །བདག་གི་ཉེར་འཚེ་མ་ལུས་ཀུན་ཞི་ནས། །མཐུན་རྐྱེན་རེ་བའི་འབྲས་བུ་མ་ལུས་པ། །འབད་མེད་ལྷུན་གྱིས་གྲུབ་པར་མཛད་དུ་གསོལ་ཅེས་བརྗོད། །(གཉིས་པ་ནི་ཡང་། ལྷ་དང་ལྷ་མིན་སོགས་བློ་ཀ་གཅིག། མཆོད་པ་དང་། མཎྜལ་བློ་ཀ །ཕྱག་འཚལ་ཉེར་གཅིག་ཚར་གསུམ།) ཕྱག་འཚལ་བ་དང་སོགས། (ཡི་གེ་བཅུ་པ་དང་ཏཱ་རེ་གཉིས་གོང་བཞིན་བགྲང་། འཕགས་མ་སྒྲོལ་མ་སོགས། གསོལ་གདབ་བྱ། གསུམ་པ་ནི།) ལྷ་དང་ལྷ་མིན་སོགས་ཀྱིས་བསྟོད་ནས། ཤབྡའི་བར་གྱིས་མཆོད། ས་གཞི་སྤོས་ཆུས་སོགས་མཎྜལ་དབུལ། ཕྱག་འཚལ་ཉེར་གཅིག་ཚར་བདུན། ཨཱར་ཏཱ་རེ་ཨརྒྷཾ་ནས། ཤབྡའི་བར། ལྷ་མོ་ལ་གུས་སོགས་དང་ཡི་གེ་བཅུ་པ་དང་ཏཱ་རེ་གཉིས་གོང་བཞིན། འཕགས་མ་སྒྲོལ་མ་བདག་ལ་དགོངས་སུ་གསོལ་ཞེས་སོགས་བློ་ཀ་གཅིག་གིས་གསོལ་བ་བཏབ། དེ་ནས་ཨརྒྷཾ་ནས་ཤཔྟའི་བར། ཕྱག་བྱར་འོས་པ་ཐམས་ཅད་ལ་ཞེས་སོགས་བློ་ཀ་གཅིག །ཕྱོགས་བཅུ་ན་བཞུགས་པའི་སངས་རྒྱས་དང་བྱང་ཆུབ་སེམས་དཔའ་ཐམས་ཅད་དང་རྗེ་བཙུན་མ་འཕགས་མ་སྒྲོལ་མའི་ལྷ་ཚོགས་ཐམས་ཅད་བདག་ལ་དགོངས་སུ་གསོལ། ཁྱེད་རྣམས་ལ་མཁྱེན་པའི་ཡེ་ཤེས་བརྩེ་བའི་ཐུགས་རྗེ་སྐྱོབ་པའི་ནུས་མཐུ་བསམ་གྱིས་མི་ཁྱབ་པ་མངའ་བ་ལགས་པས།
